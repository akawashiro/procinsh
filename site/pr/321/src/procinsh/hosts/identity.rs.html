1//! The identity a host advertises to a hub and shows in the host list.
2

4use serde::Serialize;
5use std::fs;
6
7#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
8pub(in crate::http_server) struct HostInfo {
9    pub(in crate::http_server) name: String,
10    /// `/etc/machine-id`, or the boot ID when the machine ID is unavailable.
11    pub(in crate::http_server) machine_id: String,
12    pub(in crate::http_server) version: String,
13    pub(in crate::http_server) kernel: Option<String>,
14    /// `std::env::consts::ARCH` of the procinsh build, such as `x86_64`.
15    pub(in crate::http_server) arch: String,
16}
17
18fn text(path: &str) -> Option<String> {
19    let value = fs::read_to_string(path).ok()?;
20    let value = value.trim();
21    (!value.is_empty()).then(|| value.to_owned())
22}
23
24/// The `?host=` value of the merged view.
25pub(in crate::http_server) const ALL_HOSTS: &str = "all";
26
27/// Host names are short, URL-safe labels; [`ALL_HOSTS`] names the merged view.
28pub(crate) fn valid_host_name(name: &str) -> bool {
29    (1..=63).contains(&name.len())
30        && name
31            .bytes()
32            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-'))
33        && name != ALL_HOSTS
34}
35
36impl HostInfo {
37    /// Replace identifying metadata before a remote host enters the registry.
38    /// The resulting name is also the routing key for requests and SSE events.
39    pub(super) fn protect_remote(&mut self) {
40        if crate::http_server::privacy::enabled() {
41            self.mask_identity(crate::http_server::privacy::pseudonym);
42        }
43    }
44
45    fn mask_identity(&mut self, mask: impl Fn(&str) -> String) {
46        self.name = format!("host-{}", mask(&self.name).replace('░', ""));
47        self.machine_id = mask(&self.machine_id).replace('░', "");
48    }
49
50    /// This machine, named by `name` or the kernel host name. In privacy mode
51    /// the default name and the machine ID are pseudonyms for this run.
52    pub(in crate::http_server) fn local(name: Option<String>) -> Self {
53        let machine_id = text("/etc/machine-id")
54            .or_else(|| text("/proc/sys/kernel/random/boot_id"))
55            .unwrap_or_default();
56        let private = crate::http_server::privacy::enabled();
57        let pseudonym = || crate::http_server::privacy::pseudonym(&machine_id).replace('░', "");
58        Self {
59            name: name
60                .or_else(|| {
61                    if private {
62                        Some(format!("host-{}", pseudonym()))
63                    } else {
64                        text("/proc/sys/kernel/hostname")
65                    }
66                })
67                .unwrap_or_else(|| "localhost".into()),
68            machine_id: if private { pseudonym() } else { machine_id },
69            version: env!("CARGO_PKG_VERSION").into(),
70            kernel: text("/proc/sys/kernel/osrelease"),
71            arch: std::env::consts::ARCH.into(),
72        }
73    }
74}

76#[cfg(test)]
77mod tests {
78    use super::*;
79
80    #[test]
81    fn host_names_are_short_url_safe_and_not_all() {
82        for name in ["node-a", "gpu01.cluster", "a_b", &"x".repeat(63)] {
83            assert!(valid_host_name(name), "{name}");
84        }
85        for name in ["", "all", "a b", "a/b", "a?b", &"x".repeat(64)] {
86            assert!(!valid_host_name(name), "{name}");
87        }
88        let local = HostInfo::local(Some("override".into()));
89        assert_eq!(local.name, "override");
90        assert_eq!(local.version, env!("CARGO_PKG_VERSION"));
91        assert!(!HostInfo::local(None).name.is_empty());
92    }
93}

95#[cfg(test)]
96mod privacy_tests {
97    use super::*;
98
99    #[test]
100    fn remote_identity_is_private_and_remains_routable() {
101        let mut info = HostInfo {
102            name: "private-agent".into(),
103            machine_id: "private-machine-id".into(),
104            version: "v1".into(),
105            kernel: None,
106            arch: "aarch64".into(),
107        };
108        info.mask_identity(|value| match value {
109            "private-agent" => "░12345678░".into(),
110            "private-machine-id" => "░abcdef01░".into(),
111            _ => panic!("unexpected identity"),
112        });
113        assert_eq!(info.name, "host-12345678");
114        assert!(valid_host_name(&info.name));
115        assert_eq!(info.machine_id, "abcdef01");
116        assert_eq!(info.version, "v1");
117        let json = serde_json::to_string(&info).unwrap();
118        assert!(!json.contains("private-agent"));
119        assert!(!json.contains("private-machine-id"));
120        let registry =
121            super::super::registry::HostRegistry::new(HostInfo::local(Some("hub".into())));
122        let link = std::sync::Arc::new(super::super::link::AgentLink::new().0);
123        let id = registry.admit(info.clone(), link).unwrap();
124        assert!(matches!(
125            registry.remote(&info.name),
126            super::super::registry::RemoteHost::Online(_)
127        ));
128        assert_eq!(registry.online()[0].0, info.name);
129        assert_eq!(registry.summaries()[1].machine_id, info.machine_id);
130        assert!(registry.is_current(&info.name, id));
131        registry.disconnected(&info.name, id);
132        assert!(matches!(
133            registry.remote(&info.name),
134            super::super::registry::RemoteHost::Offline
135        ));
136    }
