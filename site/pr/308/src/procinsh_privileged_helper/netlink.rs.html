1//! Bounded netlink dump requests and attribute parsing shared by socket and RDMA
2//! diagnostics. Requests only read kernel tables; no endpoint is opened or modified.
3//!
4//! # Interface
5//!
6//! | Definition | Visibility | Kind / signature |
7//! | --- | --- | --- |

13use anyhow::{Context, Result, bail, ensure};
14use std::{
15    os::fd::{AsRawFd, FromRawFd, OwnedFd},
16    time::Instant,
17};
18
19const HEADER: usize = 16;
20const NLMSG_ERROR: u16 = 2;
21const NLMSG_DONE: u16 = 3;
22const NLM_F_DUMP_INTR: u16 = 0x10;
23// NLM_F_REQUEST | NLM_F_DUMP
24const DUMP_FLAGS: u16 = 0x301;
25const REPLY_LIMIT: usize = 8 * 1024 * 1024;
26
27/// The kernel's negative errno from an error or done message.
28fn kernel_error(code: i32) -> std::io::Error {
29    std::io::Error::from_raw_os_error(code.saturating_neg())
30}
31
32pub(super) fn open_socket(protocol: i32, name: &str) -> Result<OwnedFd> {
33    let raw = unsafe {
34        libc::socket(
35            libc::AF_NETLINK,
36            libc::SOCK_RAW | libc::SOCK_CLOEXEC | libc::SOCK_NONBLOCK,
37            protocol,
38        )
39    };
40    ensure!(raw >= 0, "{name}: {}", std::io::Error::last_os_error());
41    Ok(unsafe { OwnedFd::from_raw_fd(raw) })
42}
43
44pub(super) fn u32_at(bytes: &[u8], offset: usize) -> Result<u32> {
45    Ok(u32::from_ne_bytes(
46        bytes
47            .get(offset..offset + 4)
48            .context("short netlink field")?
49            .try_into()?,
50    ))
51}

53/// Splits a 4-byte aligned attribute stream into `(type, payload)` pairs.
54/// The nested and byte-order flags are removed from the type.
55pub(super) fn attributes(bytes: &[u8]) -> Result<Vec<(u16, &[u8])>> {
56    let mut result = Vec::new();
57    let mut offset = 0;
58    while offset < bytes.len() {
59        ensure!(offset + 4 <= bytes.len(), "short netlink attribute");
60        let length = u16::from_ne_bytes(bytes[offset..offset + 2].try_into()?) as usize;
61        let kind = u16::from_ne_bytes(bytes[offset + 2..offset + 4].try_into()?) & 0x3fff;
62        ensure!(
63            length >= 4 && offset + length <= bytes.len(),
64            "invalid netlink attribute length"
65        );
66        result.push((kind, &bytes[offset + 4..offset + length]));
67        offset += (length + 3) & !3;
68    }
69    Ok(result)
70}

72/// One dump request. `label` prefixes errors, such as `"UNIX diag timed out"`.
73pub(super) struct NetlinkDumpRequest<'a> {
74    pub(super) label: &'a str,
75    pub(super) message_type: u16,
76    /// The only data message type accepted in the reply.
77    pub(super) reply_type: u16,
78    pub(super) sequence: u32,
79    /// Family header and attributes following the netlink header, 4-byte aligned.
80    pub(super) payload: &'a [u8],
81}
82
83impl NetlinkDumpRequest<'_> {
84    /// Sends the request and passes each reply body to `visit` until `NLMSG_DONE`.
85    /// Interrupted dumps, replies from other senders or sequences, error messages and
86    /// replies above 8 MiB fail. The deadline bounds the whole exchange.
87    pub(super) fn run(
88        &self,
89        fd: &OwnedFd,
90        deadline: Instant,
91        mut visit: impl FnMut(&[u8]) -> Result<()>,
92    ) -> Result<()> {
93        let label = self.label;
94        ensure!(Instant::now() < deadline, "{label} timed out");
95        let mut kernel: libc::sockaddr_nl = unsafe { std::mem::zeroed() };
96        kernel.nl_family = libc::AF_NETLINK as u16;
97        let mut request = vec![0u8; HEADER];
98        request[0..4].copy_from_slice(&((HEADER + self.payload.len()) as u32).to_ne_bytes());
99        request[4..6].copy_from_slice(&self.message_type.to_ne_bytes());
100        request[6..8].copy_from_slice(&DUMP_FLAGS.to_ne_bytes());
101        request[8..12].copy_from_slice(&self.sequence.to_ne_bytes());
102        request.extend_from_slice(self.payload);
103        let sent = unsafe {
104            libc::sendto(
105                fd.as_raw_fd(),
106                request.as_ptr().cast(),
107                request.len(),
108                0,
109                (&kernel as *const libc::sockaddr_nl).cast(),
110                std::mem::size_of_val(&kernel) as _,
111            )
112        };
113        ensure!(
114            sent == request.len() as isize,
115            "{label} send: {}",
116            std::io::Error::last_os_error()
117        );
118        let mut buffer = vec![0u8; 65536];
119        let mut total = 0;
120        loop {
121            ensure!(Instant::now() < deadline, "{label} timed out");
122            let mut poll = libc::pollfd {
123                fd: fd.as_raw_fd(),
124                events: libc::POLLIN,
125                revents: 0,
126            };
127            let remaining = deadline.saturating_duration_since(Instant::now());
128            let timeout = remaining.as_millis().clamp(1, 50) as i32;
129            let ready = unsafe { libc::poll(&mut poll, 1, timeout) };
130            if ready < 0 {
131                if std::io::Error::last_os_error().raw_os_error() == Some(libc::EINTR) {
132                    continue;
133                }
134                bail!(std::io::Error::last_os_error());
135            }
136            if ready == 0 {
137                continue;
138            }
139            let mut sender: libc::sockaddr_nl = unsafe { std::mem::zeroed() };
140            let mut sender_len = std::mem::size_of_val(&sender) as libc::socklen_t;
141            let count = unsafe {
142                libc::recvfrom(
143                    fd.as_raw_fd(),
144                    buffer.as_mut_ptr().cast(),
145                    buffer.len(),
146                    libc::MSG_TRUNC | libc::MSG_DONTWAIT,
147                    (&mut sender as *mut libc::sockaddr_nl).cast(),
148                    &mut sender_len,
149                )
150            };
151            if count < 0 {
152                let error = std::io::Error::last_os_error();
153                if matches!(error.raw_os_error(), Some(libc::EINTR) | Some(libc::EAGAIN)) {
154                    continue;
155                }
156                return Err(error.into());
157            }
158            ensure!(
159                count > 0 && count as usize <= buffer.len(),
160                "truncated {label} packet"
161            );
162            ensure!(sender.nl_pid == 0, "{label} sender is not kernel");
163            total += count as usize;
164            ensure!(total <= REPLY_LIMIT, "{label} exceeds 8 MiB limit");
165            let bytes = &buffer[..count as usize];
166            let mut offset = 0;
167            while offset < bytes.len() {
168                ensure!(offset + HEADER <= bytes.len(), "short netlink header");
169                let length = u32_at(bytes, offset)? as usize;
170                ensure!(
171                    length >= HEADER && length <= bytes.len() - offset,
172                    "invalid netlink length"
173                );
174                let kind = u16::from_ne_bytes(bytes[offset + 4..offset + 6].try_into()?);
175                let flags = u16::from_ne_bytes(bytes[offset + 6..offset + 8].try_into()?);
176                ensure!(
177                    u32_at(bytes, offset + 8)? == self.sequence && flags & NLM_F_DUMP_INTR == 0,
178                    "{label} dump interrupted"
179                );
180                let body = &bytes[offset + HEADER..offset + length];
181                match kind {
182                    // Both carry a negative errno; keep it as an io::Error so callers
183                    // can tell an unsupported dump from a failed one.
184                    NLMSG_DONE => {
185                        if body.len() >= 4 {
186                            let code = u32_at(body, 0)? as i32;
187                            if code != 0 {
188                                return Err(kernel_error(code))
189                                    .with_context(|| format!("{label} dump failed"));
190                            }
191                        }
192                        return Ok(());
193                    }
194                    NLMSG_ERROR => {
195                        let code = u32_at(body, 0)? as i32;
196                        if code != 0 {
197                            return Err(kernel_error(code)).with_context(|| label.to_owned());
198                        }
199                    }
200                    kind if kind == self.reply_type => visit(body)?,
201                    _ => bail!("unexpected netlink message {kind}"),
202                }
203                offset += (length + 3) & !3;
204            }
205        }
206    }
207}

209#[cfg(test)]
210mod tests {
211    use super::*;
212
213    fn attribute(kind: u16, data: &[u8]) -> Vec<u8> {
214        let mut bytes = Vec::new();
215        bytes.extend_from_slice(&((4 + data.len()) as u16).to_ne_bytes());
216        bytes.extend_from_slice(&kind.to_ne_bytes());
217        bytes.extend_from_slice(data);
218        bytes.resize((bytes.len() + 3) & !3, 0);
219        bytes
220    }
221
222    #[test]
223    fn attributes_strip_flags_skip_padding_and_reject_bad_lengths() {
224        let mut bytes = attribute(0x8000 | 19, &attribute(20, &[1, 2, 3]));
225        bytes.extend(attribute(2, b"mlx\0"));
226        let parsed = attributes(&bytes).unwrap();
227        assert_eq!(parsed.len(), 2);
228        assert_eq!(parsed[0].0, 19);
229        assert_eq!(attributes(parsed[0].1).unwrap()[0], (20, &[1u8, 2, 3][..]));
230        assert_eq!(parsed[1], (2, &b"mlx\0"[..]));
231        bytes[0] = 3;
232        assert!(attributes(&bytes).is_err());
233        assert!(attributes(&[8, 0, 1]).is_err());
234        assert!(attributes(&[]).unwrap().is_empty());
235    }
