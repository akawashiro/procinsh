1//! Closed monitor operations and structured replies. No request accepts a path,
2//! arbitrary process-memory read, syscall number, file descriptor, or caller-selected limit.
3//! Mapped ELF requests select an existing executable mapping by its start address;
4//! the helper independently verifies the process, mapped file identity and file ranges.
5//!
6//! Frames have a four-byte native-endian length and a native binary payload.
7//! Requests are at most 1 KiB,
8//! responses at most 1 GiB, and at most 64 requests may queue in the application.
9//! Socket I/O and application requests have 30-second deadlines. Each side
10//! treats an invalid/truncated frame or a timeout as a fatal connection error.
11//! Sessions expire after 15 seconds without polling and are capped at 32 per
12//! kind. Event polling never blocks waiting for a new observation.
13//!
14//! # Interface
15//!
16//! | Definition | Visibility | Kind / type |
17//! | --- | --- | --- |
18//! | [`MAX_REQUEST`] | `pub` | `const MAX_REQUEST: usize` |
19//! | [`MAX_RESPONSE`] | `pub` | `const MAX_RESPONSE: usize` |
20//! | [`TIMEOUT`] | `pub` | `const TIMEOUT: Duration` |
21//! | [`SESSION_LEASE`] | `pub` | `const SESSION_LEASE: Duration` |
22//! | [`ProcessId`] | `pub` | `struct ProcessId` |
23//! | [`HelperIdentity`] | `pub` | `struct HelperIdentity` |
24//! | [`ProcessResource`] | `pub` | `enum ProcessResource` |
25//! | [`MonitorRequest`] | `pub` | `enum MonitorRequest` |
26//! | [`MonitorEvent`] | `pub` | `enum MonitorEvent` |
27//! | [`AcquiredData`] | `pub` | `enum AcquiredData` |
28//! | [`MonitorResponse`] | `pub` | `enum MonitorResponse` |
29//! | [`WireResponse`] | `pub` | `type WireResponse = Result<`[`MonitorResponse`]`, String>` |
30//!
31//! Target identities use [`ProcessId`]. Acquisition snapshots use
32//! [`SystemSnapshot`](crate::ipc::model::system::SystemSnapshot).
33

36pub const MAX_REQUEST: usize = 1024;

39pub const SESSION_LEASE: Duration = Duration::from_secs(15);

51pub enum MonitorRequest {
52    Processes,
53    Process {
54        id: ProcessId,

56    },
57    MappedElf {
58        id: ProcessId,

63        id: ProcessId,

74pub enum MonitorEvent {
75    Observation(std::sync::Arc<crate::ipc::acquisition::ProcessDetailSnapshot>),
76    AcquiredSnapshot(std::sync::Arc<crate::ipc::model::system::SystemSnapshot>),
77    AcquisitionGap {
78        dropped: u64,
79        snapshot: std::sync::Arc<crate::ipc::model::system::SystemSnapshot>,
80    },
81    Activity(std::sync::Arc<crate::ipc::acquisition::SystemActivity>),
82}
83
84pub enum MonitorResponse {
85    Data(AcquiredData),
86    /// Raw ELF bytes in the native binary payload.
87    Elf(Option<crate::ipc::elf::MappedFileData>),
88    Started {
89        session: u64,
90        initial: MonitorEvent,
91    },
92    Event {
93        event: Option<MonitorEvent>,
94        closed: bool,
95    },
96    Closed,
97}

99/// Errors are data on the wire; neither HTTP status codes nor HTTP syntax are
100/// interpreted in the helper. The HTTP layer chooses the response status.
101pub type WireResponse = Result<MonitorResponse, String>;
