25#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
26pub enum SocketType {
27    Stream,
28    Dgram,
29    Raw,
30    Rdm,
31    Seqpacket,
32    Dccp,
33    Packet,
34    Unknown(u32),
35}
36impl SocketType {
37    pub fn from_code(code: u32) -> Self {
38        match code {
39            1 => Self::Stream,
40            2 => Self::Dgram,
41            3 => Self::Raw,
42            4 => Self::Rdm,
43            5 => Self::Seqpacket,
44            6 => Self::Dccp,
45            10 => Self::Packet,
46            n => Self::Unknown(n),
47        }
48    }
49}
50impl fmt::Display for SocketType {
51    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
52        f.write_str(match self {
53            Self::Stream => "STREAM",
54            Self::Dgram => "DGRAM",
55            Self::Raw => "RAW",
56            Self::Rdm => "RDM",
57            Self::Seqpacket => "SEQPACKET",
58            Self::Dccp => "DCCP",
59            Self::Packet => "PACKET",
60            Self::Unknown(code) => return write!(f, "TYPE {code}"),
61        })
62    }
63}
64#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
65pub enum SocketProtocol {
66    Tcp {
67        family: AddressFamily,
68    },
69    Udp {
70        family: AddressFamily,
71    },
72    Unix {
73        socket_type: SocketType,
74    },
75    /// Linux domain, socket type and protocol numbers outside TCP/UDP/UNIX.
76    /// A missing socket type means that only a procfs table supplied metadata.
77    Other {
78        family: u32,
79        socket_type: Option<SocketType>,
80        protocol: u32,
81    },
82}
83impl SocketProtocol {
84    pub fn is_tcp(self) -> bool {
85        matches!(self, Self::Tcp { .. })
86    }
87
88    pub fn is_tcp_or_udp(self) -> bool {
89        matches!(self, Self::Tcp { .. } | Self::Udp { .. })
90    }
91
92    pub fn from_codes(family: u32, socket_type: u32, protocol: u32) -> Self {
93        let inet = match family as i32 {
94            libc::AF_INET => Some(AddressFamily::Ipv4),
95            libc::AF_INET6 => Some(AddressFamily::Ipv6),
96            _ => None,
97        };
98        match (family as i32, socket_type as i32, protocol as i32) {
99            (_, libc::SOCK_STREAM, libc::IPPROTO_TCP) if inet.is_some() => Self::Tcp {
100                family: inet.unwrap(),
101            },
102            (_, libc::SOCK_DGRAM, libc::IPPROTO_UDP) if inet.is_some() => Self::Udp {
103                family: inet.unwrap(),
104            },
105            (libc::AF_UNIX, _, _) => Self::Unix {
106                socket_type: SocketType::from_code(socket_type),
107            },
108            _ => Self::Other {
109                family,
110                socket_type: Some(SocketType::from_code(socket_type)),
111                protocol,
112            },
113        }
114    }
115}
116impl fmt::Display for SocketProtocol {
117    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
118        match self {
119            Self::Tcp { family } => f.write_str(if *family == AddressFamily::Ipv6 {
120                "TCP6"
121            } else {
122                "TCP"
123            }),
124            Self::Udp { family } => f.write_str(if *family == AddressFamily::Ipv6 {
125                "UDP6"
126            } else {
127                "UDP"
128            }),
129            Self::Unix { socket_type } => write!(f, "UNIX {socket_type}"),
130            Self::Other {
131                family,
132                socket_type,
133                protocol,
134            } => {
135                if *family == libc::AF_NETLINK as u32 {
136                    write!(f, "NETLINK ")?;
137                    match netlink_name(*protocol) {
138                        Some(name) => f.write_str(name),
139                        None => write!(f, "PROTOCOL {protocol}"),
140                    }?;
141                    if let Some(SocketType::Unknown(code)) = socket_type {
142                        write!(f, " TYPE {code}")?;
143                    }
144                    return Ok(());
145                }
146                if matches!(*family as i32, libc::AF_INET | libc::AF_INET6)
147                    && *socket_type == Some(SocketType::Raw)
148                {
149                    write!(
150                        f,
151                        "RAW{} ",
152                        if *family == libc::AF_INET6 as u32 {
153                            "6"
154                        } else {
155                            ""
156                        }
157                    )?;
158                    return match *protocol {
159                        1 => f.write_str("ICMP"),
160                        2 => f.write_str("IGMP"),
161                        6 => f.write_str("TCP"),
162                        17 => f.write_str("UDP"),
163                        58 => f.write_str("ICMPV6"),
164                        _ => write!(f, "PROTOCOL {protocol}"),
165                    };
166                }
167                match *family as i32 {
168                    libc::AF_INET => f.write_str("INET")?,
169                    libc::AF_INET6 => f.write_str("INET6")?,
170                    libc::AF_PACKET => f.write_str("PACKET")?,
171                    libc::AF_VSOCK => f.write_str("VSOCK")?,
172                    libc::AF_BLUETOOTH => f.write_str("BLUETOOTH")?,
173                    libc::AF_CAN => f.write_str("CAN")?,
174                    libc::AF_ALG => f.write_str("ALG")?,
175                    libc::AF_XDP => f.write_str("XDP")?,
176                    _ => write!(f, "AF {family}")?,
177                }
178                if let Some(socket_type) = socket_type {
179                    write!(f, " {socket_type}")?;
180                }
181                if !matches!(*family as i32, libc::AF_PACKET | libc::AF_VSOCK) || *protocol != 0 {
182                    write!(f, " PROTOCOL {protocol}")?;
183                }
184                Ok(())
185            }
186        }
187    }
188}

190fn netlink_name(protocol: u32) -> Option<&'static str> {
191    Some(match protocol {
192        0 => "ROUTE",
193        1 => "UNUSED",
194        2 => "USERSOCK",
195        3 => "FIREWALL",
196        4 => "SOCK_DIAG",
197        5 => "NFLOG",
198        6 => "XFRM",
199        7 => "SELINUX",
200        8 => "ISCSI",
201        9 => "AUDIT",
202        10 => "FIB_LOOKUP",
203        11 => "CONNECTOR",
204        12 => "NETFILTER",
205        13 => "IP6_FW",
206        14 => "DNRTMSG",
207        15 => "KOBJECT_UEVENT",
208        16 => "GENERIC",
209        18 => "SCSITRANSPORT",
210        19 => "ECRYPTFS",
211        20 => "RDMA",
212        21 => "CRYPTO",
213        22 => "SMC",
214        _ => return None,
215    })

230#[derive(Clone, Copy, Debug, PartialEq, Eq)]
231pub enum SocketState {
232    Established,
233    SynSent,
234    SynRecv,
235    FinWait1,
236    FinWait2,
237    TimeWait,
238    Close,
239    CloseWait,
240    LastAck,
241    Listen,
242    Closing,
243    NewSynRecv,
244    Unconnected,
245    Connecting,
246    Connected,
247    Disconnecting,
248    UnknownInet(u32),
249    UnknownUnix(u32),
250}
251impl SocketState {
252    pub fn inet(code: u32) -> Self {
253        match code {
254            1 => Self::Established,
255            2 => Self::SynSent,
256            3 => Self::SynRecv,
257            4 => Self::FinWait1,
258            5 => Self::FinWait2,
259            6 => Self::TimeWait,
260            7 => Self::Close,
261            8 => Self::CloseWait,
262            9 => Self::LastAck,
263            10 => Self::Listen,
264            11 => Self::Closing,
265            12 => Self::NewSynRecv,
266            n => Self::UnknownInet(n),
267        }
268    }
269
270    pub fn unix_proc(code: u32, listening: bool) -> Self {
271        if listening {
272            Self::Listen
273        } else {
274            match code {
275                1 => Self::Unconnected,
276                2 => Self::Connecting,
277                3 => Self::Connected,
278                4 => Self::Disconnecting,
279                n => Self::UnknownUnix(n),
280            }
281        }
282    }
283
284    pub fn unix_diag(code: u32) -> Self {
285        match code {
286            1 => Self::Connected,
287            10 => Self::Listen,
288            7 => Self::Unconnected,
289            n => Self::UnknownUnix(n),
290        }
291    }
292}
293impl fmt::Display for SocketState {
294    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
295        let text = match self {
296            Self::Established => "ESTABLISHED",
297            Self::SynSent => "SYN_SENT",
298            Self::SynRecv => "SYN_RECV",
299            Self::FinWait1 => "FIN_WAIT1",
300            Self::FinWait2 => "FIN_WAIT2",
301            Self::TimeWait => "TIME_WAIT",
302            Self::Close => "CLOSE",
303            Self::CloseWait => "CLOSE_WAIT",
304            Self::LastAck => "LAST_ACK",
305            Self::Listen => "LISTEN",
306            Self::Closing => "CLOSING",
307            Self::NewSynRecv => "NEW_SYN_RECV",
308            Self::Unconnected => "UNCONNECTED",
309            Self::Connecting => "CONNECTING",
310            Self::Connected => "CONNECTED",

312            Self::UnknownInet(n) | Self::UnknownUnix(n) => return write!(f, "{n:02X}"),

314        f.write_str(text)
315    }

332#[derive(Clone, Copy, Debug, PartialEq, Eq)]
333pub enum FdAccess {
334    Read,
335    Write,
336    ReadWrite,
337    Unknown,
338}
339impl FdAccess {
340    pub fn from_flags(flags: u32) -> Self {
341        if flags == u32::MAX || flags & libc::O_PATH as u32 != 0 {
342            Self::Unknown
343        } else {
344            Self::from_mode(Some(flags & libc::O_ACCMODE as u32))
345        }
346    }
347
348    pub fn from_mode(mode: Option<u32>) -> Self {
349        match mode {
350            Some(0) => Self::Read,
351            Some(1) => Self::Write,
352            Some(2) => Self::ReadWrite,
353            _ => Self::Unknown,
354        }
355    }
356
357    pub fn opposite(self, other: Self) -> bool {
358        matches!(
359            (self, other),
360            (Self::Read, Self::Write | Self::ReadWrite)
361                | (Self::Write, Self::Read | Self::ReadWrite)
362                | (Self::ReadWrite, Self::Read | Self::Write | Self::ReadWrite)
363        )
364    }
365}
366impl fmt::Display for FdAccess {
367    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
368        f.write_str(match self {
369            Self::Read => "read",
370            Self::Write => "write",
371            Self::ReadWrite => "read/write",
372            Self::Unknown => "N/A",
373        })
374    }
375}
