19#[derive(Clone, Debug, PartialEq)]
20pub struct HardwareDevice {
21    /// A PCI function address such as `0000:85:00.0`, or a host bridge such as `pci0000:80`.
22    pub address: String,
23    pub parent: Option<String>,
24    pub role: HardwareRole,
25    pub kind: Option<HardwareKind>,
26    pub label: String,
27    /// The 24-bit class code as `0x030000`.
28    pub class: Option<String>,
29    pub vendor_id: Option<String>,
30    pub device_id: Option<String>,
31    pub driver: Option<String>,
32    pub numa_node: Option<u32>,
33    /// Null for host bridges and runtime-suspended devices.
34    pub link: Option<PcieLinkInfo>,
35    pub power_state: Option<String>,
36    pub runtime_status: Option<String>,
37    pub iommu_group: Option<u32>,
38    pub bars: Vec<BarInfo>,
39    pub gpu: Option<GpuInfo>,
40    pub rdma: Vec<RdmaDeviceInfo>,
41    /// Network interfaces of the device.
42    pub netdevs: Vec<String>,
43}

52#[derive(Clone, Debug, PartialEq)]
53pub struct GpuInfo {
54    /// `nvidia`, `amd`, `intel` or the PCI vendor ID.
55    pub vendor: String,
56    pub model: Option<String>,
57    pub uuid: Option<String>,
58    /// The NVIDIA device minor of `/dev/nvidiaN`.
59    pub minor: Option<u32>,
60    /// DRM nodes such as `card1` and `renderD128`.
61    pub drm: Vec<String>,
62}

64#[derive(Clone, Debug, PartialEq, Eq)]
65pub struct DeviceHandle {
66    pub id: String,
67    pub process_id: ProcessId,
68    pub device: String,
69    pub kind: DeviceHandleKind,

75#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
76pub enum DeviceHandleKind {

82    Uverbs,
83    /// `/dev/infiniband/umadN`.
84    Umad,

89#[derive(Clone, Debug, PartialEq)]
90pub struct HardwarePath {
91    pub id: String,
92    pub a: String,
93    pub b: String,
94    pub class: PathClass,
95    /// Nodes between the ends, in order from `a` to `b`.
96    pub via: Vec<String>,
97    /// The slowest negotiated link on the path; null when no link speed is known.
98    pub bottleneck: Option<PcieLink>,
99}
100
101#[derive(Clone, Copy, Debug, PartialEq, Eq)]
102pub enum PathClass {
103    /// At most one PCIe switch (or functions of one device).
104    Pix,
105    /// Several PCIe switches without the host bridge.
106    Pxb,
107    /// Through the host bridge of one root complex.
108    Phb,
109    /// Between host bridges of one NUMA node.
110    Node,
111    /// Between NUMA nodes.
112    Sys,
113}
