43/// Adds IPoIB remote GIDs to network destinations of the collector's own network
44/// namespace. Neighbors are read only when an IPoIB interface exists.
45fn add_remote_gids(
46    result: &mut SystemSnapshot,
47    _inode_ns: &HashMap<u64, (u64, u64)>,
48    sources: &HardwareSources,
49) {
50    if !super::ipoib::has_ipoib_interface(&sources.sys) {
51        return;
52    }
53    let Ok(own) = fs::metadata("/proc/thread-self/ns/net") else {
54        return;
55    };
56    let neighbors = match super::ipoib::ipoib_gids(Instant::now() + Duration::from_millis(200)) {
57        Ok(neighbors) => neighbors,
58        Err(error) => {
59            result
60                .warnings
61                .push(format!("IPoIB neighbors unavailable: {error:#}"));
62            return;
63        }
64    };
65    result.raw_neighbors = Some(NeighborSnapshot {
66        entries: neighbors
67            .iter()
68            .map(|(ip, gid)| (*ip, gid.clone()))
69            .collect(),
70        namespace: (own.dev(), own.ino()),
71    });
72    #[cfg(test)]
73    apply_remote_gids(
74        &mut result.fd_relations,
75        &neighbors,
76        _inode_ns,
77        (own.dev(), own.ino()),
78    );
79}

81/// Sets `remote_gid` of network destinations whose socket is in namespace `own`.
82#[cfg(test)]
83fn apply_remote_gids(
84    relations: &mut [FdRelation],
85    neighbors: &HashMap<std::net::IpAddr, String>,
86    inode_ns: &HashMap<u64, (u64, u64)>,
87    own: (u64, u64),
88) {
89    for relation in relations {
90        let same_namespace = inode_ns.get(&relation.endpoint.resource.inode) == Some(&own);
91        if let Some(socket) = &mut relation.socket
92            && socket.network_peer
93            && same_namespace
94        {
95            socket.remote_gid = socket.remote.and_then(|r| neighbors.get(&r.ip).cloned());
96        }
97    }
98}

100pub(super) fn process_from_summary(
101    summary: ProcessSummary,
102    parent_id: Option<ProcessId>,
103) -> ProcessSnapshot {
104    ProcessSnapshot {
105        identity: summary.identity,
106        parent_id,
107        name: summary.name,
108        command_line: summary.command_line,
109        uid: summary.uid,
110        username: summary.username,
111        euid: summary.euid,
112        effective_username: summary.effective_username,
113        maps: vec![],
114        maps_epoch: 0,
115        maps_error: None,
116    }
117}

123/// Collects with hardware read from `sources`, which tests point at fixtures.
124pub(super) fn collect_from(
125    discovery: &mut ProcessScanner,
126    sources: &HardwareSources,
127) -> SystemSnapshot {
128    let mut result = SystemSnapshot {
129        captured_at: process::timestamp_ms(),
130        ..Default::default()
131    };
132    result.raw_hardware = super::hardware_capture::capture(&sources.sys, &sources.proc);
133    if result.raw_hardware.truncated {
134        result
135            .warnings
136            .push("Hardware acquisition size limit reached".into());
137    }
138    #[cfg(test)]
139    let acquired = HardwareSources::acquired(result.raw_hardware.clone());
140    #[cfg(test)]
141    let hardware = HardwareObservation::observe(&acquired);
142    #[cfg(test)]
143    let mut handles = hardware.handles(&acquired);
144    let summaries = match discovery.collect() {
145        Ok(v) => v,
146        Err(e) => {
147            result.warnings.push(e.to_string());
148            return result;
149        }
150    };
151    let identities: HashMap<_, _> = summaries
152        .iter()
153        .map(|summary| (summary.identity.pid, summary.identity))
154        .collect();
155    let mut owners: HashMap<IpcIdentity, Vec<FdEndpoint>> = HashMap::new();
156    let mut infos = HashMap::new();
157    let mut inode_ns = HashMap::new();
158    let mut namespaces = HashSet::new();
159    let mut namespace_readers = Vec::new();
160    let mut denied = 0;
161    let mut metadata: HashMap<IpcIdentity, process::SocketMetadata> = HashMap::new();
162    let mut truncated = false;
163    let deadline = Instant::now() + Duration::from_secs(8);
164    for summary in summaries {
165        let parent_id = identities
166            .get(&summary.parent_pid)
167            .copied()
168            .filter(|parent| *parent != summary.identity);
169        let mut n = process_from_summary(summary, parent_id);
170        let pid = n.identity.pid;
171        let mut metadata_reader = process::SocketMetadataReader::new(n.identity);
172        if Instant::now() >= deadline {
173            n.maps_error = Some("Scan time limit reached. Retrying on the next update.".into());
174            truncated = true;
175            result.processes.push(n);
176            continue;
177        }
178        n.maps_epoch = super::monotonic_ns();
179        match process::memory_maps(pid) {
180            Ok(mut m) => {
181                if m.len() > 4096 {
182                    m.truncate(4096);
183                    truncated = true;
184                }
185                n.maps = m;
186            }
187            Err(e) => n.maps_error = Some(e.to_string()),
188        }
189        if let Ok(ns) = fs::metadata(format!("/proc/{pid}/ns/net"))
190            && namespaces.insert((ns.dev(), ns.ino()))
191        {
192            match process::NetworkNamespaceReader::open(n.identity) {
193                Ok(reader) => namespace_readers.push((n.identity, reader)),
194                Err(e) => result
195                    .warnings
196                    .push(format!("UNIX peer namespace for PID {pid}: {e:#}")),
197            }
198            for (file, protocol) in [
199                (
200                    "tcp",
201                    SocketProtocol::Tcp {
202                        family: AddressFamily::Ipv4,
203                    },
204                ),
205                (
206                    "tcp6",
207                    SocketProtocol::Tcp {
208                        family: AddressFamily::Ipv6,
209                    },
210                ),
211                (
212                    "udp",
213                    SocketProtocol::Udp {
214                        family: AddressFamily::Ipv4,
215                    },
216                ),
217                (
218                    "udp6",
219                    SocketProtocol::Udp {
220                        family: AddressFamily::Ipv6,
221                    },
222                ),
223                (
224                    "unix",
225                    SocketProtocol::Unix {
226                        socket_type: SocketType::Unknown(0),
227                    },
228                ),
229            ] {
230                if let Ok(text) = process::socket_text(&format!("/proc/{pid}/net/{file}")) {
231                    let table = if file == "unix" {
232                        process::unix_sockets(&text)
233                    } else {
234                        process::inet_sockets(&text, protocol)
235                    };
236                    for inode in table.keys() {
237                        inode_ns.insert(*inode, (ns.dev(), ns.ino()));
238                    }
239                    infos.extend(table);
240                }
241            }
242            infos.extend(process::additional_sockets(pid));
243        }
244        let mut endpoints = Vec::new();
245        match fs::read_dir(format!("/proc/{pid}/fd")) {
246            Ok(fds) => {
247                result.inspected_processes += 1;
248                for fd in fds.flatten() {
249                    if result.inspected_fds >= 100_000 || Instant::now() >= deadline {
250                        truncated = true;
251                        break;
252                    }
253                    result.inspected_fds += 1;
254                    let Ok(number) = fd.file_name().to_string_lossy().parse::<u32>() else {
255                        continue;
256                    };
257                    let meta = match fs::metadata(fd.path()) {
258                        Ok(m) => m,
259                        Err(e) => {
260                            if e.kind() == std::io::ErrorKind::PermissionDenied {
261                                denied += 1;
262                            }
263                            continue;
264                        }
265                    };
266                    let kind = if meta.file_type().is_socket() {
267                        FdKind::Socket
268                    } else if meta.file_type().is_fifo() {
269                        FdKind::Pipe
270                    } else {
271                        if meta.file_type().is_char_device() {
272                            result
273                                .raw_device_handles
274                                .push((n.identity, number, meta.rdev()));
275                            #[cfg(test)]
276                            handles.observe(n.identity, number, meta.rdev());
277                        }
278                        continue;
279                    };
280                    let resource = IpcIdentity {
281                        kind: if kind == FdKind::Pipe {
282                            IpcKind::Pipe
283                        } else {
284                            IpcKind::Socket
285                        },
286                        device: DeviceId::from_stat(meta.dev()),
287                        inode: meta.ino(),
288                    };
289                    if kind == FdKind::Socket
290                        && !metadata.contains_key(&resource)
291                        && let Ok(observed) = metadata_reader.read(number, meta.dev(), meta.ino())
292                        && observed.has_observation()
293                    {
294                        metadata.insert(resource, observed);
295                    }
296                    let flags = process::fields(&format!("/proc/{pid}/fdinfo/{number}"))
297                        .ok()
298                        .and_then(|f| f.get("flags").and_then(|v| u32::from_str_radix(v, 8).ok()))
299                        .unwrap_or(u32::MAX);
300                    endpoints.push(FdEndpoint {
301                        process_id: n.identity,
302                        fd: number,
303                        fd_count: 1,
304                        resource,
305                        connection_status: (kind == FdKind::Socket)
306                            .then_some(SocketConnectionStatus::Unknown),
307                        kind,
308                        access: FdAccess::from_flags(flags),
309                    });
310                }
311            }
312            Err(e) => {
313                if e.kind() == std::io::ErrorKind::PermissionDenied {
314                    denied += 1;
315                }
316            }
317        }
318        if process::validate_process_identity(n.identity).is_ok() {
319            for p in endpoints {
320                let list = owners.entry(p.resource).or_default();
321                if let Some(existing) = list
322                    .iter_mut()
323                    .find(|e| e.process_id == p.process_id && e.access == p.access)
324                {
325                    existing.fd = existing.fd.min(p.fd);
326                    existing.fd_count += 1;
327                } else {
328                    list.push(p);
329                }
330            }
331            result.processes.push(n);
332        }
333    }
334    let diag_deadline = Instant::now() + Duration::from_millis(500);
335    for (id, namespace) in namespace_readers {
336        if Instant::now() >= diag_deadline {
337            result.warnings.push(
338                "UNIX peer namespace scan reached its 500ms limit. Some peers may be missing."
339                    .into(),
340            );
341            break;
342        }
343        match process::unix_socket_peers(&namespace, diag_deadline) {
344            Ok(diag) => infos.extend(diag),
345            Err(e) => result.warnings.push(format!(
346                "UNIX peer for PID {} / namespace {:?}: {e:#}",
347                id.pid,
348                namespace.key()
349            )),
350        }
351    }
352    // Enrich after all table/diag reads so direct observations cannot be overwritten.
353    for (resource, observed) in &metadata {
354        if let Some(protocol) = observed.protocol {
355            infos
356                .entry(resource.inode)
357                .and_modify(|info| {
358                    if matches!(
359                        info.protocol,
360                        SocketProtocol::Other {
361                            socket_type: None,
362                            ..
363                        }
364                    ) {
365                        info.protocol = protocol;
366                    }
367                })
368                .or_insert(SocketInfo {
369                    protocol,
370                    state: None,
371                    local: None,
372                    remote: None,
373                    path: None,
374                    peer_inode: None,
375                });
376        }
377    }
378    result.raw_owners = owners.into_iter().collect();
379    let resolver = {
380        static RESOLVER: std::sync::OnceLock<super::resolver::ReverseDnsResolver> =
381            std::sync::OnceLock::new();
382        RESOLVER.get_or_init(super::resolver::ReverseDnsResolver::new)
383    };
384    let ips: HashSet<_> = infos
385        .values()
386        .filter_map(|socket| socket.remote.map(|address| address.ip()))
387        .collect();
388    result.raw_hostnames = ips
389        .into_iter()
390        .filter_map(|ip| resolver.lookup(ip).map(|name| (ip, name)))
391        .collect();
392    result.raw_sockets = infos.into_iter().collect();
393    result.raw_namespaces = inode_ns.iter().map(|(inode, ns)| (*inode, *ns)).collect();
394    result.raw_connections = metadata
395        .into_iter()
396        .map(|(id, meta)| (id, meta.connection_status))
397        .collect();
398    #[cfg(test)]
399    {
400        truncated |= crate::http_server::analysis::relations::build(&mut result);
401    }
402    add_remote_gids(&mut result, &inode_ns, sources);
403    let pids = result
404        .processes
405        .iter()
406        .map(|p| (p.identity.pid, p.identity))
407        .collect();
408    let observed = rdma::observe(
409        &sources.sys,
410        &pids,
411        Instant::now() + Duration::from_millis(300),
412    );
413    result.rdma_ports = observed.ports;
414    result.rdma_relations = observed.relations;
415    for relation in &mut result.rdma_relations {
416        if relation.peer.is_none() {
417            relation.remote_hostname = relation
418                .cm
419                .as_ref()
420                .and_then(|cm| cm.destination?.ip())
421                .and_then(|ip| resolver.lookup(ip));
422        }
423    }
424    result.warnings.extend(observed.warnings);
425    #[cfg(test)]
426    {
427        let live = result.processes.iter().map(|p| p.identity).collect();
428        result.device_handles = handles.finish(&live);
429        result.hardware_devices = hardware.devices;
430        result.hardware_paths = hardware.paths;
431    }
432    if denied > 0 {
433        result
434            .warnings
435            .push(format!("Access denied for {denied} processes/FDs"));
436    }
437    if truncated {
438        result
439            .warnings
440            .push("Scan, mapping, or connection limit reached. Some process or FD information is missing.".into());
441    }
442    result
443}

449    #[test]
450    fn remote_gids_apply_only_to_destinations_of_the_collector_namespace() {
451        use crate::privileged_helper::process::SocketInfo;
452        let info = SocketInfo {
453            protocol: SocketProtocol::Tcp {
454                family: AddressFamily::Ipv4,
455            },
456            state: Some(SocketState::Established),
457            local: Some("10.10.10.2:5000".parse().unwrap()),
458            remote: Some("10.10.10.1:20049".parse().unwrap()),
459            path: None,
460            peer_inode: None,
461        };
462        let relation = |inode: u64| FdRelation {
463            id: inode.to_string(),
464            endpoint: FdEndpoint {
465                process_id: ProcessId {
466                    pid: 1,
467                    start_time_ticks: 1,
468                },
469                fd: 3,
470                fd_count: 1,
471                resource: IpcIdentity {
472                    kind: IpcKind::Socket,
473                    device: DeviceId::from_stat(0),
474                    inode,
475                },
476                kind: FdKind::Socket,
477                access: FdAccess::ReadWrite,
478                connection_status: None,
479            },
480            peer: None,
481            label: String::new(),
482            socket: Some(SocketEndpoint::from(&info)),
483            candidate: false,
484            shared: false,
485        };
486        let mut relations = [relation(1), relation(2)];
487        let neighbors = HashMap::from([(
488            "10.10.10.1".parse().unwrap(),
489            "fe80:0000:0000:0000:f452:1403:0080:de51".to_owned(),
490        )]);
491        let namespaces = HashMap::from([(1, (5, 100)), (2, (5, 200))]);
492        apply_remote_gids(&mut relations, &neighbors, &namespaces, (5, 100));
493        let gid = |r: &FdRelation| r.socket.as_ref().unwrap().remote_gid.clone();
494        assert_eq!(
495            gid(&relations[0]).as_deref(),
496            Some("fe80:0000:0000:0000:f452:1403:0080:de51")
497        );
498        assert_eq!(
499            gid(&relations[1]),
500            None,
501            "another namespace has its own neighbors"
502        );
503        assert_eq!(
504            serde_json::to_value(relations[1].socket.as_ref().unwrap()).unwrap()["remote_gid"],
505            serde_json::Value::Null
506        );
507    }

509    #[test]
510    fn socket_destination_classification() {
511        use crate::privileged_helper::process::SocketInfo;
512        let mut info = SocketInfo {
513            protocol: SocketProtocol::Tcp {
514                family: AddressFamily::Ipv6,
515            },
516            state: Some(SocketState::Established),
517            local: Some("[::1]:5000".parse().unwrap()),
518            remote: Some("[2001:db8::1]:443".parse().unwrap()),
519            path: None,
520            peer_inode: None,
521        };
522        let endpoint = SocketEndpoint::from(&info);
523        assert!(endpoint.network_peer);
524        assert_eq!(
525            serde_json::to_value(endpoint).unwrap()["remote"],
526            serde_json::json!({"ip":"2001:db8::1","port":443})
527        );
528        info.state = Some(SocketState::Listen);
529        assert!(!SocketEndpoint::from(&info).network_peer);
530        info.protocol = SocketProtocol::Udp {
531            family: AddressFamily::Ipv4,
532        };
533        info.state = Some(SocketState::Unconnected);
534        for address in ["0.0.0.0:0", "127.0.0.1:0", "[::]:123"] {
535            info.remote = Some(address.parse().unwrap());
536            assert!(!SocketEndpoint::from(&info).network_peer);
537        }
538        info.remote = None;
539        assert!(!SocketEndpoint::from(&info).network_peer);
540        info.protocol = SocketProtocol::Unix {
541            socket_type: SocketType::Unknown(0),
542        };
543        assert!(!SocketEndpoint::from(&info).network_peer);
544        info.protocol = SocketProtocol::from_codes(
545            libc::AF_INET as u32,
546            libc::SOCK_RAW as u32,
547            libc::IPPROTO_TCP as u32,
548        );
549        info.remote = Some("127.0.0.1:443".parse().unwrap());
550        assert!(!SocketEndpoint::from(&info).network_peer);
551    }
