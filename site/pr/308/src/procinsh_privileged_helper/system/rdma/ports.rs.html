1//! Local RDMA port addresses and link state from sysfs.
2
3use std::{fs, path::Path};
4
5pub(in crate::privileged_helper::system) use crate::ipc::rdma::*;
6
7fn text(path: &Path) -> Option<String> {
8    let value = fs::read_to_string(path).ok()?;
9    let value = value.trim();
10    (!value.is_empty()).then(|| value.to_owned())
11}
12
13fn names(path: &Path) -> Vec<String> {
14    let mut names: Vec<_> = fs::read_dir(path)
15        .into_iter()
16        .flatten()
17        .flatten()
18        .map(|entry| entry.file_name().to_string_lossy().into_owned())
19        .collect();
20    names.sort();
21    names
22}
23
24/// Device names under `root` (`/sys/class/infiniband`).
25pub(super) fn devices(root: &Path) -> Vec<String> {
26    names(root)
27}
28
29pub(super) fn read_ports(root: &Path) -> Vec<RdmaPortInfo> {
30    let mut result = Vec::new();
31    for device in devices(root) {
32        let ports = root.join(&device).join("ports");
33        let mut numbers: Vec<u32> = names(&ports)
34            .iter()
35            .filter_map(|name| name.parse().ok())
36            .collect();
37        numbers.sort();
38        for port in numbers {
39            let path = ports.join(port.to_string());
40            result.push(RdmaPortInfo {
41                device: device.clone(),
42                port,
43                link_layer: text(&path.join("link_layer")),
44                state: text(&path.join("state")).map(|s| {
45                    s.split_once(": ")
46                        .map_or(s.clone(), |(_, name)| name.into())
47                }),
48                lid: text(&path.join("lid"))
49                    .and_then(|s| u32::from_str_radix(s.trim_start_matches("0x"), 16).ok())
50                    .filter(|lid| *lid != 0),
51                gid: text(&path.join("gids/0")).and_then(|s| IbGid::parse(&s)),
52                rate: text(&path.join("rate")),
53            });
54        }
55    }
56    result
57}

59#[cfg(test)]
60mod tests {
61    use super::*;
62
63    #[test]
64    fn sysfs_ports_parse_addresses_and_keep_unassigned_values_null() {
65        let root = std::env::temp_dir().join(format!("procinsh-rdma-{}", std::process::id()));
66        let _ = fs::remove_dir_all(&root);
67        for (device, port, lid, gid) in [
68            (
69                "mlx4_0",
70                "1",
71                "0x2",
72                "fe80:0000:0000:0000:0002:c903:0028:45c7",
73            ),
74            (
75                "mlx4_0",
76                "2",
77                "0x0",
78                "0000:0000:0000:0000:0000:0000:0000:0000",
79            ),
80            (
81                "mlx4_0",
82                "10",
83                "0x10",
84                "fe80:0000:0000:0000:0002:c903:0028:45c8",
85            ),
86        ] {
87            let path = root.join(device).join("ports").join(port);
88            fs::create_dir_all(path.join("gids")).unwrap();
89            fs::write(path.join("lid"), format!("{lid}\n")).unwrap();
90            fs::write(path.join("gids/0"), format!("{gid}\n")).unwrap();
91            fs::write(path.join("state"), "4: ACTIVE\n").unwrap();
92            fs::write(path.join("link_layer"), "InfiniBand\n").unwrap();
93        }
94        fs::create_dir_all(root.join("rxe0")).unwrap();
95        let ports = read_ports(&root);
96        assert_eq!(devices(&root), ["mlx4_0", "rxe0"]);
97        assert_eq!(ports.iter().map(|p| p.port).collect::<Vec<_>>(), [1, 2, 10]);
98        assert_eq!(
99            serde_json::to_value(&ports[0]).unwrap(),
100            serde_json::json!({
101                "device": "mlx4_0",
102                "port": 1,
103                "link_layer": "InfiniBand",
104                "state": "ACTIVE",
105                "lid": 2,
106                "gid": "fe80:0000:0000:0000:0002:c903:0028:45c7",
107                "rate": null,
108            })
109        );
110        assert_eq!((ports[1].lid, ports[1].gid), (None, None));
111        fs::remove_dir_all(&root).unwrap();
112    }
113}
