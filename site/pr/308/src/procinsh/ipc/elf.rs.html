1//! Bounded ranges of a verified mapped file, transferred without descriptors.
2//!
3//! # Interface
4//! - [`ElfFileId`] (`pub`): `type ElfFileId = (u64,u64,u64,i64,i64)`.
5//! - [`FileRange`] (`pub`): `struct FileRange`.
6//! - [`FileChunk`] (`pub`): `struct FileChunk`.
7//! - [`MappedFileData`] (`pub`): `struct MappedFileData`.
8//! - [`MAX_FILE_SIZE`] (`pub`): `const MAX_FILE_SIZE: u64`.
9//! - [`MAX_READ_SIZE`] (`pub`): `const MAX_READ_SIZE: u64`.
10//! - [`MAX_RANGES`] (`pub`): `const MAX_RANGES: usize`.
11pub type ElfFileId = (u64, u64, u64, i64, i64);
12pub const MAX_FILE_SIZE: u64 = 128 * 1024 * 1024;
13pub const MAX_READ_SIZE: u64 = 64 * 1024 * 1024;
14pub const MAX_RANGES: usize = 60;
15#[derive(Clone, Copy, Debug)]
16pub struct FileRange {
17    pub offset: u64,
18    pub length: u64,
19}
20pub struct FileChunk {
21    pub offset: u64,
22    pub bytes: Vec<u8>,
23}
24pub struct MappedFileData {
25    pub identity: ElfFileId,
26    pub chunks: Vec<FileChunk>,
27}
28crate::ipc::codec::wire_struct!(FileRange { offset, length });
29crate::ipc::codec::wire_struct!(MappedFileData { identity, chunks });
30impl crate::ipc::codec::Wire for FileChunk {
31    fn write(&self, o: &mut crate::ipc::codec::Encoder) -> anyhow::Result<()> {
32        self.offset.write(o)?;
33        o.blob(&self.bytes)
34    }
35    fn read(i: &mut crate::ipc::codec::Decoder<'_>) -> anyhow::Result<Self> {
36        let offset = crate::ipc::codec::Wire::read(i)?;
37        let bytes = i.blob()?;
38        anyhow::ensure!(
39            bytes.len() as u64 <= MAX_READ_SIZE,
40            "mapped file read exceeds limit"
41        );
42        Ok(Self {
43            offset,
44            bytes: bytes.to_vec(),
45        })
46    }
47}
