1//! Connections whose two ends are observed on different hosts.
2//!
3//! Each host reports a connection to another host as an external endpoint. A TCP
4//! or UDP pair matches by reversed address tuples; an RDMA pair by swapped QPNs
5//! with remote GID or connection manager addresses as evidence. Only unique
6//! matches link, so NAT and duplicated tuples stay external.
7
8use serde::{Deserialize, Serialize};
9use serde_json::Value;

12/// One end of a cross-host link: a host and a relation ID in its snapshot.
13#[derive(Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord, Serialize)]
14pub(in crate::http_server) struct LinkEnd {
15    pub(in crate::http_server) host: String,
16    pub(in crate::http_server) relation: String,
17}

98fn entries<T: for<'de> Deserialize<'de>>(snapshot: &Value, name: &str) -> Vec<T> {
99    // Entries that do not fit the expected shape are skipped, not fatal.
100    snapshot
101        .get(name)
102        .and_then(Value::as_array)
103        .into_iter()
104        .flatten()
105        .filter_map(|entry| T::deserialize(entry).ok())
106        .collect()
107}

109fn routable(address: Address) -> bool {
110    address.port != 0 && !address.ip.is_unspecified() && !address.ip.is_loopback()
111}
112
113fn socket_links(hosts: &[(&str, &Value)]) -> Vec<SocketLink> {
114    type Tuple = (String, Address, Address);
115    let mut by_tuple: HashMap<Tuple, Vec<LinkEnd>> = HashMap::new();
116    for (host, snapshot) in hosts {
117        for relation in entries::<FdRelation>(snapshot, "fd_relations") {
118            let Some(socket) = relation.socket else {
119                continue;
120            };
121            if relation.peer.is_some() || relation.shared || !socket.network_peer {
122                continue;
123            }
124            let (Some(local), Some(remote)) = (socket.local, socket.remote) else {
125                continue;
126            };
127            if !routable(local) || !routable(remote) {
128                continue;
129            }
130            by_tuple
131                .entry((socket.protocol.kind, local, remote))
132                .or_default()
133                .push(LinkEnd {
134                    host: (*host).to_owned(),
135                    relation: relation.id,
136                });
137        }
138    }
139    let mut links = Vec::new();
140    for ((kind, local, remote), ends) in &by_tuple {
141        let Some(reverse) = by_tuple.get(&(kind.clone(), *remote, *local)) else {
142            continue;
143        };
144        // Each tuple once, from its smaller end; both sides must be unique.
145        if ends.len() != 1 || reverse.len() != 1 || ends[0] >= reverse[0] {
146            continue;
147        }
148        if ends[0].host != reverse[0].host {
149            links.push(SocketLink {
150                a: ends[0].clone(),
151                b: reverse[0].clone(),
152            });
153        }
154    }
155    links.sort_by(|x, y| (&x.a, &x.b).cmp(&(&y.a, &y.b)));
156    links
157}

159struct Qp<'a> {
160    end: LinkEnd,
161    relation: RdmaRelation,
162    gid: Option<&'a str>,
163    link_layer: Option<&'a str>,
164}
165
166fn rdma_links(hosts: &[(&str, &Value)]) -> Vec<RdmaLink> {
167    let ports: Vec<Vec<RdmaPort>> = hosts
168        .iter()
169        .map(|(_, snapshot)| entries(snapshot, "rdma_ports"))
170        .collect();
171    let mut qps = Vec::new();
172    for ((host, snapshot), ports) in hosts.iter().zip(&ports) {
173        for relation in entries::<RdmaRelation>(snapshot, "rdma_relations") {
174            let connected = matches!(
175                relation.endpoint.qp_type.kind.as_str(),
176                "rc" | "uc" | "xrc_ini" | "xrc_tgt"
177            );
178            if relation.peer.is_some() || !connected || relation.remote_qpn.is_none() {
179                continue;
180            }
181            let port = ports.iter().find(|p| {
182                p.device == relation.endpoint.device && Some(p.port) == relation.endpoint.port
183            });
184            qps.push(Qp {
185                end: LinkEnd {
186                    host: (*host).to_owned(),
187                    relation: relation.id.clone(),
188                },
189                relation,
190                gid: port.and_then(|p| p.gid.as_deref()),
191                link_layer: port.and_then(|p| p.link_layer.as_deref()),
192            });
193        }
194    }
195    let mut by_qpn: HashMap<u32, Vec<usize>> = HashMap::new();
196    for (index, qp) in qps.iter().enumerate() {
197        by_qpn
198            .entry(qp.relation.endpoint.qpn)
199            .or_default()
200            .push(index);
201    }
202    let evidence = |a: &Qp, b: &Qp| -> Option<RdmaEvidence> {
203        let gid = |from: &Qp, to: &Qp| match (from.relation.remote_gid.as_deref(), to.gid) {
204            (Some(remote), Some(local)) => Some(remote == local),
205            _ => None,
206        };
207        let cm = match (&a.relation.cm, &b.relation.cm) {
208            (Some(x), Some(y)) => match (&x.source, &x.destination, &y.source, &y.destination) {
209                (Some(xs), Some(xd), Some(ys), Some(yd)) => Some(xs == yd && xd == ys),
210                _ => None,
211            },
212            _ => None,
213        };
214        let gids = [gid(a, b), gid(b, a)];
215        if gids.contains(&Some(false)) || cm == Some(false) {
216            return None;
217        }
218        Some(RdmaEvidence {
219            gid: gids.contains(&Some(true)),
220            cm: cm == Some(true),
221        })
222    };
223    let candidates = |index: usize| -> Vec<(usize, RdmaEvidence)> {
224        let a = &qps[index];
225        by_qpn
226            .get(&a.relation.remote_qpn.unwrap())
227            .into_iter()
228            .flatten()
229            .filter(|&&b| {
230                qps[b].end.host != a.end.host
231                    && qps[b].relation.remote_qpn == Some(a.relation.endpoint.qpn)
232            })
233            .filter_map(|&b| Some((b, evidence(a, &qps[b])?)))
234            .collect()
235    };
236    let mut links = Vec::new();
237    for index in 0..qps.len() {
238        let found = candidates(index);
239        let [(other, evidence)] = found.as_slice() else {
240            continue;
241        };
242        let (a, b) = (&qps[index], &qps[*other]);
243        // A bare QPN swap links only when unique in both directions on one link layer.
244        let unique = matches!(candidates(*other).as_slice(), [(back, _)] if *back == index);
245        let addressed = evidence.gid || evidence.cm;
246        if a.end >= b.end || !unique || (!addressed && a.link_layer != b.link_layer) {
247            continue;
248        }
249        links.push(RdmaLink {
250            a: a.end.clone(),
251            b: b.end.clone(),
252            evidence: *evidence,
253        });
254    }
255    links.sort_by(|x, y| (&x.a, &x.b).cmp(&(&y.a, &y.b)));
256    links
257}

259/// Links between `hosts`, given as host names and their snapshots.
260pub(in crate::http_server) fn correlate(hosts: &[(&str, &Value)]) -> CrossHostLinks {
261    CrossHostLinks {
262        sockets: socket_links(hosts),
263        rdma: rdma_links(hosts),
264    }
265}

272    fn tcp(id: &str, local: &str, remote: &str) -> Value {
273        let address = |text: &str| {
274            let a: std::net::SocketAddr = text.parse().unwrap();
275            json!({"ip": a.ip(), "port": a.port()})
276        };
277        json!({"id": id, "peer": null, "shared": false, "socket": {
278            "protocol": {"kind": "tcp", "family": "ipv4"}, "state": {"kind": "established"},
279            "local": address(local), "remote": address(remote), "network_peer": true}})
280    }
281
282    fn qp(id: &str, qpn: u32, remote: u32, gid: Option<&str>, cm: Option<(&str, &str)>) -> Value {
283        let inet = |text: &str| {
284            let a: std::net::SocketAddr = text.parse().unwrap();
285            json!({"kind": "inet", "ip": a.ip(), "port": a.port()})
286        };
287        json!({"id": id, "peer": null, "remote_qpn": remote, "remote_gid": gid,
288            "endpoint": {"device": "ib0", "port": 1, "qpn": qpn, "qp_type": {"kind": "rc"}},
289            "cm": cm.map(|(s, d)| json!({"source": inet(s), "destination": inet(d)}))})
290    }

292    fn host<'a>(name: &'a str, fd: Vec<Value>, rdma: Vec<Value>, gid: &str) -> (&'a str, Value) {
293        (
294            name,
295            json!({"fd_relations": fd, "rdma_relations": rdma,
296                "rdma_ports": [{"device": "ib0", "port": 1, "link_layer": "InfiniBand", "gid": gid}]}),
297        )
298    }

300    fn correlate(hosts: &[(&str, Value)]) -> CrossHostLinks {
301        let hosts: Vec<_> = hosts.iter().map(|(name, s)| (*name, s)).collect();
302        super::correlate(&hosts)
303    }

308    #[test]
309    fn reversed_unique_tuples_link_sockets_across_hosts() {
310        let links = correlate(&[
311            host(
312                "a",
313                vec![
314                    tcp("x", "10.0.0.1:5000", "10.0.0.2:443"),
315                    tcp("dup1", "10.0.0.1:1", "10.0.0.9:9"),
316                ],
317                vec![],
318                GA,
319            ),
320            host(
321                "b",
322                vec![
323                    tcp("y", "10.0.0.2:443", "10.0.0.1:5000"),
324                    tcp("dup2", "10.0.0.9:9", "10.0.0.1:1"),
325                    tcp("lo", "127.0.0.1:80", "127.0.0.1:81"),
326                ],
327                vec![],
328                GB,
329            ),
330            host(
331                "c",
332                vec![
333                    tcp("dup3", "10.0.0.9:9", "10.0.0.1:1"),
334                    tcp("lo2", "127.0.0.1:81", "127.0.0.1:80"),
335                ],
336                vec![],
337                GA,
338            ),
339        ]);
340        assert_eq!(
341            serde_json::to_value(&links.sockets).unwrap(),
342            json!([{"a": {"host": "a", "relation": "x"}, "b": {"host": "b", "relation": "y"}}])
343        );
344        // The same host never links to itself, even with reversed tuples.
345        let single = correlate(&[host(
346            "a",
347            vec![
348                tcp("x", "10.0.0.1:1", "10.0.0.2:2"),
349                tcp("y", "10.0.0.2:2", "10.0.0.1:1"),
350            ],
351            vec![],
352            GA,
353        )]);
354        assert!(single.sockets.is_empty());
355    }

357    #[test]
358    fn rdma_links_need_swapped_qpns_and_consistent_addresses() {
359        // Remote GIDs naming each other's ports.
360        let links = correlate(&[
361            host("a", vec![], vec![qp("qa", 10, 20, Some(GB), None)], GA),
362            host("b", vec![], vec![qp("qb", 20, 10, Some(GA), None)], GB),
363        ]);
364        assert_eq!(
365            serde_json::to_value(&links.rdma).unwrap(),
366            json!([{"a": {"host": "a", "relation": "qa"}, "b": {"host": "b", "relation": "qb"},
367                "evidence": {"gid": true, "cm": false}}])
368        );
369        // Swapped CM addresses, and a unique bare swap on the same link layer.
370        let cm = correlate(&[
371            host(
372                "a",
373                vec![],
374                vec![qp("qa", 10, 20, None, Some(("10.0.0.1:1", "10.0.0.2:2")))],
375                GA,
376            ),
377            host(
378                "b",
379                vec![],
380                vec![qp("qb", 20, 10, None, Some(("10.0.0.2:2", "10.0.0.1:1")))],
381                GB,
382            ),
383        ]);
384        assert_eq!(
385            cm.rdma[0].evidence,
386            RdmaEvidence {
387                gid: false,
388                cm: true
389            }
390        );
391        let bare = correlate(&[
392            host("a", vec![], vec![qp("qa", 10, 20, None, None)], GA),
393            host("b", vec![], vec![qp("qb", 20, 10, None, None)], GB),
394        ]);
395        assert_eq!(bare.rdma.len(), 1);
396        // A remote GID of a third machine contradicts the swap.
397        let contradicted = correlate(&[
398            host(
399                "a",
400                vec![],
401                vec![qp(
402                    "qa",
403                    10,
404                    20,
405                    Some("fe80:0000:0000:0000:0000:0000:0000:00ff"),
406                    None,
407                )],
408                GA,
409            ),
410            host("b", vec![], vec![qp("qb", 20, 10, None, None)], GB),
411        ]);
412        assert!(contradicted.rdma.is_empty());
413        // An ambiguous bare swap stays external.
414        let ambiguous = correlate(&[
415            host("a", vec![], vec![qp("qa", 10, 20, None, None)], GA),
416            host("b", vec![], vec![qp("qb", 20, 10, None, None)], GB),
417            host("c", vec![], vec![qp("qc", 20, 10, None, None)], GB),
418        ]);
419        assert!(ambiguous.rdma.is_empty());
420        // Malformed entries are ignored.
421        let malformed = correlate(&[host("a", vec![json!({"id": 1})], vec![json!("x")], GA)]);
422        assert_eq!(malformed, CrossHostLinks::default());
423    }
