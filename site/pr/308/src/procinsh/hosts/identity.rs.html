1//! The identity a host advertises to a hub and shows in the host list.
2
3use serde::Serialize;
4use std::fs;
5
6#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
7pub(in crate::http_server) struct HostInfo {
8    pub(in crate::http_server) name: String,
9    /// `/etc/machine-id`, or the boot ID when the machine ID is unavailable.
10    pub(in crate::http_server) machine_id: String,
11    pub(in crate::http_server) version: String,
12    pub(in crate::http_server) kernel: Option<String>,
13    /// `std::env::consts::ARCH` of the procinsh build, such as `x86_64`.
14    pub(in crate::http_server) arch: String,
15}
16
17fn text(path: &str) -> Option<String> {
18    let value = fs::read_to_string(path).ok()?;
19    let value = value.trim();
20    (!value.is_empty()).then(|| value.to_owned())
21}
22
23/// The `?host=` value of the merged view.
24pub(in crate::http_server) const ALL_HOSTS: &str = "all";
25
26/// Host names are short, URL-safe labels; [`ALL_HOSTS`] names the merged view.
27pub(crate) fn valid_host_name(name: &str) -> bool {
28    (1..=63).contains(&name.len())
29        && name
30            .bytes()
31            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-'))
32        && name != ALL_HOSTS
33}
34
35impl HostInfo {
36    /// This machine, named by `name` or the kernel host name.
37    pub(in crate::http_server) fn local(name: Option<String>) -> Self {
38        Self {
39            name: name
40                .or_else(|| text("/proc/sys/kernel/hostname"))
41                .unwrap_or_else(|| "localhost".into()),
42            machine_id: text("/etc/machine-id")
43                .or_else(|| text("/proc/sys/kernel/random/boot_id"))
44                .unwrap_or_default(),
45            version: env!("CARGO_PKG_VERSION").into(),
46            kernel: text("/proc/sys/kernel/osrelease"),
47            arch: std::env::consts::ARCH.into(),
48        }
49    }
50}

52#[cfg(test)]
53mod tests {
54    use super::*;
55
56    #[test]
57    fn host_names_are_short_url_safe_and_not_all() {
58        for name in ["node-a", "gpu01.cluster", "a_b", &"x".repeat(63)] {
59            assert!(valid_host_name(name), "{name}");
60        }
61        for name in ["", "all", "a b", "a/b", "a?b", &"x".repeat(64)] {
62            assert!(!valid_host_name(name), "{name}");
63        }
64        let local = HostInfo::local(Some("override".into()));
65        assert_eq!(local.name, "override");
66        assert_eq!(local.version, env!("CARGO_PKG_VERSION"));
67        assert!(!HostInfo::local(None).name.is_empty());
68    }
