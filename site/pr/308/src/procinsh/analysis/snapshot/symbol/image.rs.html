1//! Main-owned sparse ELF bytes and selection of symbol, DWARF and unwind sections.
2//!
3//! # Interface
4//! [`ElfImage`] (`pub`): `struct ElfImage` stores verified file ranges for parsing.
5use crate::ipc::elf::{
6    ElfFileId, FileChunk, FileRange, MAX_FILE_SIZE, MAX_READ_SIZE, MappedFileData,
7};
8use anyhow::{Context, Result, ensure};
9use object::read::ReadRef;
10use std::ops::Range;
11
12pub struct ElfImage {
13    pub identity: ElfFileId,
14    chunks: Vec<FileChunk>,
15}
16impl ElfImage {
17    pub fn new(data: MappedFileData) -> Result<Self> {
18        ensure!(data.identity.2 <= MAX_FILE_SIZE, "ELF file exceeds limit");
19        let mut image = Self {
20            identity: data.identity,
21            chunks: Vec::new(),
22        };
23        image.extend(data)?;
24        Ok(image)
25    }
26    pub fn extend(&mut self, data: MappedFileData) -> Result<()> {
27        ensure!(
28            data.identity == self.identity,
29            "mapped file changed during acquisition"
30        );
31        self.chunks.extend(data.chunks);
32        self.chunks.sort_by_key(|chunk| chunk.offset);
33        let mut merged: Vec<FileChunk> = Vec::new();
34        for chunk in self.chunks.drain(..) {
35            let end = chunk
36                .offset
37                .checked_add(chunk.bytes.len() as u64)
38                .context("ELF range overflow")?;
39            ensure!(end <= self.identity.2, "ELF range exceeds file");
40            if let Some(last) = merged.last_mut()
41                && chunk.offset <= last.offset + last.bytes.len() as u64
42            {
43                let overlap = (last.offset + last.bytes.len() as u64 - chunk.offset) as usize;
44                if overlap < chunk.bytes.len() {
45                    last.bytes.extend_from_slice(&chunk.bytes[overlap..]);
46                }
47            } else {
48                merged.push(chunk);
49            }
50        }
51        ensure!(
52            merged
53                .iter()
54                .map(|chunk| chunk.bytes.len() as u64)
55                .sum::<u64>()
56                <= MAX_READ_SIZE,
57            "ELF acquisition exceeds limit"
58        );
59        self.chunks = merged;
60        Ok(())
61    }
62    fn bytes(&self, offset: u64, length: u64) -> Result<&[u8]> {
63        self.read_bytes_at(offset, length)
64            .map_err(|()| anyhow::anyhow!("missing ELF range"))
65    }
66    fn number(&self, offset: u64, length: u64) -> Result<u64> {
67        let bytes = self.bytes(offset, length)?;
68        Ok(match length {
69            2 => u16::from_le_bytes(bytes.try_into()?) as u64,
70            4 => u32::from_le_bytes(bytes.try_into()?) as u64,
71            8 => u64::from_le_bytes(bytes.try_into()?),
72            _ => unreachable!(),
73        })
74    }
75    fn is64(&self) -> Result<bool> {
76        let ident = self.bytes(0, 16)?;
77        ensure!(
78            &ident[..4] == b"\x7fELF" && ident[5] == 1 && ident[6] == 1,
79            "unsupported ELF header"
80        );
81        match ident[4] {
82            1 => Ok(false),
83            2 => Ok(true),
84            _ => anyhow::bail!("unsupported ELF class"),
85        }
86    }
87    fn section_table(&self) -> Result<(u64, u64, u64, u64)> {
88        let wide = self.is64()?;
89        let offset = self.number(if wide { 40 } else { 32 }, if wide { 8 } else { 4 })?;
90        let size = self.number(if wide { 58 } else { 46 }, 2)?;
91        let count = self.number(if wide { 60 } else { 48 }, 2)?;
92        let names = self.number(if wide { 62 } else { 50 }, 2)?;
93        ensure!(
94            count > 0 && count <= 4096 && size == if wide { 64 } else { 40 } && names < count,
95            "unsupported ELF section table"
96        );
97        Ok((offset, size, count, names))
98    }
99    pub fn tables(&self) -> Result<Vec<FileRange>> {
100        let (offset, size, count, _) = self.section_table()?;
101        let wide = self.is64()?;
102        let programs = self.number(if wide { 32 } else { 28 }, if wide { 8 } else { 4 })?;
103        let entry_size = self.number(if wide { 54 } else { 42 }, 2)?;
104        let entries = self.number(if wide { 56 } else { 44 }, 2)?;
105        ensure!(entries <= 4096, "too many ELF segments");
106        self.ranges(vec![
107            FileRange {
108                offset,
109                length: size * count,
110            },
111            FileRange {
112                offset: programs,
113                length: entry_size * entries,
114            },
115        ])
116    }
117    fn section(&self, index: u64) -> Result<(u64, u64, u64, u64)> {
118        let (table, size, count, _) = self.section_table()?;
119        ensure!(index < count, "invalid ELF section index");
120        let at = table + index * size;
121        let wide = self.is64()?;
122        Ok((
123            self.number(at, 4)?,
124            self.number(at + 4, 4)?,
125            self.number(at + if wide { 24 } else { 16 }, if wide { 8 } else { 4 })?,
126            self.number(at + if wide { 32 } else { 20 }, if wide { 8 } else { 4 })?,
127        ))
128    }
129    pub fn names(&self) -> Result<Vec<FileRange>> {
130        let (_, _, _, index) = self.section_table()?;
131        let (_, _, offset, length) = self.section(index)?;
132        self.ranges(vec![FileRange { offset, length }])
133    }
134    pub fn sections(&self) -> Result<Vec<FileRange>> {
135        let (_, _, count, index) = self.section_table()?;
136        let (_, _, offset, length) = self.section(index)?;
137        let names = self.bytes(offset, length)?;
138        let mut ranges = Vec::new();
139        for index in 0..count {
140            let (name, kind, offset, length) = self.section(index)?;
141            if kind == 8 || length == 0 {
142                continue;
143            } // SHT_NOBITS has no file bytes.
144            let name = names
145                .get(name as usize..)
146                .context("invalid ELF section name")?;
147            let end = name
148                .iter()
149                .position(|byte| *byte == 0)
150                .context("unterminated ELF section name")?;
151            let name = std::str::from_utf8(&name[..end])?;
152            if needed_section(name) || matches!(kind, 2 | 3 | 11 | 18) {
153                ensure!(length <= 32 * 1024 * 1024, "ELF section exceeds limit");
154                ranges.push(FileRange { offset, length });
155            }
156        }
157        self.ranges(ranges)
158    }
159    fn ranges(&self, mut ranges: Vec<FileRange>) -> Result<Vec<FileRange>> {
160        ranges.retain(|range| {
161            range.length != 0 && self.read_bytes_at(range.offset, range.length).is_err()
162        });
163        ranges.sort_by_key(|range| range.offset);
164        let mut merged: Vec<FileRange> = Vec::new();
165        for range in ranges {
166            let end = range
167                .offset
168                .checked_add(range.length)
169                .context("ELF range overflow")?;
170            ensure!(end <= self.identity.2, "ELF range exceeds file");
171            if let Some(last) = merged.last_mut()
172                && range.offset <= last.offset + last.length
173            {
174                last.length = last.length.max(end - last.offset);
175            } else {
176                merged.push(range);
177            }
178        }
179        ensure!(
180            merged.iter().map(|range| range.length).sum::<u64>() <= MAX_READ_SIZE,
181            "ELF ranges exceed limit"
182        );
183        Ok(merged)
184    }
185}
186pub(super) fn needed_section(name: &str) -> bool {
187    name.starts_with(".debug_")
188        || name.starts_with(".zdebug_")
189        || matches!(
190            name,
191            ".eh_frame"
192                | ".eh_frame_hdr"
193                | ".symtab"
194                | ".dynsym"
195                | ".strtab"
196                | ".dynstr"
197                | ".shstrtab"
198                | ".symtab_shndx"
199        )
200}
201impl<'a> ReadRef<'a> for &'a ElfImage {
202    fn len(self) -> std::result::Result<u64, ()> {
203        Ok(self.identity.2)
204    }
205    fn read_bytes_at(self, offset: u64, size: u64) -> std::result::Result<&'a [u8], ()> {
206        if offset > self.identity.2 {
207            return Err(());
208        }
209        if size == 0 {
210            return Ok(&[]);
211        }
212        let end = offset.checked_add(size).ok_or(())?;
213        self.chunks
214            .iter()
215            .find_map(|chunk| {
216                (offset >= chunk.offset && end <= chunk.offset + chunk.bytes.len() as u64).then(
217                    || {
218                        &chunk.bytes
219                            [(offset - chunk.offset) as usize..(end - chunk.offset) as usize]
220                    },
221                )
222            })
223            .ok_or(())
224    }
225    fn read_bytes_at_until(
226        self,
227        range: Range<u64>,
228        delimiter: u8,
229    ) -> std::result::Result<&'a [u8], ()> {
230        if range.start > range.end || range.end > self.identity.2 {
231            return Err(());
232        }
233        let chunk = self
234            .chunks
235            .iter()
236            .find(|chunk| {
237                range.start >= chunk.offset && range.start < chunk.offset + chunk.bytes.len() as u64
238            })
239            .ok_or(())?;
240        let end = range.end.min(chunk.offset + chunk.bytes.len() as u64);
241        let bytes = chunk
242            .bytes
243            .get((range.start - chunk.offset) as usize..(end - chunk.offset) as usize)
244            .ok_or(())?;
245        let length = bytes.iter().position(|byte| *byte == delimiter).ok_or(())?;
246        Ok(&bytes[..length])
247    }
248}
