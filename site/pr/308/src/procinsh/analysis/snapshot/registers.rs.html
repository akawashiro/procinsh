35fn optional_hex<S: serde::Serializer>(
36    value: &Option<u64>,
37    serializer: S,
38) -> Result<S::Ok, S::Error> {
39    match value {
40        Some(value) => serializer.serialize_some(&format!("0x{value:x}")),
41        None => serializer.serialize_none(),
42    }

65pub fn classify(name: &str, role: RegisterRole, value: u64, maps: &[MemoryMap]) -> Register {
66    let map = maps.iter().find(|m| m.contains(value));
67    Register {
68        name: name.into(),
69        role,
70        value,
71        decimal: value.to_string(),
72        kind: map.map_or(MemoryKind::Integer, MemoryMap::kind),
73        mapping: map.map(|m| RegisterMapping {
74            pathname: m.pathname.clone(),
75            readable: m.readable,
76            writable: m.writable,
77            executable: m.executable,
78            private: m.private,
79        }),
80        offset: map.map(|m| value - m.start),
81    }
82}

106pub fn from_sample(r: &RegisterSet, maps: &[MemoryMap]) -> Vec<Register> {
107    REGISTERS
108        .iter()
109        .filter(|&&(_, role, _)| role != RegisterRole::Flags || r.flags_available)
110        .map(|&(name, role, index)| classify(name, role, r.values[index], maps))
111        .collect()
112}

194    #[cfg(target_arch = "x86_64")]
195    #[test]
196    fn x86_64_sample_keeps_order_and_marks_pc_sp_fp_and_flags() {
197        let registers = from_sample(
198            &RegisterSet {
199                flags_available: true,
200                ..RegisterSet::default()
201            },
202            &[],
203        );
204        let roles: Vec<_> = registers
205            .iter()
206            .filter(|r| r.role != RegisterRole::General)
207            .map(|r| (r.name.as_str(), r.role))
208            .collect();
209        assert_eq!(
210            roles,
211            [
212                ("RIP", RegisterRole::Pc),
213                ("RSP", RegisterRole::Sp),
214                ("RBP", RegisterRole::Fp),
215                ("RFLAGS", RegisterRole::Flags),
216            ]
217        );
218        assert_eq!(registers.len(), 18);
219        assert_eq!(registers[0].name, "RIP");
220        assert_eq!(registers[17].name, "RFLAGS");
221    }

223    #[test]
224    fn register_mapping_keeps_fields_and_offset_or_integer() {
225        let map = crate::http_server::model::memory::parse_map(
226            "1000-2000 rw-p 00000000 08:01 42 /tmp/a [b]",
227        )
228        .unwrap();
229        let register = classify("RAX", RegisterRole::General, 0x1008, &[map]);
230        assert_eq!(register.kind, MemoryKind::File);
231        assert_eq!(register.offset, Some(8));
232        let json = serde_json::to_value(register).unwrap();
233        assert_eq!(
234            json["mapping"],
235            serde_json::json!({"pathname":"/tmp/a [b]","readable":true,"writable":true,"executable":false,"private":true})
236        );
237        assert_eq!(json["offset"], "0x8");
238        assert_eq!(json["role"], "general");
239        let integer = classify("RAX", RegisterRole::General, u64::MAX, &[]);
240        assert_eq!(integer.kind, MemoryKind::Integer);
241        assert!(integer.mapping.is_none() && integer.offset.is_none());
242        assert_eq!(
243            serde_json::to_value(integer).unwrap()["offset"],
244            serde_json::Value::Null
245        );
246    }
247    #[test]
248    fn native_registers_round_trip_and_reject_wrong_counts() {
249        let mut registers = RegisterSet {
250            flags_available: true,
251            code_pac_mask: Some(0xff00_0000_0000_0000),
252            ..RegisterSet::default()
253        };
254        registers.values = (0..RegisterSet::COUNT).map(|i| i as u64 + 100).collect();
255        let encoded = crate::ipc::codec::encode(&registers, 4096).unwrap();
256        let decoded: RegisterSet = crate::ipc::codec::decode(&encoded).unwrap();
257        assert_eq!(decoded.values, registers.values);
258        assert_eq!(decoded.code_pac_mask, registers.code_pac_mask);
259        assert!(decoded.flags_available);
260        let json = serde_json::to_value(&registers).unwrap();
261        let decoded: RegisterSet = serde_json::from_value(json.clone()).unwrap();
262        assert_eq!(decoded.pc(), 100 + RegisterSet::PC as u64);
263        let mut invalid = json;
264        invalid["values"].as_array_mut().unwrap().pop();
265        assert!(serde_json::from_value::<RegisterSet>(invalid).is_err());
266        let mut invalid = encoded;
267        invalid[..4].copy_from_slice(&((RegisterSet::COUNT - 1) as u32).to_ne_bytes());
268        assert!(crate::ipc::codec::decode::<RegisterSet>(&invalid).is_err());
269    }

273    fn arm64_names_roles_and_optional_pstate() {
274        let mut r = RegisterSet {
275            flags_available: true,
276            ..RegisterSet::default()
277        };
278        r.values = (0..34).collect();
279        let displayed = from_sample(&r, &[]);

282            ("PC", RegisterRole::Pc, 32),
283            ("SP", RegisterRole::Sp, 31),
284            ("X29", RegisterRole::Fp, 29),
285            ("X30", RegisterRole::Lr, 30),
286            ("PSTATE", RegisterRole::Flags, 33),
287        ] {
288            let register = displayed.iter().find(|r| r.name == name).unwrap();
289            assert_eq!((register.role, register.value), (role, value));
290        }
291        r.flags_available = false;
292        let displayed = from_sample(&r, &[]);
293        assert_eq!(displayed.len(), 33);
294        assert!(displayed.iter().all(|r| r.name != "PSTATE"));
295    }
