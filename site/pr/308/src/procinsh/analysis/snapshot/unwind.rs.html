28#[derive(Default)]
29/// Unwinds sampled stacks using cached rules and executable mapping metadata.
30pub struct StackUnwinder {
31    unwinder: NativeUnwinder<Arc<[u8]>>,
32    cache: NativeCache<framehop::MayAllocateDuringUnwind>,
33    mappings: Vec<Mapping>,
34    modules: Vec<(u64, u64, bool)>,
35}
36
37impl StackUnwinder {
38    /// Rebuild on executable-map changes, including unload/reload and ASLR.
39    /// Failed acquisitions remain diagnostic until executable mappings change.
40    pub fn refresh(&mut self, maps: &[MemoryMap], symbols: &ElfCache) {
41        let mappings: Vec<_> = maps
42            .iter()
43            .filter(|m| m.executable)
44            .map(|m| {
45                (
46                    m.start,
47                    m.end,
48                    m.file_offset,
49                    m.inode,
50                    m.device.major,
51                    m.device.minor,
52                )
53            })
54            .collect();
55        if mappings == self.mappings {
56            return;
57        }
58        self.unwinder = NativeUnwinder::new();
59        self.cache = NativeCache::new();
60        self.modules.clear();
61        self.mappings = mappings;
62        for map in maps.iter().filter(|m| m.executable) {
63            let Some(elf) = symbols.get(map) else {
64                continue;
65            };
66            let Some(svma) = elf_address(map.start, map, &elf, unsafe {
67                libc::sysconf(libc::_SC_PAGESIZE) as u64
68            }) else {
69                continue;
70            };
71            let Some(base) = map.start.checked_sub(svma) else {
72                continue;
73            };
74            let metadata = elf.unwind.eh_frame.is_some() || elf.unwind.debug_frame.is_some();
75            self.unwinder.add_module(Module::new(
76                map.pathname.clone().unwrap_or_default(),
77                map.start..map.end,
78                base,
79                elf.unwind.clone(),
80            ));
81            self.modules.push((map.start, map.end, metadata));
82        }
83    }
84
85    /// Unwinds from the supplied register values and captured stack bytes.
86    pub fn walk(
87        &mut self,
88        registers: &RegisterSet,
89        maps: &[MemoryMap],
90        stack: &[u8],
91    ) -> (Vec<StackFrame>, String) {
92        let (rip, rsp, rbp) = (registers.pc(), registers.sp(), registers.fp());
93        let mut frames = Vec::new();
94        if !maps.iter().any(|m| m.readable && m.contains(rsp)) {
95            return (
96                vec![StackFrame::raw(rip)],
97                "SP is outside readable mappings".into(),
98            );
99        }
100        let mut read = |address: u64| -> Result<u64, ()> {
101            let offset = usize::try_from(address.checked_sub(rsp).ok_or(())?).map_err(|_| ())?;
102            let bytes = stack
103                .get(offset..offset.checked_add(8).ok_or(())?)
104                .ok_or(())?;
105            Ok(u64::from_ne_bytes(bytes.try_into().map_err(|_| ())?))
106        };
107        #[cfg(target_arch = "x86_64")]
108        let regs = UnwindRegsX86_64::new(rip, rsp, rbp);
109        #[cfg(target_arch = "aarch64")]
110        let regs = UnwindRegsAarch64::new_with_ptr_auth_mask(
111            registers
112                .code_pac_mask
113                .map_or(PtrAuthMask::new_no_strip(), |mask| PtrAuthMask(!mask)),
114            registers.lr(),
115            rsp,
116            rbp,
117        );
118        let mut iter = self
119            .unwinder
120            .iter_frames(rip, regs, &mut self.cache, &mut read);
121        let mut diagnostic = None;
122        let reason = loop {
123            match iter.next() {
124                Ok(Some(frame)) => {
125                    let address = frame.address();
126                    let lookup = frame.address_for_lookup();
127                    if cfg!(target_arch = "aarch64") && !address.is_multiple_of(4) {
128                        if frames.is_empty() {
129                            frames.push(StackFrame::raw(address));
130                        }
131                        break "unaligned AArch64 instruction address".into();
132                    }
133                    if !maps.iter().any(|m| m.executable && m.contains(lookup)) {
134                        if frames.is_empty() {
135                            frames.push(StackFrame::raw(address));
136                        }
137                        break "address outside executable mappings".into();
138                    }
139                    match self
140                        .modules
141                        .iter()
142                        .find(|&&(start, end, _)| start <= lookup && lookup < end)
143                    {
144                        None => diagnostic = Some("module lookup failure; frame-pointer fallback"),
145                        Some((_, _, false)) => {
146                            diagnostic = Some("unwind metadata unavailable; frame-pointer fallback")
147                        }
148                        _ => {}
149                    }
150                    frames.push(StackFrame::raw(address));
151                    if frames.len() >= 256 {
152                        break "256-frame limit".into();
153                    }
154                }
155                Ok(None) => break "end of stack".into(),
156                Err(framehop::Error::CouldNotReadStack(address)) => {
157                    let end = rsp.saturating_add(stack.len() as u64);
158                    break if address >= rsp && address.saturating_add(8) > end {
159                        format!("stack snapshot exhausted at 0x{address:x}")
160                    } else {
161                        format!("stack address outside snapshot at 0x{address:x}")
162                    };
163                }
164                Err(error) => break format!("unwind failed: {error}"),
165            }
166        };
167        let reason = match diagnostic {
168            Some(diagnostic) => format!("{reason}; {diagnostic}"),
169            None => reason,
170        };
171        (frames, reason)
172    }
173}

181    // Capture actual native call-site state, exercising CFI without assuming a
182    // compiler's prologue instructions or instruction-set-specific stack layout.
183    #[test]
184    fn omitted_frame_pointer_snapshot_reaches_main() {
185        for name in [
186            "recursive_no_fp",
187            "recursive_no_fp_nopie",
188            "recursive_debug_frame",
189        ] {
190            let target = TestTarget::new(name);
191            let maps = process::maps::read_maps(target.id.pid).unwrap();
192            let mut symbols = ElfCache::default();
193            for map in maps.iter().filter(|map| map.executable && map.inode != 0) {
194                let _ = process::load(&mut symbols, target.id.pid, map);
195            }
196            let map = maps
197                .iter()
198                .find(|m| m.executable && m.pathname.as_deref().is_some_and(|p| p.ends_with(name)))
199                .unwrap();
200            let elf = process::load(&mut symbols, target.id.pid, map).unwrap();
201            let base = map.start - elf_address(map.start, map, &elf, procfs::page_size()).unwrap();
202            let bytes = std::fs::read(format!("tests/targets/bin/{name}")).unwrap();
203            let file = object::File::parse(bytes.as_slice()).unwrap();
204            let expected: Vec<_> = ["baz", "bar", "foo", "main"]
205                .map(|name| {
206                    base + file
207                        .symbols()
208                        .find(|s| s.name().ok() == Some(name))
209                        .unwrap()
210                        .address()
211                })
212                .into();
213            let (registers, stack) = process::capture_registers(target.id.pid, &maps).unwrap();
214            for indexed in [true, false] {
215                let mut state = StackUnwinder::default();
216                state.refresh(&maps, &symbols);
217                if !indexed {
218                    let mut sections = elf.unwind.clone();
219                    sections.eh_frame_hdr = None;
220                    sections.eh_frame_hdr_svma = None;
221                    state.unwinder.add_module(Module::new(
222                        name.into(),
223                        map.start..map.end,
224                        base,
225                        sections,
226                    ));
227                }
228                let (frames, stop) = state.walk(&registers, &maps, &stack);
229                assert!(frames.len() >= 4, "{name}: {frames:?}, {stop}");
230                for (frame, start) in frames.iter().zip(&expected) {
231                    assert!(
232                        frame.address >= *start && frame.address - start < 256,
233                        "{name}: {frames:?}"
234                    );
235                }
236                let (_, short_stop) = state.walk(&registers, &maps, &[]);
237                assert!(
238                    short_stop.contains("stack snapshot exhausted"),
239                    "{short_stop}"
240                );
241            }
242            let mut state = StackUnwinder::default();
243            state.refresh(&maps, &symbols);
244            assert!(!state.modules.is_empty());
245            state.refresh(&[], &symbols);
246            assert!(state.modules.is_empty());
247            target.assert_detached();
248        }
249    }

251    fn registers(pc: u64, sp: u64, fp: u64) -> RegisterSet {
252        let mut r = RegisterSet::default();
253        r.values[RegisterSet::PC] = pc;
254        r.values[RegisterSet::SP] = sp;
255        r.values[RegisterSet::FP] = fp;
256        r
257    }

259    #[test]
260    fn fallback_reports_module_failure_and_bounds_stack_reads() {
261        let maps = vec![
262            process::maps::parse_map("1000-2000 rw-p 0 00:00 0 [stack]").unwrap(),
263            process::maps::parse_map("3000-4000 r-xp 0 00:00 0 /missing").unwrap(),
264        ];
265        let mut state = StackUnwinder::default();
266        let (_, reason) = state.walk(&registers(0x3010, 0x1000, 0x1100), &maps, &[]);
267        assert!(reason.contains("snapshot exhausted"), "{reason}");
268        assert!(reason.contains("module lookup failure"), "{reason}");
269        state.modules.push((0x3000, 0x4000, false));
270        let (_, reason) = state.walk(&registers(0x3010, 0x1000, 0x1100), &maps, &[]);
271        assert!(reason.contains("unwind metadata unavailable"), "{reason}");
272        let (frames, reason) = state.walk(&registers(0x3010, 0x2000, 0), &maps, &[]);
273        assert_eq!(frames.len(), 1);
274        assert!(reason.contains("SP"));
275    }

278    fn kernel_pac_mask_strips_saved_return_addresses_without_guessing() {
279        let maps = vec![
280            process::maps::parse_map("1000-2000 rw-p 0 00:00 0 [stack]").unwrap(),
281            process::maps::parse_map("3000-4000 r-xp 0 00:00 0 /missing").unwrap(),

285        stack[16..24].copy_from_slice(&0x1020u64.to_ne_bytes());
286        stack[24..32].copy_from_slice(&0xab00_0000_0000_3020u64.to_ne_bytes());
287        let (frames, reason) = StackUnwinder::default().walk(&r, &maps, &stack);
288        assert_eq!(frames.len(), 1);
289        assert!(reason.contains("outside executable"), "{reason}");
290        r.code_pac_mask = Some(0xff00_0000_0000_0000);
291        let (frames, reason) = StackUnwinder::default().walk(&r, &maps, &stack);
292        assert_eq!(
293            frames.iter().map(|f| f.address).collect::<Vec<_>>(),
294            [0x3010, 0x3020],
295            "{reason}"
296        );
