101/// `Key: value` lines of `/proc/driver/nvidia/gpus/<address>/information`.
102fn nvidia_information(
103    sources: &HardwareSources,
104    proc: &Path,
105    address: &str,
106) -> Vec<(String, String)> {
107    sources
108        .text(
109            &proc
110                .join("driver/nvidia/gpus")
111                .join(address)
112                .join("information"),
113        )
114        .unwrap_or_default()
115        .lines()
116        .filter_map(|line| {
117            let (key, value) = line.split_once(':')?;
118            Some((key.trim().to_owned(), value.trim().to_owned()))
119        })
120        .collect()
121}
122
123fn short(address: &str) -> &str {
124    address.strip_prefix("0000:").unwrap_or(address)
125}

127pub fn device(sources: &HardwareSources, node: &PciNode, proc: &Path) -> HardwareDevice {
128    let hex = |value: Option<u32>, width: usize| value.map(|v| format!("0x{v:0width$x}"));
129    let gpu = (node.kind == Some(HardwareKind::Gpu)).then(|| {
130        let info = nvidia_information(sources, proc, &node.address);
131        let field = |key: &str| info.iter().find(|(k, _)| k == key).map(|(_, v)| v.clone());
132        GpuInfo {
133            vendor: match node.vendor_id {
134                Some(0x10de) => "nvidia".into(),
135                Some(0x1002) => "amd".into(),
136                Some(0x8086) => "intel".into(),
137                other => hex(other, 4).unwrap_or_default(),
138            },
139            model: field("Model"),
140            uuid: field("GPU UUID"),
141            minor: field("Device Minor").and_then(|m| m.parse().ok()),
142            drm: sources
143                .names(&node.path.join("drm"))
144                .into_iter()
145                .filter(|n| n.starts_with("card") || n.starts_with("renderD"))
146                .filter(|n| !n.contains('-'))
147                .collect(),
148        }
149    });
150    let rdma: Vec<_> = sources
151        .names(&node.path.join("infiniband"))
152        .into_iter()
153        .map(|name| {
154            let dir = node.path.join("infiniband").join(&name);
155            RdmaDeviceInfo {
156                hca_type: sources.text(&dir.join("hca_type")),
157                fw_ver: sources.text(&dir.join("fw_ver")),
158                name,
159            }
160        })
161        .collect();
162    let label = match (node.role, node.kind) {
163        (HardwareRole::HostBridge, _) => {
164            format!("Host bridge {}", node.address.trim_start_matches("pci"))
165        }
166        (HardwareRole::RootPort, _) => format!("Root port {}", short(&node.address)),
167        (HardwareRole::SwitchPort, _) => format!("Switch port {}", short(&node.address)),
168        (_, Some(HardwareKind::Gpu)) => gpu
169            .as_ref()
170            .and_then(|g| g.model.clone())
171            .unwrap_or_else(|| format!("GPU {}", short(&node.address))),
172        (_, Some(HardwareKind::Rdma)) if !rdma.is_empty() => rdma
173            .iter()
174            .map(|d| d.name.as_str())
175            .collect::<Vec<_>>()
176            .join(", "),
177        (_, Some(HardwareKind::Rdma)) => format!("RDMA NIC {}", short(&node.address)),
178        (_, Some(HardwareKind::Accelerator)) => format!("Accelerator {}", short(&node.address)),
179        (_, Some(HardwareKind::NvSwitch)) => format!("NVSwitch {}", short(&node.address)),
180        (_, None) => short(&node.address).to_owned(),
181    };
182    HardwareDevice {
183        address: node.address.clone(),
184        parent: node.parent.clone(),
185        role: node.role,
186        kind: node.kind,
187        label,
188        class: hex(node.class, 6),
189        vendor_id: hex(node.vendor_id, 4),
190        device_id: hex(node.device_id, 4),
191        driver: node.driver.clone(),
192        numa_node: node.numa_node,
193        link: node.link,
194        power_state: node.power_state.clone(),
195        runtime_status: node.runtime_status.clone(),
196        iommu_group: node.iommu_group,
197        bars: node.bars.clone(),
198        gpu,
199        rdma,
200        netdevs: if node.role == HardwareRole::Endpoint {
201            sources.names(&node.path.join("net"))
202        } else {
203            vec![]
204        },
205    }
206}
