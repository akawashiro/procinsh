19struct ProcessHistory {
20    sample: crate::http_server::model::process::ProcessSample,
21    observation: crate::http_server::model::process::ProcessObservation,
22    points: std::collections::VecDeque<crate::http_server::analysis::history::HistoryPoint>,
23}
24/// Owns analysis caches and independent per-session histories and BPF deltas.
25/// IPC-only capture fields are consumed before any HTTP serialization.
26#[derive(Default)]
27pub(super) struct ObservationProcessor {
28    histories: HashMap<(u64, ProcessId), ProcessHistory>,
29    activity: HashMap<u64, crate::http_server::analysis::activity::ActivityProcessor>,
30    known: HashMap<i32, ProcessId>,
31    discovery: HashMap<ProcessId, (u64, u64)>,
32    symbols: ElfCache,
33    loaded: HashSet<(ProcessId, u64, u64, u64)>,
34    unwinders: HashMap<ProcessId, StackUnwinder>,
35}
36impl ObservationProcessor {
37    pub(super) fn close(&mut self, session: u64) {
38        let processes: Vec<_> = self
39            .histories
40            .keys()
41            .filter(|(key, _)| *key == session)
42            .map(|(_, id)| *id)
43            .collect();
44        self.histories.retain(|(key, _), _| *key != session);
45        for id in processes {
46            if !self.histories.keys().any(|(_, other)| *other == id) {
47                self.unwinders.remove(&id);
48                self.loaded.retain(|(process, _, _, _)| *process != id);
49            }
50        }
51        self.activity.remove(&session);
52    }
53
54    fn snapshot(
55        &mut self,
56        mut snapshot: crate::http_server::model::system::SystemSnapshot,
57    ) -> Result<Value> {
58        self.known = snapshot
59            .processes
60            .iter()
61            .map(|process| (process.identity.pid, process.identity))
62            .collect();
63        if crate::http_server::analysis::relations::build(&mut snapshot) {
64            snapshot.warnings.push("Scan, mapping, or connection limit reached. Some process or FD information is missing.".into());
65        }
66        let sources = crate::http_server::analysis::hardware::HardwareSources::acquired(
67            std::mem::take(&mut snapshot.raw_hardware),
68        );
69        let hardware =
70            crate::http_server::analysis::hardware::HardwareObservation::observe(&sources);
71        let mut handles = hardware.handles(&sources);
72        for (process, fd, rdev) in std::mem::take(&mut snapshot.raw_device_handles) {
73            handles.observe(process, fd, rdev);
74        }
75        let live = snapshot
76            .processes
77            .iter()
78            .map(|process| process.identity)
79            .collect();
80        snapshot.device_handles = handles.finish(&live);
81        snapshot.hardware_devices = hardware.devices;
82        snapshot.hardware_paths = hardware.paths;
83        let hostnames: HashMap<_, _> = std::mem::take(&mut snapshot.raw_hostnames)
84            .into_iter()
85            .collect();
86        for relation in &mut snapshot.fd_relations {
87            if let Some(socket) = &mut relation.socket
88                && socket.network_peer
89            {
90                socket.remote_hostname = socket
91                    .remote
92                    .and_then(|address| hostnames.get(&address.ip).cloned());
93            }
94        }
95        Ok(serde_json::to_value(snapshot)?)
96    }
97    pub(super) async fn process(
98        &mut self,
99        client: &MonitorClient,
100        session: Option<u64>,
101        response: &mut MonitorResponse,
102    ) -> Result<()> {
103        let session = match response {
104            MonitorResponse::Started { session, .. } => Some(*session),
105            _ => session,
106        };
107        let mut snapshot_processed = false;
108        let event = match response {
109            MonitorResponse::Started { initial, .. } => Some(initial),
110            MonitorResponse::Event {
111                event: Some(event), ..
112            } => Some(event),
113            _ => None,
114        };
115        if let Some(event) = event {
116            let acquired = matches!(
117                event,
118                MonitorEvent::AcquiredSnapshot(_) | MonitorEvent::AcquisitionGap { .. }
119            );
120            if acquired {
121                let raw = std::mem::replace(event, MonitorEvent::Snapshot(Value::Null));
122                *event = match raw {
123                    MonitorEvent::AcquiredSnapshot(snapshot) => MonitorEvent::Snapshot(
124                        self.snapshot(std::sync::Arc::unwrap_or_clone(snapshot))?,
125                    ),
126                    MonitorEvent::AcquisitionGap { dropped, snapshot } => MonitorEvent::Gap {
127                        dropped,
128                        snapshot: self.snapshot(std::sync::Arc::unwrap_or_clone(snapshot))?,
129                    },
130                    _ => unreachable!(),
131                };
132                snapshot_processed = true;
133            }
134        }
135        let value = match response {
136            MonitorResponse::Data(value) => value,
137            MonitorResponse::Started {
138                initial:
139                    MonitorEvent::Observation(value)
140                    | MonitorEvent::Snapshot(value)
141                    | MonitorEvent::Activity(value),
142                ..
143            }
144            | MonitorResponse::Event {
145                event:
146                    Some(
147                        MonitorEvent::Observation(value)
148                        | MonitorEvent::Snapshot(value)
149                        | MonitorEvent::Activity(value),
150                    ),
151                ..
152            } => value,
153            MonitorResponse::Started {
154                initial: MonitorEvent::Gap { snapshot, .. },
155                ..
156            }
157            | MonitorResponse::Event {
158                event: Some(MonitorEvent::Gap { snapshot, .. }),
159                ..
160            } => snapshot,
161            _ => return Ok(()),
162        };
163        crate::http_server::analysis::details::complete(value)?;
164        let name = |record: &mut Value| {
165            if let Some(object) = record.as_object_mut() {
166                object.remove("raw_ticks");
167                object.remove("raw_measured_ns");
168            }
169        };
170        if let Some(list) = value.as_array_mut() {
171            let mut next = HashMap::new();
172            for record in list.iter_mut() {
173                if let (Ok(id), Some(ticks), Some(now)) = (
174                    serde_json::from_value::<ProcessId>(record["identity"].clone()),
175                    record["raw_ticks"].as_u64(),
176                    record["raw_measured_ns"].as_u64(),
177                ) {
178                    let cpu = self
179                        .discovery
180                        .get(&id)
181                        .and_then(|(old, then)| ticks.checked_sub(*old).zip(now.checked_sub(*then)))
182                        .filter(|(_, elapsed)| *elapsed > 0)
183                        .map(|(delta, elapsed)| {
184                            delta as f64
185                                / crate::http_server::model::process::ticks_per_second()
186                                / (elapsed as f64 / 1_000_000_000.0)
187                                * 100.0
188                        });
189                    record["cpu_percent"] = serde_json::json!(cpu);
190                    next.insert(id, (ticks, now));
191                }
192                name(record);
193            }
194            if !next.is_empty() {
195                self.discovery = next;
196                list.sort_by(|a, b| {
197                    b["cpu_percent"]
198                        .as_f64()
199                        .unwrap_or_default()
200                        .total_cmp(&a["cpu_percent"].as_f64().unwrap_or_default())
201                        .then(
202                            a["identity"]["pid"]
203                                .as_i64()
204                                .cmp(&b["identity"]["pid"].as_i64()),
205                        )
206                });
207            }
208        }
209        if let Some(summary) = value.get_mut("summary") {
210            name(summary);
211        }
212        if let Some(processes) = value.get_mut("processes").and_then(Value::as_array_mut) {
213            for process in processes {
214                name(process);
215            }
216        }
217        if let Some(raw) = value
218            .as_object_mut()
219            .and_then(|value| value.remove("raw_activity"))
220            && !raw.is_null()
221        {
222            let raw = serde_json::from_value(raw)?;
223            let session = session.unwrap_or_default();
224            if self.activity.len() >= 64 && !self.activity.contains_key(&session) {
225                self.activity.clear();
226            }
227            let batch = self
228                .activity
229                .entry(session)
230                .or_default()
231                .process(raw, &self.known);
232            value["files"] = serde_json::to_value(batch.files)?;
233            value["ipc"] = serde_json::to_value(batch.ipc)?;
234            value["signals"] = serde_json::to_value(batch.signals)?;
235            value["cpu"] = serde_json::to_value(batch.cpu)?;
236            value["status"]["unresolved"] = serde_json::json!(batch.unresolved);
237            value["status"]["files_lost"] = serde_json::json!(
238                value["status"]["files_lost"]
239                    .as_u64()
240                    .unwrap_or_default()
241                    .saturating_add(batch.files_dropped)
242            );
243        }
244        if let Some(raw) = value
245            .as_object_mut()
246            .and_then(|value| value.remove("process_sample"))
247        {
248            use crate::http_server::model::process::{
249                ProcessSample, initial_observation, next_observation,
250            };
251            let current: ProcessSample = serde_json::from_value(raw)?;
252            let mut observed = initial_observation(&current);
253            if value.get("summary").is_some() {
254                let key = (session.unwrap_or_default(), current.process_id);
255                if self.histories.len() >= 64 && !self.histories.contains_key(&key) {
256                    self.histories.clear();
257                }
258                let history = self.histories.entry(key).or_insert_with(|| ProcessHistory {
259                    sample: current.clone(),
260                    observation: observed.clone(),
261                    points: Default::default(),
262                });
263                if current.measured_at > history.sample.measured_at {
264                    observed = next_observation(&history.sample, &current);
265                } else {
266                    observed = history.observation.clone();
267                }
268                if history
269                    .points
270                    .back()
271                    .is_none_or(|point| point.timestamp < current.timestamp)
272                {
273                    crate::http_server::analysis::history::push(&mut history.points, &observed);
274                }
275                history.sample = current;
276                history.observation = observed.clone();
277                value["history"] = serde_json::to_value(&history.points)?;
278                value["observation"] = serde_json::to_value(observed)?;
279            } else {
280                let threads_only = value["threads_only"] == true;
281                *value = if threads_only {
282                    serde_json::to_value(observed.threads)?
283                } else {
284                    serde_json::to_value(observed)?
285                };
286            }
287        }
288        if value.get("raw_fd_owners").is_some() || value.get("raw_peer_keys").is_some() {
289            let mut descriptors: crate::http_server::analysis::descriptors::FileDescriptors =
290                serde_json::from_value(value.take())?;
291            crate::http_server::analysis::descriptors::complete(&mut descriptors);
292            *value = serde_json::to_value(descriptors)?;
293            value.as_object_mut().unwrap().remove("raw_fd_owners");
294            value.as_object_mut().unwrap().remove("raw_peer_keys");
295            for entry in value["entries"].as_array_mut().unwrap() {
296                for field in ["key", "mode", "socket_device"] {
297                    entry.as_object_mut().unwrap().remove(field);
298                }
299            }
300        }
301        if !snapshot_processed
302            && value.get("processes").is_some()
303            && value.get("fd_relations").is_some()
304        {
305            *value = self.snapshot(serde_json::from_value(value.take())?)?;
306        }
307        let contexts: Vec<MappingSnapshot> = value
308            .as_object_mut()
309            .and_then(|value| value.remove("mapping_snapshots"))
310            .map(serde_json::from_value)
311            .transpose()?
312            .unwrap_or_default();
313        let contexts: HashMap<_, _> = contexts
314            .into_iter()
315            .map(|context| (context.epoch, context.maps))
316            .collect();
317        if value
318            .get("live_samples")
319            .and_then(Value::as_array)
320            .is_none()
321        {
322            return Ok(());
323        }
324        let id: ProcessId = serde_json::from_value(value["summary"]["identity"].clone())?;
325        let epoch = value["maps_captured_at"].as_u64().unwrap_or_default();
326        let mut maps: Vec<MemoryMap> = contexts.values().flatten().cloned().collect();
327        maps.sort_by_key(|map| map.start);
328        maps.dedup_by_key(|map| (map.start, map.inode));
329        if self.loaded.len() > 65536 {
330            self.loaded.clear();
331        }
332        for map in maps.iter().filter(|map| map.executable && map.inode != 0) {
333            if !self.loaded.insert((id, epoch, map.start, map.inode)) {
334                continue;
335            }
336            if let Ok(Some(file)) =
337                super::elf_reader::acquire(client, id, map.start, self.symbols.identity(map)).await
338            {
339                self.symbols.insert_image(map, &file);
340            }
341        }
342        if self.unwinders.len() >= 64 && !self.unwinders.contains_key(&id) {
343            self.unwinders.clear();
344        }
345        let unwinder = self.unwinders.entry(id).or_default();
346        for sample in value["live_samples"].as_array_mut().unwrap() {
347            let raw = sample.as_object_mut().unwrap();
348            if let Some(time) = raw.remove("sample_time_ns").and_then(|time| time.as_u64()) {
349                raw.insert(
350                    "sample_age_ms".into(),
351                    serde_json::json!(
352                        crate::http_server::model::process::monotonic_ns().saturating_sub(time)
353                            / 1_000_000
354                    ),
355                );
356            }
357            let registers = raw.remove("raw_registers").unwrap_or(Value::Null);
358            let stack = raw.remove("stack_bytes").unwrap_or(Value::Null);
359            let epoch = raw.remove("map_epoch").and_then(|epoch| epoch.as_u64());
360            if registers.is_null() {
361                continue;
362            }
363            if raw.get("architecture").and_then(Value::as_str) != Some(std::env::consts::ARCH) {
364                let message = "Only native 64-bit process snapshots are supported";
365                let error = raw
366                    .get("error")
367                    .and_then(Value::as_str)
368                    .map_or_else(|| message.to_owned(), |error| format!("{error}; {message}"));
369                raw.insert("error".into(), Value::String(error));
370                raw.insert("registers".into(), serde_json::json!([]));
371                raw.insert("call_stack".into(), serde_json::json!([]));
372                raw.insert("unwind_stop".into(), Value::String(message.into()));
373                raw.insert("disassembly".into(), Value::Null);
374                continue;
375            }
376            let registers: RegisterSet = serde_json::from_value(registers)?;
377            let stack: Vec<u8> = serde_json::from_value(stack)?;
378            let maps = epoch
379                .and_then(|epoch| contexts.get(&epoch))
380                .map(Vec::as_slice)
381                .unwrap_or(&[]);
382            unwinder.refresh(maps, &self.symbols);
383            let (mut frames, reason) = unwinder.walk(&registers, maps, &stack);
384            for (index, frame) in frames.iter_mut().enumerate() {
385                let address = instruction_address(frame.address, index > 0);
386                let info = maps
387                    .iter()
388                    .find(|map| map.contains(address) && map.inode != 0)
389                    .and_then(|map| {
390                        let elf = self.symbols.get(map)?;
391                        resolve_frame(
392                            elf_address(address, map, &elf, unsafe {
393                                libc::sysconf(libc::_SC_PAGESIZE) as u64
394                            })?,
395                            &elf,
396                        )
397                    })
398                    .unwrap_or_default();
399                frame.symbol = info.name;
400                frame.symbol_offset = info.offset.map(|offset| format!("0x{offset:x}"));
401                frame.source_file = info.file;
402                frame.line = info.line;
403                frame.inline_frames = info.inline_frames;
404            }
405            raw.insert(
406                "registers".into(),
407                serde_json::to_value(from_sample(&registers, maps))?,
408            );
409            raw.insert("call_stack".into(), serde_json::to_value(frames)?);
410            raw.insert("unwind_stop".into(), Value::String(reason));
411        }
412        Ok(())
413    }

425    #[tokio::test]
426    async fn retained_samples_use_their_captured_mapping_epoch() {
427        use crate::http_server::model::memory::parse_map;
428        let mut processor = ObservationProcessor::default();
429        let mut registers = RegisterSet::default();
430        registers.values[0] = 0x1008;
431        registers.values[RegisterSet::PC] = 0x1008;
432        registers.values[RegisterSet::SP] = 0x1000;
433        let sample = |epoch| json!({"architecture":std::env::consts::ARCH,"raw_registers":registers,"stack_bytes":[],"map_epoch":epoch,"sample_time_ns":crate::http_server::model::process::monotonic_ns().saturating_sub(3_000_000_000)});
434        let mut response = MonitorResponse::Data(json!({
435            "summary":{"identity":{"pid":42,"start_time_ticks":10}},
436            "mapping_snapshots":[
437                MappingSnapshot {epoch:1,maps:vec![parse_map("1000-2000 r--p 0 00:00 0 /old").unwrap()]},
438                MappingSnapshot {epoch:2,maps:vec![parse_map("1000-2000 r--p 0 00:00 0 /new").unwrap()]}
439            ],
440            "live_samples":[sample(1),sample(2)]
441        }));
442        processor
443            .process(&MonitorClient::unavailable(), None, &mut response)
444            .await
445            .unwrap();
446        let MonitorResponse::Data(value) = response else {
447            panic!()
448        };
449        for (index, path) in [(0, "/old"), (1, "/new")] {
450            let sample = &value["live_samples"][index];
451            let rax = sample["registers"]
452                .as_array()
453                .unwrap()
454                .iter()
455                .find(|register| {
456                    register["name"]
457                        == if cfg!(target_arch = "aarch64") {
458                            "X0"
459                        } else {
460                            "RAX"
461                        }
462                })
463                .unwrap();
464            assert_eq!(rax["mapping"]["pathname"], path);
465            assert!(sample.get("map_epoch").is_none());
466            assert!(sample.get("sample_time_ns").is_none());
467            assert!(sample["sample_age_ms"].as_u64().unwrap() >= 3000);
468        }
469        assert!(value.get("mapping_snapshots").is_none());
470    }

472    #[tokio::test]
473    async fn process_histories_are_independent_and_repeated_samples_preserve_rates() {
474        let mut processor = ObservationProcessor::default();
475        let client = MonitorClient::unavailable();
476        let mut first = MonitorResponse::Started {
477            session: 1,
478            initial: MonitorEvent::Observation(observation(100, 1_000_000_000)),
479        };
480        processor.process(&client, None, &mut first).await.unwrap();
481        let mut next = MonitorResponse::Event {
482            event: Some(MonitorEvent::Observation(observation(150, 2_000_000_000))),
483            closed: false,
484        };
485        processor
486            .process(&client, Some(1), &mut next)
487            .await
488            .unwrap();
489        let MonitorResponse::Event {
490            event: Some(MonitorEvent::Observation(value)),
491            ..
492        } = next
493        else {
494            panic!()
495        };
496        assert!(value["observation"]["cpu_percent"].as_f64().unwrap() > 0.0);
497        assert_eq!(value["history"].as_array().unwrap().len(), 2);
498        assert!(value.get("process_sample").is_none());
499        assert!(value["live_samples"][0].get("raw_registers").is_none());
500        let mut repeated = MonitorResponse::Event {
501            event: Some(MonitorEvent::Observation(observation(150, 2_000_000_000))),
502            closed: false,
503        };
504        processor
505            .process(&client, Some(1), &mut repeated)
506            .await
507            .unwrap();
508        let MonitorResponse::Event {
509            event: Some(MonitorEvent::Observation(repeated)),
510            ..
511        } = repeated
512        else {
513            panic!()
514        };
515        assert_eq!(
516            repeated["observation"]["cpu_percent"],
517            value["observation"]["cpu_percent"]
518        );
519        assert_eq!(repeated["history"].as_array().unwrap().len(), 2);
520        let mut second = MonitorResponse::Started {
521            session: 2,
522            initial: MonitorEvent::Observation(observation(150, 2_000_000_000)),
523        };
524        processor.process(&client, None, &mut second).await.unwrap();
525        let MonitorResponse::Started {
526            initial: MonitorEvent::Observation(value),
527            ..
528        } = second
529        else {
530            panic!()
531        };
532        assert!(value["observation"]["cpu_percent"].is_null());
533        assert_eq!(value["history"].as_array().unwrap().len(), 1);
534        processor.close(1);
535        assert_eq!(processor.histories.len(), 1);
536    }
