19/// Logical roots and an in-memory view of helper-acquired attributes.
20#[derive(Clone)]
21pub struct HardwareSources {
22    pub sys: PathBuf,
23    pub proc: PathBuf,
24    texts: HashMap<PathBuf, String>,
25    directories: HashMap<PathBuf, Vec<String>>,
26    links: HashMap<PathBuf, PathBuf>,
27}
28impl Default for HardwareSources {
29    fn default() -> Self {
30        Self::acquired(crate::ipc::hardware::HardwareFiles::default())
31    }
32}
33impl HardwareSources {
34    pub fn acquired(files: crate::ipc::hardware::HardwareFiles) -> Self {
35        Self::at("/sys".into(), "/proc".into(), files)
36    }
37    fn at(sys: PathBuf, proc: PathBuf, files: crate::ipc::hardware::HardwareFiles) -> Self {
38        Self {
39            sys,
40            proc,
41            texts: files
42                .texts
43                .into_iter()
44                .map(|(p, v)| (p.into(), v))
45                .collect(),
46            directories: files
47                .directories
48                .into_iter()
49                .map(|(p, v)| (p.into(), v))
50                .collect(),
51            links: files
52                .links
53                .into_iter()
54                .map(|(p, v)| (p.into(), v.into()))
55                .collect(),
56        }
57    }
58    pub(super) fn text(&self, path: &std::path::Path) -> Option<String> {
59        let value = self.texts.get(path)?.trim();
60        (!value.is_empty()).then(|| value.to_owned())
61    }
62    pub(super) fn names(&self, path: &std::path::Path) -> Vec<String> {
63        self.directories.get(path).cloned().unwrap_or_default()
64    }
65    pub(super) fn link(&self, path: &std::path::Path) -> Option<PathBuf> {
66        self.links.get(path).cloned()
67    }
68    pub(super) fn is_dir(&self, path: &std::path::Path) -> bool {
69        self.directories.contains_key(path)
70    }
71    #[cfg(test)]
72    pub(super) fn fixture(sys: PathBuf, proc: PathBuf) -> Self {
73        let files = super::capture_fixture::capture(&sys, &proc);
74        Self::at(sys, proc, files)
75    }
76}

81#[derive(Default)]
82pub struct HardwareObservation {
83    pub devices: Vec<HardwareDevice>,
84    pub paths: Vec<HardwarePath>,
85}
86
87impl HardwareObservation {
88    /// Reads GPUs, RDMA NICs, accelerators and NVSwitches with their bridges and paths.
89    pub fn observe(sources: &HardwareSources) -> Self {
90        let nodes = read_topology(sources, &sources.sys);
91        Self {
92            paths: paths(&nodes, online_numa_nodes(sources, &sources.sys)),
93            devices: nodes
94                .iter()
95                .map(|n| device(sources, n, &sources.proc))
96                .collect(),
97        }
98    }
99
100    /// A collector for descriptors of this observation's devices.
101    pub fn handles(&self, sources: &HardwareSources) -> DeviceHandleCollector {
102        let addresses: HashSet<_> = self.devices.iter().map(|d| d.address.clone()).collect();
103        let minors: HashMap<_, _> = self
104            .devices
105            .iter()
106            .filter_map(|d| Some((d.gpu.as_ref()?.minor?, d.address.clone())))
107            .collect();
108        DeviceHandleCollector::new(sources, addresses, minors)
109    }
110}

125    #[test]
126    fn captured_hardware_survives_removal_of_all_source_files() {
127        let f = fixture::workstation("captured-only");
128        let sys = f.sys();
129        let proc = f.proc();
130        let files = super::super::capture_fixture::capture(&sys, &proc);
131        assert!(!files.truncated);
132        let encoded = crate::ipc::codec::encode(&files, 16 * 1024 * 1024).unwrap();
133        let captured = crate::ipc::codec::decode(&encoded).unwrap();
134        std::fs::remove_dir_all(&f.root).unwrap();
135        let sources = HardwareSources::at(sys, proc, captured);
136        let observation = HardwareObservation::observe(&sources);
137        let gpu = observation
138            .devices
139            .iter()
140            .find(|d| d.address == "0000:01:00.0")
141            .unwrap();
142        assert_eq!(
143            gpu.gpu.as_ref().unwrap().model.as_deref(),
144            Some("NVIDIA GeForce RTX 5070 Ti")
145        );
146        assert!(!observation.paths.is_empty());
147    }

149    #[test]
150    fn devices_carry_gpu_and_rdma_identity_as_json() {
151        let f = fixture::workstation("observation");
152        let observation = observe_at(&f.root);
153        let json = serde_json::to_value(&observation.devices).unwrap();
154        let gpu = json
155            .as_array()
156            .unwrap()
157            .iter()
158            .find(|d| d["address"] == "0000:01:00.0")
159            .unwrap();
160        assert_eq!(gpu["role"], "endpoint");
161        assert_eq!(gpu["kind"], "gpu");
162        assert_eq!(gpu["label"], "NVIDIA GeForce RTX 5070 Ti");
163        assert_eq!(gpu["class"], "0x030000");
164        assert_eq!(gpu["vendor_id"], "0x10de");
165        assert_eq!(gpu["gpu"]["minor"], 0);
166        assert_eq!(
167            gpu["gpu"]["drm"],
168            serde_json::json!(["card1", "renderD128"])
169        );
170        assert_eq!(
171            gpu["link"],
172            serde_json::json!({"current": {"speed_gts": 8.0, "width": 16}, "max": {"speed_gts": 32.0, "width": 16}})
173        );
174        let nic = json
175            .as_array()
176            .unwrap()
177            .iter()
178            .find(|d| d["address"] == "0000:85:00.0")
179            .unwrap();
180        assert_eq!(nic["kind"], "rdma");
181        assert_eq!(nic["label"], "ibp133s0");
182        assert_eq!(nic["rdma"][0]["hca_type"], "MT26428");
183        assert_eq!(nic["netdevs"], serde_json::json!(["ibp133s0"]));
184        let host = json
185            .as_array()
186            .unwrap()
187            .iter()
188            .find(|d| d["address"] == "pci0000:80")
189            .unwrap();
190        assert_eq!(
191            (host["role"].clone(), host["label"].clone()),
192            (
193                serde_json::json!("host_bridge"),
194                serde_json::json!("Host bridge 0000:80")
195            )
196        );
197        assert_eq!(
198            serde_json::to_value(&observation.paths).unwrap()[0]["class"],
199            "node"
200        );
201    }
