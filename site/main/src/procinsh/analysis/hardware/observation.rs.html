26/// Logical roots and an in-memory view of helper-acquired attributes.
27#[derive(Clone)]
28pub(crate) struct HardwareSources {
29    pub(crate) sys: PathBuf,
30    pub(crate) proc: PathBuf,
31    texts: HashMap<PathBuf, String>,
32    directories: HashMap<PathBuf, Vec<String>>,
33    links: HashMap<PathBuf, PathBuf>,
34}
35impl Default for HardwareSources {
36    fn default() -> Self {
37        acquired_hardware_sources(crate::ipc::hardware::HardwareFiles::default())
38    }
39}
40pub(crate) fn acquired_hardware_sources(
41    files: crate::ipc::hardware::HardwareFiles,
42) -> HardwareSources {
43    at_hardware_sources("/sys".into(), "/proc".into(), files)
44}
45
46fn at_hardware_sources(
47    sys: PathBuf,
48    proc: PathBuf,
49    files: crate::ipc::hardware::HardwareFiles,
50) -> HardwareSources {
51    HardwareSources {
52        sys,
53        proc,
54        texts: files
55            .texts
56            .into_iter()
57            .map(|(p, v)| (p.into(), v))
58            .collect(),
59        directories: files
60            .directories
61            .into_iter()
62            .map(|(p, v)| (p.into(), v))
63            .collect(),
64        links: files
65            .links
66            .into_iter()
67            .map(|(p, v)| (p.into(), v.into()))
68            .collect(),
69    }
70}
71
72pub(super) fn text_hardware_sources(
73    state: &HardwareSources,
74    path: &std::path::Path,
75) -> Option<String> {
76    let value = state.texts.get(path)?.trim();
77    (!value.is_empty()).then(|| value.to_owned())
78}
79
80pub(super) fn names_hardware_sources(
81    state: &HardwareSources,
82    path: &std::path::Path,
83) -> Vec<String> {
84    state.directories.get(path).cloned().unwrap_or_default()
85}
86
87pub(super) fn link_hardware_sources(
88    state: &HardwareSources,
89    path: &std::path::Path,
90) -> Option<PathBuf> {
91    state.links.get(path).cloned()
92}
93
94pub(super) fn hardware_sources_is_dir(state: &HardwareSources, path: &std::path::Path) -> bool {
95    state.directories.contains_key(path)
96}

113/// Reads GPUs, RDMA NICs, accelerators and NVSwitches with their bridges and paths.
114pub(crate) fn observe_hardware_observation(sources: &HardwareSources) -> HardwareObservation {
115    let nodes = read_topology(sources, &sources.sys);
116    HardwareObservation {
117        paths: build_hardware_paths(&nodes, online_numa_nodes(sources, &sources.sys)),
118        devices: nodes
119            .iter()
120            .map(|n| build_hardware_device(sources, n, &sources.proc))
121            .collect(),
122    }
123}

125/// A collector for descriptors of this observation's devices.
126pub(crate) fn handles_hardware_observation(
127    state: &HardwareObservation,
128    sources: &HardwareSources,
129) -> DeviceHandleCollector {
130    use crate::http_server::analysis::hardware::handles::create_device_handle_collector;
131
132    let addresses: HashSet<_> = state.devices.iter().map(|d| d.address.clone()).collect();
133    let minors: HashMap<_, _> = state
134        .devices
135        .iter()
136        .filter_map(|d| Some((d.gpu.as_ref()?.minor?, d.address.clone())))
137        .collect();
138    create_device_handle_collector(sources, addresses, minors)
139}

153    #[test]
154    fn captured_hardware_survives_removal_of_all_source_files() {
155        use crate::http_server::analysis::hardware::pci::fixture::{
156            proc_sysfs_fixture, sys_sysfs_fixture,
157        };
158
159        let f = fixture::workstation("captured-only");
160        let sys = sys_sysfs_fixture(&f);
161        let proc = proc_sysfs_fixture(&f);
162        let files = super::super::acquisition::collect_hardware_files(&sys, &proc);
163        assert!(!files.truncated);
164        let encoded = crate::ipc::codec::encode(&files, 16 * 1024 * 1024).unwrap();
165        let captured = crate::ipc::codec::decode(&encoded).unwrap();
166        std::fs::remove_dir_all(&f.root).unwrap();
167        let sources = super::at_hardware_sources(sys, proc, captured);
168        let observation = super::observe_hardware_observation(&sources);
169        let gpu = observation
170            .devices
171            .iter()
172            .find(|d| d.address == "0000:01:00.0")
173            .unwrap();
174        assert_eq!(
175            gpu.gpu.as_ref().unwrap().model.as_deref(),
176            Some("NVIDIA GeForce RTX 5070 Ti")
177        );
178        assert!(!observation.paths.is_empty());
179    }

181    #[test]
182    fn devices_carry_gpu_and_rdma_identity_as_json() {
183        let f = fixture::workstation("observation");
184        let observation = observe_at(&f.root);
185        let json = serde_json::to_value(&observation.devices).unwrap();
186        let gpu = json
187            .as_array()
188            .unwrap()
189            .iter()
190            .find(|d| d["address"] == "0000:01:00.0")
191            .unwrap();
192        assert_eq!(gpu["role"], "endpoint");
193        assert_eq!(gpu["kind"], "gpu");
194        assert_eq!(gpu["label"], "NVIDIA GeForce RTX 5070 Ti");
195        assert_eq!(gpu["class"], "0x030000");
196        assert_eq!(gpu["vendor_id"], "0x10de");
197        assert_eq!(gpu["gpu"]["minor"], 0);
198        assert_eq!(
199            gpu["gpu"]["drm"],
200            serde_json::json!(["card1", "renderD128"])
201        );
202        assert_eq!(
203            gpu["link"],
204            serde_json::json!({"current": {"speed_gts": 8.0, "width": 16}, "max": {"speed_gts": 32.0, "width": 16}})
205        );
206        let nic = json
207            .as_array()
208            .unwrap()
209            .iter()
210            .find(|d| d["address"] == "0000:85:00.0")
211            .unwrap();
212        assert_eq!(nic["kind"], "rdma");
213        assert_eq!(nic["label"], "ibp133s0");
214        assert_eq!(nic["rdma"][0]["hca_type"], "MT26428");
215        assert_eq!(nic["netdevs"], serde_json::json!(["ibp133s0"]));
216        let host = json
217            .as_array()
218            .unwrap()
219            .iter()
220            .find(|d| d["address"] == "pci0000:80")
221            .unwrap();
222        assert_eq!(
223            (host["role"].clone(), host["label"].clone()),
224            (
225                serde_json::json!("host_bridge"),
226                serde_json::json!("Host bridge 0000:80")
227            )
228        );
229        assert_eq!(
230            serde_json::to_value(&observation.paths).unwrap()[0]["class"],
231            "node"
232        );
233    }
