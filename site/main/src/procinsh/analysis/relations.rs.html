1//! Build FD relationships from captured endpoints and socket observations.
2//!
3//! # Interface
4//!
5//! | Definition | Visibility | Kind / signature |
6//! | --- | --- | --- |
7//! | [`build`] | `pub` | `fn build(result: &mut SystemSnapshot) -> bool` |
8//!
9//! Input and output values are defined by [`SystemSnapshot`].
10
11use crate::http_server::model::{resource::*, socket_types::*, system::*};
12use std::collections::{HashMap, HashSet};
13pub fn build(result: &mut SystemSnapshot) -> bool {
14    let mut owners: HashMap<_, _> = std::mem::take(&mut result.raw_owners).into_iter().collect();
15    let infos: HashMap<_, _> = std::mem::take(&mut result.raw_sockets)
16        .into_iter()
17        .collect();
18    let inode_ns: HashMap<_, _> = std::mem::take(&mut result.raw_namespaces)
19        .into_iter()
20        .collect();
21    let metadata: HashMap<_, _> = std::mem::take(&mut result.raw_connections)
22        .into_iter()
23        .collect();
24    let mut truncated = false;
25    let mut unknown_socket_types = 0;
26    let mut unknown_connection_statuses = 0;
27    for (resource, endpoints) in &mut owners {
28        if resource.kind != IpcKind::Socket {
29            continue;
30        }
31        let observed = metadata
32            .get(resource)
33            .copied()
34            .unwrap_or(SocketConnectionStatus::Unknown);
35        let info = infos.get(&resource.inode);
36        let status = info.map_or(observed, |info| info.connection_status(observed));
37        for endpoint in endpoints {
38            endpoint.connection_status = Some(status);
39            if info.is_none() {
40                unknown_socket_types += endpoint.fd_count;
41            }
42            if status == SocketConnectionStatus::Unknown {
43                unknown_connection_statuses += endpoint.fd_count;
44            }
45        }
46    }
47    // Socket inodes are global, but reversed tuples must also share a namespace. Build candidates from same net table below only if unique.
48    let mut reverse: HashMap<_, Vec<u64>> = HashMap::new();
49    for (&inode, info) in &infos {
50        if info.protocol.is_tcp_or_udp()
51            && let (Some(local), Some(remote)) = (info.local, info.remote)
52            && remote.port() != 0
53            && !remote.ip().is_unspecified()
54            && info.state != Some(SocketState::Listen)
55        {
56            reverse
57                .entry((
58                    inode_ns.get(&inode).copied(),
59                    info.protocol.is_tcp(),
60                    local,
61                    remote,
62                ))
63                .or_default()
64                .push(inode);
65        }
66    }
67    let mut by_inode = HashMap::new();
68    for key in owners.keys() {
69        if key.kind == IpcKind::Socket {
70            by_inode.insert(key.inode, *key);
71        }
72    }
73    for endpoints in owners.values_mut() {
74        endpoints.sort_by_key(|p| (p.process_id.pid, p.process_id.start_time_ticks, p.fd));
75    }
76    let mut seen = HashSet::new();
77    for (key, endpoints) in &owners {
78        for endpoint in endpoints {
79            let mut matches: Vec<(&FdEndpoint, bool, bool, String)> = Vec::new();
80            for peer in endpoints {
81                if endpoint.process_id == peer.process_id && endpoint.fd == peer.fd {
82                    continue;
83                }
84                let shared =
85                    endpoint.kind == FdKind::Socket || !endpoint.access.opposite(peer.access);
86                // Shared ownership is a group, not a full mesh of communication links.
87                if shared
88                    && !std::ptr::eq(endpoint, &endpoints[0])
89                    && !std::ptr::eq(peer, &endpoints[0])
90                {
91                    continue;
92                }
93                matches.push((peer, false, shared, endpoint.kind.to_string()));
94            }
95            let inode = key.inode;
96            let info = if endpoint.kind == FdKind::Socket {
97                infos.get(&inode)
98            } else {
99                None
100            };
101            if let Some(info) = info {
102                let mut peers = Vec::new();
103                if let Some(peer) = info.peer_inode {
104                    peers.push((peer, false));
105                } else if info.protocol.is_tcp_or_udp()
106                    && let (Some(local), Some(remote)) = (info.local, info.remote)
107                    && let Some(ids) = reverse.get(&(
108                        inode_ns.get(&inode).copied(),
109                        info.protocol.is_tcp(),
110                        remote,
111                        local,
112                    ))
113                {
114                    peers.extend(ids.iter().filter(|&&v| v != inode).map(|&v| (v, true)));
115                }
116                for (peer, candidate) in peers {
117                    if let Some(peer_endpoints) = by_inode.get(&peer).and_then(|k| owners.get(k)) {
118                        for peer in peer_endpoints {
119                            matches.push((peer, candidate, false, info.protocol.to_string()));
120                        }
121                    }
122                }
123            }
124            // Shared ownership does not identify a network receiver.
125            if matches.is_empty()
126                || (info.is_some_and(|info| SocketEndpoint::from(info).network_peer)
127                    && matches.iter().all(|(_, _, shared, _)| *shared))
128            {
129                let label = info
130                    .map(|i| {
131                        format!(
132                            "{} {} {}",
133                            i.protocol,
134                            i.state.map(|s| s.to_string()).unwrap_or_default(),
135                            i.remote
136                                .map(|v| v.to_string())
137                                .or(i.path.clone())
138                                .unwrap_or_default()
139                        )
140                    })
141                    .map(|label| label.trim().to_owned())
142                    .unwrap_or_else(|| {
143                        if endpoint.kind == FdKind::Socket {
144                            "Socket · Type unavailable".into()
145                        } else {
146                            "Unknown peer".into()
147                        }
148                    });
149                result.fd_relations.push(FdRelation {
150                    id: format!(
151                        "{}:{}:{}:external",
152                        endpoint.process_id.pid, endpoint.process_id.start_time_ticks, endpoint.fd
153                    ),
154                    endpoint: endpoint.clone(),
155                    peer: None,
156                    label,
157                    socket: info.map(SocketEndpoint::from),
158                    candidate: false,
159                    shared: false,
160                });
161            }
162            matches.sort_by_key(|(_, _, shared, _)| *shared);
163            if matches.len() > 64 {
164                truncated = true;
165            }
166            for (peer, candidate, shared, label) in matches.into_iter().take(64) {
167                let label = if endpoint.kind == FdKind::Socket {
168                    info.map(|i| i.protocol.to_string())
169                        .unwrap_or_else(|| "Socket · Type unavailable".into())
170                } else {
171                    label
172                };
173                let mut ends = [
174                    format!(
175                        "{}:{}:{}",
176                        endpoint.process_id.pid, endpoint.process_id.start_time_ticks, endpoint.fd
177                    ),
178                    format!(
179                        "{}:{}:{}",
180                        peer.process_id.pid, peer.process_id.start_time_ticks, peer.fd
181                    ),
182                ];
183                ends.sort();
184                let id = ends.join("/");
185                if seen.insert(id.clone()) {
186                    result.fd_relations.push(FdRelation {
187                        id,
188                        endpoint: endpoint.clone(),
189                        peer: Some(peer.clone()),
190                        label,
191                        socket: info.map(SocketEndpoint::from),
192                        candidate,
193                        shared,
194                    });
195                }
196            }
197            if result.fd_relations.len() >= 20_000 {
198                truncated = true;
199                break;
200            }
201        }
202        if result.fd_relations.len() >= 20_000 {
203            break;
204        }
205    }
206    if unknown_socket_types > 0 {
207        result.warnings.push(format!(
208            "Socket type unavailable for {unknown_socket_types} FDs (closed, unobserved, unsupported, or access denied)."
209        ));
210    }
211    if unknown_connection_statuses > 0 {
212        result.warnings.push(format!(
213            "Socket connection status unavailable for {unknown_connection_statuses} FDs (closed, unobserved, unsupported, or access denied)."
214        ));
215    }
216
217    if let Some(captured) = result.raw_neighbors.take() {
218        let own = captured.namespace;
219        let neighbors: HashMap<_, _> = captured.entries.into_iter().collect();
220        for relation in &mut result.fd_relations {
221            if inode_ns.get(&relation.endpoint.resource.inode) == Some(&own)
222                && let Some(socket) = &mut relation.socket
223                && socket.network_peer
224            {
225                socket.remote_gid = socket
226                    .remote
227                    .and_then(|remote| neighbors.get(&remote.ip).cloned());
228            }
229        }
230    }
231    truncated
232}
