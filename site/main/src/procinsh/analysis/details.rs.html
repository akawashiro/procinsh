13use anyhow::{Result, bail, ensure};
14use serde::Serialize;
15#[derive(Debug, Serialize, serde::Deserialize)]
16pub struct EnvironmentEntry {
17    pub name: String,
18    pub value: Option<String>,
19}
20
21pub fn parse_environment(bytes: &[u8]) -> Vec<EnvironmentEntry> {
22    bytes
23        .split(|b| *b == 0)
24        .filter(|entry| !entry.is_empty())
25        .map(|entry| {
26            let (name, value) = match entry.iter().position(|b| *b == b'=') {
27                Some(index) => (&entry[..index], Some(&entry[index + 1..])),
28                None => (entry, None),
29            };
30            EnvironmentEntry {
31                name: String::from_utf8_lossy(name).into_owned(),
32                value: value.map(|v| String::from_utf8_lossy(v).into_owned()),
33            }
34        })
35        .collect()
36}

38#[derive(Debug, Serialize, serde::Deserialize)]
39pub struct AuxEntry {

47    pub decimal: String,
48    pub kind: String,

50    pub text: Option<String>,

54fn aux_info(tag: u64) -> (&'static str, &'static str, &'static str) {
55    match tag {
56        0 => ("AT_NULL", "number", "End of vector"),
57        1 => ("AT_IGNORE", "number", "Ignored entry"),
58        2 => ("AT_EXECFD", "number", "Executable file descriptor"),
59        3 => ("AT_PHDR", "address", "Program header address"),
60        4 => ("AT_PHENT", "number", "Size of one program header"),
61        5 => ("AT_PHNUM", "number", "Number of program headers"),
62        6 => ("AT_PAGESZ", "number", "Page size (bytes)"),
63        7 => ("AT_BASE", "address", "ELF interpreter base address"),
64        8 => ("AT_FLAGS", "bitmask", "Flags"),
65        9 => ("AT_ENTRY", "address", "Program entry point"),
66        10 => ("AT_NOTELF", "number", "Non-ELF program"),
67        11 => ("AT_UID", "number", "Real UID"),

94pub fn parse_auxv(bytes: &[u8], word_bytes: usize) -> Result<Vec<AuxEntry>> {
95    ensure!(matches!(word_bytes, 4 | 8), "unsupported auxv word size");
96    let mut entries = Vec::new();
97    for pair in bytes.chunks_exact(word_bytes * 2) {
98        let word = |b: &[u8]| {
99            if word_bytes == 8 {
100                u64::from_le_bytes(b.try_into().unwrap())
101            } else {
102                u32::from_le_bytes(b.try_into().unwrap()) as u64
103            }
104        };
105        let tag = word(&pair[..word_bytes]);
106        let value = word(&pair[word_bytes..]);
107        let (name, kind, description) = aux_info(tag);
108        entries.push(AuxEntry {
109            tag: tag.to_string(),
110            name: if name == "AT_UNKNOWN" {
111                format!("AT_UNKNOWN_{tag}")
112            } else {
113                name.into()
114            },
115            value,
116            decimal: value.to_string(),
117            kind: kind.into(),
118            description: description.into(),
119            text: None,
120            text_error: None,
121        });
122        if tag == 0 {
123            ensure!(value == 0, "auxv: invalid AT_NULL value");
124            return Ok(entries);
125        }
126    }
127    bail!("Incomplete auxv (AT_NULL is missing).")
128}

130pub fn complete(value: &mut serde_json::Value) -> Result<()> {
131    if value.get("raw_environment").is_some() {
132        let bytes: Vec<u8> = serde_json::from_value(
133            value
134                .as_object_mut()
135                .unwrap()
136                .remove("raw_environment")
137                .unwrap(),
138        )?;
139        value["entries"] = serde_json::to_value(parse_environment(&bytes))?;
140        value["lossy_utf8"] = serde_json::json!(std::str::from_utf8(&bytes).is_err());
141    }
142    if value.get("raw_auxv").is_some() {
143        let bytes: Vec<u8> =
144            serde_json::from_value(value.as_object_mut().unwrap().remove("raw_auxv").unwrap())?;
145        let word_bytes = value["word_bits"].as_u64().unwrap() as usize / 8;
146        let texts: Vec<(u64, Option<String>, Option<String>)> = serde_json::from_value(
147            value
148                .as_object_mut()
149                .unwrap()
150                .remove("raw_aux_strings")
151                .unwrap(),
152        )?;
153        let mut entries = parse_auxv(&bytes, word_bytes)?;
154        for entry in &mut entries {
155            if let Some((_, text, error)) = texts
156                .iter()
157                .find(|(tag, _, _)| entry.tag == tag.to_string())
158            {
159                entry.text = text.clone();
160                entry.text_error = error.clone();
161            }
162        }
163        value["entries"] = serde_json::to_value(entries)?;
164    }
165    Ok(())
166}
