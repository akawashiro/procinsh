32struct ProcessHistory {
33    sample: crate::ipc::model::process::ProcessSample,
34    observation: crate::ipc::model::process::ProcessObservation,
35    points: std::collections::VecDeque<crate::ipc::history::HistoryPoint>,
36}
37/// Owns analysis caches and independent per-session histories and BPF deltas.
38/// IPC-only capture fields are consumed before any HTTP serialization.
39#[derive(Default)]
40pub(super) struct ObservationProcessor {
41    histories: HashMap<(u64, ProcessId), ProcessHistory>,
42    activity: HashMap<u64, crate::http_server::analysis::activity::ActivityProcessor>,
43    known: HashMap<i32, ProcessId>,
44    discovery: HashMap<ProcessId, (u64, u64)>,
45    symbols: ElfCache,
46    loaded: HashSet<(ProcessId, u64, u64, u64)>,
47    unwinders: HashMap<ProcessId, StackUnwinder>,
48}
49pub(super) fn close_observation_processor(state: &mut ObservationProcessor, session: u64) {
50    let processes: Vec<_> = state
51        .histories
52        .keys()
53        .filter(|(key, _)| *key == session)
54        .map(|(_, id)| *id)
55        .collect();
56    state.histories.retain(|(key, _), _| *key != session);
57    for id in processes {
58        if !state.histories.keys().any(|(_, other)| *other == id) {
59            state.unwinders.remove(&id);
60            state.loaded.retain(|(process, _, _, _)| *process != id);
61        }
62    }
63    state.activity.remove(&session);
64}

66fn snapshot_observation_processor(
67    state: &mut ObservationProcessor,
68    mut snapshot: crate::ipc::model::system::SystemSnapshot,
69) -> Result<Value> {
70    use crate::http_server::analysis::hardware::{
71        finish_device_handle_collector,
72        observation::{
73            acquired_hardware_sources, handles_hardware_observation, observe_hardware_observation,
74        },
75        observe_device_handle_collector,
76    };
77
78    state.known = snapshot
79        .processes
80        .iter()
81        .map(|process| (process.identity.pid, process.identity))
82        .collect();
83    if crate::http_server::analysis::relations::build_ipc_relations(&mut snapshot) {
84        snapshot.warnings.push("Scan, mapping, or connection limit reached. Some process or FD information is missing.".into());
85    }
86    let sources = acquired_hardware_sources(std::mem::take(&mut snapshot.raw_hardware));
87    let hardware = observe_hardware_observation(&sources);
88    let mut handles = handles_hardware_observation(&hardware, &sources);
89    for (process, fd, rdev) in std::mem::take(&mut snapshot.raw_device_handles) {
90        observe_device_handle_collector(&mut handles, process, fd, rdev);
91    }
92    let live = snapshot
93        .processes
94        .iter()
95        .map(|process| process.identity)
96        .collect();
97    snapshot.device_handles = finish_device_handle_collector(handles, &live);
98    snapshot.hardware_devices = hardware.devices;
99    snapshot.hardware_paths = hardware.paths;
100    let hostnames: HashMap<_, _> = std::mem::take(&mut snapshot.raw_hostnames)
101        .into_iter()
102        .collect();
103    for relation in &mut snapshot.fd_relations {
104        if let Some(socket) = &mut relation.socket
105            && socket.network_peer
106        {
107            socket.remote_hostname = socket
108                .remote
109                .and_then(|address| hostnames.get(&address.ip).cloned());
110        }
111    }
112    Ok(serde_json::to_value(snapshot)?)
113}

115pub(super) async fn process_observation_processor(
116    state: &mut ObservationProcessor,
117    client: &MonitorClient,
118    session: Option<u64>,
119    response: &mut MonitorResponse,
120) -> Result<()> {
121    use crate::{
122        http_server::analysis::{
123            activity::process_activity_processor,
124            snapshot::{
125                symbol::{
126                    cache_elf_symbols_from_image, lookup_cached_elf_file_identity,
127                    lookup_elf_symbols,
128                },
129                unwind::{refresh_stack_unwinder, unwind_stack},
130            },
131        },
132        ipc::model::memory::memory_map_contains,
133    };
134
135    let session = match response {
136        MonitorResponse::Started { session, .. } => Some(*session),
137        _ => session,
138    };
139    let mut snapshot_processed = false;
140    let event = match response {
141        MonitorResponse::Started { initial, .. } => Some(initial),
142        MonitorResponse::Event {
143            event: Some(event), ..
144        } => Some(event),
145        _ => None,
146    };
147    if let Some(event) = event {
148        let acquired = matches!(
149            event,
150            MonitorEvent::AcquiredSnapshot(_) | MonitorEvent::AcquisitionGap { .. }
151        );
152        if acquired {
153            let raw = std::mem::replace(event, MonitorEvent::Snapshot(Value::Null));
154            *event = match raw {
155                MonitorEvent::AcquiredSnapshot(snapshot) => {
156                    MonitorEvent::Snapshot(snapshot_observation_processor(
157                        state,
158                        std::sync::Arc::unwrap_or_clone(snapshot),
159                    )?)
160                }
161                MonitorEvent::AcquisitionGap { dropped, snapshot } => MonitorEvent::Gap {
162                    dropped,
163                    snapshot: snapshot_observation_processor(
164                        state,
165                        std::sync::Arc::unwrap_or_clone(snapshot),
166                    )?,
167                },
168                _ => unreachable!(),
169            };
170            snapshot_processed = true;
171        }
172    }
173    let value = match response {
174        MonitorResponse::Data(value) => value,
175        MonitorResponse::Started {
176            initial:
177                MonitorEvent::Observation(value)
178                | MonitorEvent::Snapshot(value)
179                | MonitorEvent::Activity(value),
180            ..
181        }
182        | MonitorResponse::Event {
183            event:
184                Some(
185                    MonitorEvent::Observation(value)
186                    | MonitorEvent::Snapshot(value)
187                    | MonitorEvent::Activity(value),
188                ),
189            ..
190        } => value,
191        MonitorResponse::Started {
192            initial: MonitorEvent::Gap { snapshot, .. },
193            ..
194        }
195        | MonitorResponse::Event {
196            event: Some(MonitorEvent::Gap { snapshot, .. }),
197            ..
198        } => snapshot,
199        _ => return Ok(()),
200    };
201    crate::http_server::analysis::details::format_process_metadata(value)?;
202    let name = |record: &mut Value| {
203        if let Some(object) = record.as_object_mut() {
204            object.remove("raw_ticks");
205            object.remove("raw_measured_ns");
206        }
207    };
208    if let Some(list) = value.as_array_mut() {
209        let mut next = HashMap::new();
210        for record in list.iter_mut() {
211            if let (Ok(id), Some(ticks), Some(now)) = (
212                serde_json::from_value::<ProcessId>(record["identity"].clone()),
213                record["raw_ticks"].as_u64(),
214                record["raw_measured_ns"].as_u64(),
215            ) {
216                let cpu = state
217                    .discovery
218                    .get(&id)
219                    .and_then(|(old, then)| ticks.checked_sub(*old).zip(now.checked_sub(*then)))
220                    .filter(|(_, elapsed)| *elapsed > 0)
221                    .map(|(delta, elapsed)| {
222                        delta as f64
223                            / crate::ipc::model::process::ticks_per_second()
224                            / (elapsed as f64 / 1_000_000_000.0)
225                            * 100.0
226                    });
227                record["cpu_percent"] = serde_json::json!(cpu);
228                next.insert(id, (ticks, now));
229            }
230            name(record);
231        }
232        if !next.is_empty() {
233            state.discovery = next;
234            list.sort_by(|a, b| {
235                b["cpu_percent"]
236                    .as_f64()
237                    .unwrap_or_default()
238                    .total_cmp(&a["cpu_percent"].as_f64().unwrap_or_default())
239                    .then(
240                        a["identity"]["pid"]
241                            .as_i64()
242                            .cmp(&b["identity"]["pid"].as_i64()),
243                    )
244            });
245        }
246    }
247    if let Some(summary) = value.get_mut("summary") {
248        name(summary);
249    }
250    if let Some(processes) = value.get_mut("processes").and_then(Value::as_array_mut) {
251        for process in processes {
252            name(process);
253        }
254    }
255    if let Some(raw) = value
256        .as_object_mut()
257        .and_then(|value| value.remove("raw_activity"))
258        && !raw.is_null()
259    {
260        let raw = serde_json::from_value(raw)?;
261        let session = session.unwrap_or_default();
262        if state.activity.len() >= 64 && !state.activity.contains_key(&session) {
263            state.activity.clear();
264        }
265        let batch = process_activity_processor(
266            state.activity.entry(session).or_default(),
267            raw,
268            &state.known,
269        );
270        value["files"] = serde_json::to_value(batch.files)?;
271        value["ipc"] = serde_json::to_value(batch.ipc)?;
272        value["signals"] = serde_json::to_value(batch.signals)?;
273        value["cpu"] = serde_json::to_value(batch.cpu)?;
274        value["status"]["unresolved"] = serde_json::json!(batch.unresolved);
275        value["status"]["files_lost"] = serde_json::json!(
276            value["status"]["files_lost"]
277                .as_u64()
278                .unwrap_or_default()
279                .saturating_add(batch.files_dropped)
280        );
281    }
282    if let Some(raw) = value
283        .as_object_mut()
284        .and_then(|value| value.remove("process_sample"))
285    {
286        use crate::{
287            http_server::model::process::{initial_observation, next_observation},
288            ipc::model::process::ProcessSample,
289        };
290        let current: ProcessSample = serde_json::from_value(raw)?;
291        let mut observed = initial_observation(&current);
292        if value.get("summary").is_some() {
293            let key = (session.unwrap_or_default(), current.process_id);
294            if state.histories.len() >= 64 && !state.histories.contains_key(&key) {
295                state.histories.clear();
296            }
297            let history = state
298                .histories
299                .entry(key)
300                .or_insert_with(|| ProcessHistory {
301                    sample: current.clone(),
302                    observation: observed.clone(),
303                    points: Default::default(),
304                });
305            if current.measured_at > history.sample.measured_at {
306                observed = next_observation(&history.sample, &current);
307            } else {
308                observed = history.observation.clone();
309            }
310            if history
311                .points
312                .back()
313                .is_none_or(|point| point.timestamp < current.timestamp)
314            {
315                crate::http_server::analysis::history::append_history_observation(
316                    &mut history.points,
317                    &observed,
318                );
319            }
320            history.sample = current;
321            history.observation = observed.clone();
322            value["history"] = serde_json::to_value(&history.points)?;
323            value["observation"] = serde_json::to_value(observed)?;
324        } else {
325            let threads_only = value["threads_only"] == true;
326            *value = if threads_only {
327                serde_json::to_value(observed.threads)?
328            } else {
329                serde_json::to_value(observed)?
330            };
331        }
332    }
333    if value.get("raw_fd_owners").is_some() || value.get("raw_peer_keys").is_some() {
334        let mut descriptors: crate::ipc::descriptors::FileDescriptors =
335            serde_json::from_value(value.take())?;
336        crate::http_server::analysis::descriptors::resolve_descriptor_relations(&mut descriptors);
337        *value = serde_json::to_value(descriptors)?;
338        value.as_object_mut().unwrap().remove("raw_fd_owners");
339        value.as_object_mut().unwrap().remove("raw_peer_keys");
340        for entry in value["entries"].as_array_mut().unwrap() {
341            for field in ["key", "mode", "socket_device"] {
342                entry.as_object_mut().unwrap().remove(field);
343            }
344        }
345    }
346    if !snapshot_processed
347        && value.get("processes").is_some()
348        && value.get("fd_relations").is_some()
349    {
350        *value = snapshot_observation_processor(state, serde_json::from_value(value.take())?)?;
351    }
352    let contexts: Vec<MappingSnapshot> = value
353        .as_object_mut()
354        .and_then(|value| value.remove("mapping_snapshots"))
355        .map(serde_json::from_value)
356        .transpose()?
357        .unwrap_or_default();
358    let contexts: HashMap<_, _> = contexts
359        .into_iter()
360        .map(|context| (context.epoch, context.maps))
361        .collect();
362    if value
363        .get("live_samples")
364        .and_then(Value::as_array)
365        .is_none()
366    {
367        return Ok(());
368    }
369    let id: ProcessId = serde_json::from_value(value["summary"]["identity"].clone())?;
370    let epoch = value["maps_captured_at"].as_u64().unwrap_or_default();
371    let mut maps: Vec<MemoryMap> = contexts.values().flatten().cloned().collect();
372    maps.sort_by_key(|map| map.start);
373    maps.dedup_by_key(|map| (map.start, map.inode));
374    if state.loaded.len() > 65536 {
375        state.loaded.clear();
376    }
377    for map in maps.iter().filter(|map| map.executable && map.inode != 0) {
378        if !state.loaded.insert((id, epoch, map.start, map.inode)) {
379            continue;
380        }
381        if let Ok(Some(file)) = super::elf_reader::acquire_mapped_elf_image(
382            client,
383            id,
384            map.start,
385            lookup_cached_elf_file_identity(&state.symbols, map),
386        )
387        .await
388        {
389            cache_elf_symbols_from_image(&mut state.symbols, map, &file);
390        }
391    }
392    if state.unwinders.len() >= 64 && !state.unwinders.contains_key(&id) {
393        state.unwinders.clear();
394    }
395    let unwinder = state.unwinders.entry(id).or_default();
396    for sample in value["live_samples"].as_array_mut().unwrap() {
397        let raw = sample.as_object_mut().unwrap();
398        if let Some(time) = raw.remove("sample_time_ns").and_then(|time| time.as_u64()) {
399            raw.insert(
400                "sample_age_ms".into(),
401                serde_json::json!(
402                    crate::ipc::model::process::monotonic_ns().saturating_sub(time) / 1_000_000
403                ),
404            );
405        }
406        let registers = raw.remove("raw_registers").unwrap_or(Value::Null);
407        let stack = raw.remove("stack_bytes").unwrap_or(Value::Null);
408        let epoch = raw.remove("map_epoch").and_then(|epoch| epoch.as_u64());
409        if registers.is_null() {
410            continue;
411        }
412        if raw.get("architecture").and_then(Value::as_str) != Some(std::env::consts::ARCH) {
413            let message = "Only native 64-bit process snapshots are supported";
414            let error = raw
415                .get("error")
416                .and_then(Value::as_str)
417                .map_or_else(|| message.to_owned(), |error| format!("{error}; {message}"));
418            raw.insert("error".into(), Value::String(error));
419            raw.insert("registers".into(), serde_json::json!([]));
420            raw.insert("call_stack".into(), serde_json::json!([]));
421            raw.insert("unwind_stop".into(), Value::String(message.into()));
422            raw.insert("disassembly".into(), Value::Null);
423            continue;
424        }
425        let registers: RegisterSet = serde_json::from_value(registers)?;
426        let stack: Vec<u8> = serde_json::from_value(stack)?;
427        let maps = epoch
428            .and_then(|epoch| contexts.get(&epoch))
429            .map(Vec::as_slice)
430            .unwrap_or(&[]);
431        refresh_stack_unwinder(&mut *unwinder, maps, &state.symbols);
432        let (mut frames, reason) = unwind_stack(&mut *unwinder, &registers, maps, &stack);
433        for (index, frame) in frames.iter_mut().enumerate() {
434            let address = instruction_address(frame.address, index > 0);
435            let info = maps
436                .iter()
437                .find(|map| memory_map_contains(map, address) && map.inode != 0)
438                .and_then(|map| {
439                    let elf = lookup_elf_symbols(&state.symbols, map)?;
440                    resolve_frame(
441                        elf_address(address, map, &elf, unsafe {
442                            libc::sysconf(libc::_SC_PAGESIZE) as u64
443                        })?,
444                        &elf,
445                    )
446                })
447                .unwrap_or_default();
448            frame.symbol = info.name;
449            frame.symbol_offset = info.offset.map(|offset| format!("0x{offset:x}"));
450            frame.source_file = info.file;
451            frame.line = info.line;
452            frame.inline_frames = info.inline_frames;
453        }
454        raw.insert(
455            "registers".into(),
456            serde_json::to_value(classify_sample_registers(&registers, maps))?,
457        );
458        raw.insert("call_stack".into(), serde_json::to_value(frames)?);
459        raw.insert("unwind_stop".into(), Value::String(reason));
460    }
461    Ok(())
462}

606            let mut response = MonitorResponse::Data(json!({
607                "summary":{"identity":{"pid":42,"start_time_ticks":10}},
608                "live_samples":[{"architecture":architecture,"raw_registers":RegisterSet::default(),
609                    "stack_bytes":[1,2,3],"map_epoch":1,"error":"Partial capture",
610                    "registers":[],"call_stack":[],"disassembly":{"address":"0x1000","bytes":[144],"error":null}}]
611            }));
612            super::process_observation_processor(
613                &mut ObservationProcessor::default(),
614                &create_unavailable_monitor_client(),
615                None,
616                &mut response,
617            )
618            .await
619            .unwrap();
620            let MonitorResponse::Data(value) = response else {
621                panic!()
622            };
623            let sample = &value["live_samples"][0];
624            assert!(
625                sample["error"]
626                    .as_str()
627                    .unwrap()
628                    .starts_with("Partial capture; Only native")
629            );
630            assert!(sample["disassembly"].is_null());
631            assert!(sample["registers"].as_array().unwrap().is_empty());
632            for field in ["raw_registers", "stack_bytes", "map_epoch"] {
633                assert!(sample.get(field).is_none());
634            }
635        }
636    }
