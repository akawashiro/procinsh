1use super::super::{
2    AppState,

6use super::process::ApiError;
7use axum::{Json, extract::State, http::StatusCode};
8use serde::Deserialize;
9use serde_json::{Value, json};
10use std::sync::Arc;

12/// `?host=`: a host name; absent or the local name selects this machine.
13#[derive(Deserialize)]
14pub(super) struct HostQuery {
15    pub(super) host: Option<String>,
16}
17
18/// `{"local": name, "hosts": [summary...]}` with the local host first.
19pub(super) async fn hosts(State(s): State<Arc<AppState>>) -> Json<Value> {
20    Json(json!({
21        "local": s.hosts.local().name,
22        "hosts": s.hosts.summaries(),
23    }))
24}
25
26/// The helper of `host`: this machine's own, or a remote host's through its agent link.
27pub(super) fn client(s: &AppState, host: Option<&str>) -> Result<MonitorClient, ApiError> {
28    let Some(host) = host.filter(|host| *host != s.hosts.local().name) else {
29        return Ok(s.client.clone());
30    };
31    match s.hosts.remote(host) {
32        RemoteHost::Online(link) => Ok(MonitorClient::agent(link)),
33        RemoteHost::Offline => Err(ApiError::new(
34            StatusCode::SERVICE_UNAVAILABLE,
35            format!("Host {host} is offline"),
36        )),
37        RemoteHost::Unknown if host == ALL_HOSTS => Err(ApiError::new(
38            StatusCode::BAD_REQUEST,
39            "Only SPACE events support all hosts".into(),
40        )),
41        RemoteHost::Unknown => Err(ApiError::new(
42            StatusCode::NOT_FOUND,
43            format!("Unknown host {host}"),
44        )),
45    }
46}
