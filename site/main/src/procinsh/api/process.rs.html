36pub(super) struct ApiError(StatusCode, String);
37pub(super) fn create_api_error(status: StatusCode, message: String) -> ApiError {
38    ApiError(status, message)
39}
40impl IntoResponse for ApiError {
41    fn into_response(self) -> Response {
42        log::debug!("API error status={} detail={}", self.0, self.1);
43        (self.0, Json(json!({"error":self.1}))).into_response()
44    }
45}
46impl From<anyhow::Error> for ApiError {
47    fn from(e: anyhow::Error) -> Self {
48        let message = format!("{e:#}");
49        let status = if message.contains("Server is stopping")
50            || message.contains("monitor IPC")
51            || message.contains("helper unavailable")
52            || message.contains("agent link")
53        {
54            StatusCode::SERVICE_UNAVAILABLE
55        } else if message.contains("Too many viewers") {
56            StatusCode::TOO_MANY_REQUESTS
57        } else if message.contains("Process exited") {
58            StatusCode::GONE
59        } else {
60            StatusCode::UNPROCESSABLE_ENTITY
61        };
62        Self(status, message)
63    }
64}
65type ApiResult = Result<Json<Value>, ApiError>;

67pub(super) async fn get_server_config(State(s): State<Arc<AppState>>) -> Json<Value> {
68    Json(
69        json!({"version":env!("CARGO_PKG_VERSION"),"interval_ms":s.interval.as_millis(),"history_seconds":60}),
70    )
71}

73/// A process on `host`, which defaults to this machine.
74#[derive(Deserialize)]
75#[serde(deny_unknown_fields)]
76pub(super) struct ProcessQuery {
77    host: Option<String>,
78    pid: i32,
79    start_time_ticks: u64,
80}
81fn id_process_query(state: &ProcessQuery) -> ProcessId {
82    ProcessId {
83        pid: state.pid,
84        start_time_ticks: state.start_time_ticks,
85    }
86}

88async fn request_process_data(
89    state: &AppState,
90    host: Option<&str>,
91    request: MonitorRequest,
92) -> ApiResult {
93    use crate::http_server::client::request_monitor_client;
94
95    match request_monitor_client(&(resolve_host_client(state, host)?), request).await? {
96        MonitorResponse::Data(value) => Ok(Json(value)),
97        _ => Err(anyhow::anyhow!("unexpected monitor IPC response").into()),
98    }
99}

150async fn request_process_resource(
151    state: &AppState,
152    query: ProcessQuery,
153    resource: ProcessResource,
154) -> ApiResult {
155    let id = id_process_query(&query);
156    request_process_data(
157        state,
158        query.host.as_deref(),
159        MonitorRequest::Process { id, resource },
160    )
161    .await
162}
163
164pub(super) async fn stream_process_events(
165    State(s): State<Arc<AppState>>,
166    Query(query): Query<ProcessQuery>,
167) -> Result<Response, ApiError> {
168    use crate::http_server::client::observe_monitor_client;
169
170    let session = observe_monitor_client(
171        &(resolve_host_client(&s, query.host.as_deref())?),
172        MonitorRequest::ObserveProcess {
173            id: id_process_query(&query),
174        },
175    )
176    .await?;
177    Ok(super::stream::build_sse_response(s, session))
178}
