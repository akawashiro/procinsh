20pub fn deserialize_hex<'de, D: serde::Deserializer<'de>>(deserializer: D) -> Result<u64, D::Error> {
21    let value = String::deserialize(deserializer)?;
22    u64::from_str_radix(value.trim_start_matches("0x"), 16).map_err(serde::de::Error::custom)
23}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    #[test]
137    fn maps_and_smaps_keep_paths_and_boundaries() {
138        let maps = parse_smaps("1000-2000 r-xp 00001000 08:01 10 /tmp/a b (deleted)\nRss: 4 kB\nPss: 2 kB\n2000-3000 rw-p 0 00:00 0\n").unwrap();
139        assert_eq!(
140            maps[0].mapping.pathname.as_deref(),
141            Some("/tmp/a b (deleted)")
142        );
143        assert_eq!(maps[0].rss_bytes, Some(4096));
144        let detailed = serde_json::to_value(&maps[0]).unwrap();
145        assert_eq!(detailed["rss_bytes"], 4096);
146        assert_eq!(detailed["pss_bytes"], 2048);
147        assert_eq!(detailed["start"], "0x0000000000001000");
148        assert!(detailed.get("mapping").is_none());
149        assert!(maps[0].mapping.contains(0x1000));
150        assert!(!maps[0].mapping.contains(0x2000));
151        assert!(maps[1].mapping.pathname.is_none());
152        assert!(parse_map("2000-1000 rw-p 0 00:00 0").is_err());
153    }
154
155    #[test]
156    fn structural_maps_exclude_usage_and_fallback_observations_keep_shape() {
157        let mapping = parse_map("1000-2000 r-xp 0 00:00 0 /tmp/test").unwrap();
158        let structural = serde_json::to_value(&mapping).unwrap();
159        assert!(structural.get("rss_bytes").is_none());
160        assert!(structural.get("pss_bytes").is_none());
161        let fallback = SmapsEntry::from(mapping.clone());
162        let json = serde_json::to_value(&fallback).unwrap();
163        assert!(json["rss_bytes"].is_null());
164        assert!(json["pss_bytes"].is_null());
165        assert_eq!(json["pathname"], "/tmp/test");
166        let measured =
167            parse_smaps("1000-2000 r-xp 0 00:00 0 /tmp/test\nRss: 8 kB\nPss: 4 kB\n").unwrap();
168        assert_eq!(measured[0].mapping, mapping);
169        assert_eq!(measured[0].rss_bytes, Some(8192));
170    }
