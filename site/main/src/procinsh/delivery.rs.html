17/// Application-side full/delta selection over complete structured observations.
18#[derive(Default)]
19pub(super) struct SnapshotEncoder {
20    previous: Option<Arc<Value>>,
21    last_full: Option<Instant>,
22    epochs: HashMap<String, Value>,
23    sequence: u64,
24}
25
26fn field_identity_key(value: &Value, field: &str) -> String {
27    value[field].to_string()
28}
29
30fn build_array_delta(old: &[Value], current: &[Value], field: &str) -> Option<Value> {
31    let previous: HashMap<_, _> = old
32        .iter()
33        .map(|v| (field_identity_key(v, field), v))
34        .collect();
35    let live: std::collections::HashSet<_> = current
36        .iter()
37        .map(|v| field_identity_key(v, field))
38        .collect();
39    let upsert: Vec<_> = current
40        .iter()
41        .filter(|v| {
42            previous
43                .get(&field_identity_key(v, field))
44                .is_none_or(|p| *p != *v)
45        })
46        .collect();
47    let remove: Vec<_> = old
48        .iter()
49        .filter(|v| !live.contains(&field_identity_key(v, field)))
50        .map(|v| &v[field])
51        .collect();
52    let delta = json!({"upsert":upsert,"remove":remove});
53    (delta.to_string().len() + 6 < serde_json::to_vec(current).unwrap().len()).then_some(delta)
54}
55
56fn snapshot_array_entries(value: &Value) -> &[Value] {
57    value.as_array().map(Vec::as_slice).unwrap_or_default()
58}
59
60pub(super) fn encode_snapshot_encoder(
61    state: &mut SnapshotEncoder,
62    snapshot: Arc<Value>,
63    force_full: bool,
64    now: Instant,
65) -> serde_json::Result<String> {
66    let full = force_full
67        || state
68            .last_full
69            .is_none_or(|last| now.duration_since(last) >= Duration::from_secs(60));
70    let mut output = (*snapshot).clone();
71    let old = state.previous.as_deref().unwrap_or(&Value::Null);
72    let previous: HashMap<_, _> = snapshot_array_entries(&old["processes"])
73        .iter()
74        .map(|p| (field_identity_key(p, "identity"), p))
75        .collect();
76    let mut epochs = HashMap::new();
77    if let Some(processes) = output["processes"].as_array_mut() {
78        for p in processes {
79            let id = field_identity_key(p, "identity");
80            let before = previous.get(&id);
81            let unchanged = !full && before.is_some_and(|old| old["maps"] == p["maps"]);
82            if unchanged {
83                p["maps_epoch"] = state.epochs[&id].clone();
84                p.as_object_mut().unwrap().remove("maps");
85            } else if !full
86                && let Some(before) = before
87                && let Some(change) = build_array_delta(
88                    snapshot_array_entries(&before["maps"]),
89                    snapshot_array_entries(&p["maps"]),
90                    "start",
91                )
92            {
93                p.as_object_mut().unwrap().remove("maps");
94                p["maps_delta"] = change;
95            }
96            epochs.insert(id, p["maps_epoch"].clone());
97        }
98    }
99    // RDMA ports are small and always sent whole.
100    for field in ["fd_relations", "rdma_relations", "device_handles"] {
101        if !full
102            && let Some(change) = build_array_delta(
103                snapshot_array_entries(&old[field]),
104                snapshot_array_entries(&output[field]),
105                "id",
106            )
107        {
108            output.as_object_mut().unwrap().remove(field);
109            output[format!("{field}_delta")] = change;
110        }
111    }
112    state.sequence += 1;
113    output["kind"] = json!(if full { "full" } else { "delta" });
114    output["sequence"] = json!(state.sequence);
115    if !full {
116        output["base_sequence"] = json!(state.sequence - 1);
117    }
118    let data = serde_json::to_string(&output)?;
119    log::debug!(
120        "snapshot_delta kind={} payload_bytes={}",
121        if full { "full" } else { "delta" },
122        data.len()
123    );
124    state.previous = Some(snapshot);
125    state.epochs = epochs;
126    if full {
127        state.last_full = Some(now);
128    }
129    Ok(data)
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135    #[test]
136    fn delta_baseline_refresh_and_pid_reuse() {
137        let now = Instant::now();
138        let sample = |start, epoch| {
139            Arc::new(
140                json!({"processes":[{"identity":{"pid":42,"start_time_ticks":start},"maps":[{"start":"0x1000"}],"maps_epoch":epoch}], "fd_relations":[],"device_handles":[]}),
141            )
142        };
143        let mut encoder = SnapshotEncoder::default();
144        let parse = |s: String| serde_json::from_str::<Value>(&s).unwrap();
145        assert_eq!(
146            parse(super::encode_snapshot_encoder(&mut encoder, sample(1, 1), false, now).unwrap())
147                ["kind"],
148            "full"
149        );
150        let next =
151            parse(super::encode_snapshot_encoder(&mut encoder, sample(1, 2), false, now).unwrap());
152        assert!(next["processes"][0].get("maps").is_none());
153        assert_eq!(next["processes"][0]["maps_epoch"], 1);
154        assert!(
155            parse(super::encode_snapshot_encoder(&mut encoder, sample(2, 3), false, now).unwrap())
156                ["processes"][0]["maps"]
157                .is_array()
158        );
159        assert_eq!(
160            parse(
161                super::encode_snapshot_encoder(
162                    &mut encoder,
163                    sample(2, 4),
164                    false,
165                    now + Duration::from_secs(60)
166                )
167                .unwrap()
168            )["kind"],
169            "full"
170        );
171        assert_eq!(
172            parse(
173                super::encode_snapshot_encoder(
174                    &mut encoder,
175                    sample(2, 5),
176                    true,
177                    now + Duration::from_secs(61)
178                )
179                .unwrap()
180            )["kind"],
181            "full"
182        );
183    }

185    #[test]
186    fn rdma_relations_use_id_deltas_and_ports_are_sent_whole() {
187        let now = Instant::now();
188        let relation = |qpn: u32, label: &str| json!({"id": format!("rdma:ib0/1:{qpn}"), "label": label, "padding": "x".repeat(40)});
189        let snapshot = |relations: Vec<Value>| {
190            Arc::new(
191                json!({"processes": [], "fd_relations": [], "device_handles": [],
192                "rdma_ports": [{"device": "ib0", "port": 1}], "rdma_relations": relations}),
193            )
194        };
195        let initial: Vec<_> = (0..20).map(|q| relation(q, "a")).collect();
196        let mut encoder = SnapshotEncoder::default();
197        let parse = |s: String| serde_json::from_str::<Value>(&s).unwrap();
198        let full = parse(
199            super::encode_snapshot_encoder(&mut encoder, snapshot(initial.clone()), false, now)
200                .unwrap(),
201        );
202        assert_eq!(full["rdma_relations"].as_array().unwrap().len(), 20);
203        let mut current = initial.clone();
204        current.remove(3);
205        current[0] = relation(0, "changed");
206        let delta = parse(
207            super::encode_snapshot_encoder(&mut encoder, snapshot(current), false, now).unwrap(),
208        );
209        assert!(delta.get("rdma_relations").is_none());
210        assert_eq!(
211            delta["rdma_relations_delta"],
212            json!({"upsert": [relation(0, "changed")], "remove": ["rdma:ib0/1:3"]})
213        );
214        assert_eq!(delta["rdma_ports"], full["rdma_ports"]);
215    }
