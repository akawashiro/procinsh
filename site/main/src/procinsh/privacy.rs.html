1//! Privacy mode: mask sensitive text before a reply leaves this process.
2//!
3//! # Interface
4//! - [`configure_privacy`] (`pub(super)`): `fn configure_privacy(level: u8) -> anyhow::Result<()>`.
5//! - [`privacy_enabled`] (`pub(super)`): `fn privacy_enabled() -> bool`.
6//! - [`scrub_response`] (`pub(super)`): `fn scrub_response(response: &mut MonitorResponse)`.
7//! - [`pseudonym`] (`pub(super)`): `fn pseudonym(value: &str) -> String`.
8//!
9//! The privileged helper does not read environments or process memory in this
10//! mode. Everything else it returns passes through [`scrub_response`]:
11//! identifying fields are replaced whole, and other strings are scanned by
12//! logblur, with this machine's host name, IDs and login users preloaded. A masked
13//! value becomes `░` + 8 hex digits + `░`, a keyed hash that is consistent
14//! within this run, so distinct addresses stay distinct, but cannot be linked
15//! across runs or hosts. The original text is never sent.
16use super::protocol::{MonitorEvent, MonitorResponse};
17use anyhow::{Context, Result};
18use logblur_core::{Category, Engine, JobConfig, scan::Found};
19use serde_json::Value;
20use sha2::{Digest, Sha256};
21use std::{
22    collections::HashMap,
23    io::Read,
24    sync::{Mutex, OnceLock},
25};
26
27/// Fields that identify a machine or peer whatever their text looks like.
28const WHOLE_FIELDS: &[&str] = &["remote_hostname", "gid", "remote_gid", "uuid", "serial"];
29/// Cached results; snapshots repeat most strings, so the cache is cleared rather than evicted.
30const CACHE_LIMIT: usize = 200_000;
31
32struct Privacy {
33    engine: Engine,
34    key: [u8; 32],
35    cache: Mutex<HashMap<String, String>>,
36}
37
38static PRIVACY: OnceLock<Privacy> = OnceLock::new();
39
40/// Users that log in (UID 1000 and above, except `nobody`) are personal.
41fn login_users(passwd: &str) -> Vec<String> {
42    passwd
43        .lines()
44        .filter_map(|line| {
45            let mut fields = line.split(':');
46            let name = fields.next()?;
47            let uid: u32 = fields.nth(1)?.parse().ok()?;
48            (uid >= 1000 && uid != 65534 && !name.is_empty()).then(|| name.to_owned())
49        })
50        .collect()
51}
52
53/// This machine's IDs, which programs such as dbus-broker pass in arguments.
54fn machine_ids() -> Vec<String> {
55    ["/etc/machine-id", "/proc/sys/kernel/random/boot_id"]
56        .into_iter()
57        .filter_map(|path| std::fs::read_to_string(path).ok())
58        .map(|id| id.trim().to_owned())
59        .filter(|id| !id.is_empty())
60        .collect()
61}
62
63fn host_names() -> Vec<String> {
64    let Ok(name) = std::fs::read_to_string("/proc/sys/kernel/hostname") else {
65        return Vec::new();
66    };
67    let name = name.trim();
68    let mut names = vec![name.to_owned()];
69    if let Some((short, _)) = name.split_once('.') {
70        names.push(short.to_owned());
71    }
72    names
73}

75impl Privacy {
76    fn new(level: u8, learned: Vec<(Category, String)>, key: [u8; 32]) -> Result<Self> {
77        let config = JobConfig {
78            level,
79            ..JobConfig::default()
80        };
81        config.validate()?;
82        let mut engine = Engine::new(&config)?;
83        engine.set_learned(learned)?;
84        Ok(Self {
85            engine,
86            key,
87            cache: Mutex::new(HashMap::new()),
88        })
89    }
90
91    fn pseudonym(&self, value: &[u8]) -> String {
92        let digest = Sha256::new()
93            .chain_update(self.key)
94            .chain_update(value)
95            .finalize();
96        let hex: String = digest[..4].iter().map(|b| format!("{b:02x}")).collect();
97        format!("░{hex}░")
98    }
99
100    /// Scans `items` as one space-separated text, so `-p` and its password in
101    /// separate arguments are seen together, and masks each item's share of a hit.
102    fn scan(&self, items: &[&str]) -> Vec<String> {
103        // Arguments cannot contain NUL, and pseudonyms never do.
104        let key = items.join("\0");
105        if let Some(masked) = self.cache.lock().unwrap().get(&key) {
106            return masked.split('\0').map(str::to_owned).collect();
107        }
108        let text = items.join(" ");
109        let bytes = text.as_bytes();
110        let mut found: Vec<Found> = Vec::new();
111        self.engine.find(bytes, &mut found);
112        let mut masked = Vec::with_capacity(bytes.len());
113        let mut start = 0;
114        for (index, item) in items.iter().enumerate() {
115            if index > 0 {
116                masked.push(0);
117            }
118            let end = start + item.len();
119            let mut done = start;
120            for hit in found
121                .iter()
122                .filter(|hit| hit.start < end && hit.end > start)
123            {
124                let (from, to) = (hit.start.max(start), hit.end.min(end));
125                masked.extend_from_slice(&bytes[done..from]);
126                masked.extend_from_slice(self.pseudonym(&bytes[from..to]).as_bytes());
127                done = to;
128            }
129            masked.extend_from_slice(&bytes[done..end]);
130            start = end + 1;
131        }
132        let masked = String::from_utf8_lossy(&masked).into_owned();
133        let mut cache = self.cache.lock().unwrap();
134        if cache.len() >= CACHE_LIMIT {
135            cache.clear();
136        }
137        cache.insert(key, masked.clone());
138        masked.split('\0').map(str::to_owned).collect()
139    }
140
141    fn scrub(&self, value: &mut Value, field: Option<&str>) {
142        match value {
143            Value::String(text) if text.is_empty() => {}
144            Value::String(text) => {
145                let whole = match field {
146                    Some("ip") => text
147                        .parse::<std::net::IpAddr>()
148                        .is_ok_and(|ip| !ip.is_loopback() && !ip.is_unspecified()),
149                    Some(field) => WHOLE_FIELDS.contains(&field),
150                    None => false,
151                };
152                *text = if whole {
153                    self.pseudonym(text.as_bytes())
154                } else {
155                    self.scan(&[text]).remove(0)
156                };
157            }
158            Value::Array(items)
159                if items.len() > 1
160                    && !field.is_some_and(|field| WHOLE_FIELDS.contains(&field))
161                    && items.iter().all(Value::is_string) =>
162            {
163                let texts: Vec<&str> = items.iter().filter_map(Value::as_str).collect();
164                let masked = self.scan(&texts);
165                for (item, masked) in items.iter_mut().zip(masked) {
166                    *item = Value::String(masked);
167                }
168            }
169            Value::Array(items) => {
170                for item in items {
171                    self.scrub(item, field);
172                }
173            }
174            Value::Object(fields) => {
175                for (name, item) in fields {
176                    self.scrub(item, Some(name));
177                }
178            }
179            Value::Null | Value::Bool(_) | Value::Number(_) => {}
180        }
181    }
182}

184/// Turns privacy mode on for this process. `level` is logblur's 1–4.
185pub(super) fn configure_privacy(level: u8) -> Result<()> {
186    let mut key = [0; 32];
187    std::fs::File::open("/dev/urandom")
188        .and_then(|mut random| random.read_exact(&mut key))
189        .context("read a random privacy key")?;
190    let mut learned: Vec<_> = host_names()
191        .into_iter()
192        .map(|name| (Category::Hostname, name))
193        .chain(
194            machine_ids()
195                .into_iter()
196                .map(|id| (Category::MachineId, id)),
197        )
198        .collect();
199    if let Ok(passwd) = std::fs::read_to_string("/etc/passwd") {
200        learned.extend(
201            login_users(&passwd)
202                .into_iter()
203                .map(|name| (Category::Username, name)),
204        );
205    }
206    let privacy = Privacy::new(level, learned, key).context("start privacy mode")?;
207    let _ = PRIVACY.set(privacy);
208    Ok(())
209}
210
211pub(super) fn privacy_enabled() -> bool {
212    PRIVACY.get().is_some()
213}
214
215/// The masked form of a whole value, such as the machine ID a host advertises.
216pub(super) fn pseudonym(value: &str) -> String {
217    match PRIVACY.get() {
218        Some(privacy) => privacy.pseudonym(value.as_bytes()),
219        None => value.to_owned(),
220    }
221}
222
223/// Masks every string in a reply when privacy mode is on.
224pub(super) fn scrub_response(response: &mut MonitorResponse) {
225    let Some(privacy) = PRIVACY.get() else {
226        return;
227    };
228    let event = match response {
229        MonitorResponse::Data(value) => return privacy.scrub(value, None),
230        MonitorResponse::Started { initial, .. } => initial,
231        MonitorResponse::Event {
232            event: Some(event), ..
233        } => event,
234        MonitorResponse::Event { event: None, .. } | MonitorResponse::Closed => return,
235    };
236    match event {
237        MonitorEvent::Observation(value)
238        | MonitorEvent::Snapshot(value)
239        | MonitorEvent::Activity(value)
240        | MonitorEvent::Gap {
241            snapshot: value, ..
242        } => privacy.scrub(value, None),
243        // Converted to JSON by the observation processor before this point.
244        MonitorEvent::AcquiredSnapshot(_) | MonitorEvent::AcquisitionGap { .. } => {}
245    }
246}

253    fn privacy() -> Privacy {
254        Privacy::new(
255            3,
256            vec![
257                (Category::Hostname, "workstation-7".into()),
258                (Category::Username, "alice".into()),
259            ],
260            [7; 32],

263    }
264
265    fn masked(text: &str) -> bool {
266        text.contains('░')
267    }
268
269    #[test]
270    fn login_users_skip_system_accounts_and_nobody() {
271        let passwd = "root:x:0:0::/root:/bin/sh\nalice:x:1000:1000::/home/alice:/bin/sh\n\
272                      nobody:x:65534:65534::/:/sbin/nologin\nbob:x:1001:1001::/home/bob:/bin/sh\n";
273        assert_eq!(login_users(passwd), ["alice", "bob"]);
274    }
275
276    #[test]
277    fn secrets_users_and_hosts_are_masked_in_any_string() {
278        let privacy = privacy();
279        let mut value = json!({
280            "command_line": ["sshpass", "-p", "hunter2secret", "ssh", "alice@workstation-7"],
281            "path": "/home/alice/.ssh/id_ed25519",
282            "comm": "sshd-session: alice@pts/4",
283            "username": "alice",
284            "uid": 1000,
285        });
286        privacy.scrub(&mut value, None);
287        let text = value.to_string();
288        for secret in ["hunter2secret", "alice", "workstation-7"] {
289            assert!(!text.contains(secret), "{secret} in {text}");
290        }
291        assert_eq!(value["uid"], 1000);
292        assert!(masked(value["username"].as_str().unwrap()));
293        assert_eq!(value["command_line"][0], "sshpass");
294    }
295
296    #[test]
297    fn identifying_fields_are_masked_whole_and_consistently() {
298        let privacy = privacy();
299        let mut value = json!([
300            {"remote_hostname": "peer", "remote": {"ip": "192.0.2.1", "port": 22}},
301            {"remote_hostname": "peer", "remote": {"ip": "192.0.2.2", "port": 22}},
302            {"local": {"ip": "127.0.0.1", "port": 8080}, "gid": "fe80::2:c903:10:1234"},
303        ]);
304        privacy.scrub(&mut value, None);
305        assert!(masked(value[0]["remote_hostname"].as_str().unwrap()));
306        assert_eq!(value[0]["remote_hostname"], value[1]["remote_hostname"]);
307        assert_ne!(value[0]["remote"]["ip"], value[1]["remote"]["ip"]);
308        assert_eq!(value[0]["remote"]["port"], 22);
309        assert_eq!(value[2]["local"]["ip"], "127.0.0.1");
310        assert!(masked(value[2]["gid"].as_str().unwrap()));
311    }

313    #[test]
314    fn ordinary_text_is_unchanged_and_keys_differ_per_run() {
315        let privacy = privacy();
316        let mut value =
317            json!({"comm": "cargo", "command_line": ["make", "-j", "16"], "text": "mov rax, 0x10"});
318        let before = value.clone();
319        privacy.scrub(&mut value, None);
320        assert_eq!(value, before);
321        let other = Privacy::new(3, Vec::new(), [8; 32]).unwrap();
322        assert_ne!(privacy.pseudonym(b"10.0.0.1"), other.pseudonym(b"10.0.0.1"));
323        assert_eq!(
324            privacy.pseudonym(b"10.0.0.1"),
325            privacy.pseudonym(b"10.0.0.1")
326        );
