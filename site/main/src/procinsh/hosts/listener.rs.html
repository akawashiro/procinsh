23const HANDSHAKE_TIMEOUT: Duration = Duration::from_secs(5);
24pub(super) const PING_INTERVAL: Duration = Duration::from_secs(10);
25/// A link without any message for this long is closed.
26pub(super) const IDLE_TIMEOUT: Duration = Duration::from_secs(30);

28/// Accepts agent links until `shutdown` changes. `token`, when set, must match each
29/// agent's hello.
30pub(in crate::http_server) async fn run_agent_listener(
31    listener: TcpListener,
32    registry: Arc<HostRegistry>,
33    token: Option<Arc<str>>,
34    mut shutdown: watch::Receiver<bool>,
35) {
36    loop {
37        let (stream, peer) = tokio::select! {
38            accepted = listener.accept() => match accepted {
39                Ok(accepted) => accepted,
40                Err(error) => {
41                    log::warn!("agent accept failed: {error}");
42                    tokio::time::sleep(Duration::from_millis(100)).await;
43                    continue;
44                }
45            },
46            _ = shutdown.changed() => return,
47        };
48        let (registry, token, shutdown) = (registry.clone(), token.clone(), shutdown.clone());
49        tokio::spawn(async move {
50            if let Err(error) = serve_agent(stream, peer, &registry, token, shutdown).await {
51                log::warn!("agent link from {peer} closed: {error:#}");
52            }
53        });
54    }
55}

57async fn accept_agent_handshake(
58    stream: &mut TcpStream,
59    registry: &HostRegistry,
60    token: Option<&str>,
61    link: Arc<AgentLink>,
62) -> Result<(String, u64)> {
63    use crate::http_server::hosts::{
64        identity::protect_remote_host_info, registry::admit_host_registry,
65    };
66
67    let hello = timeout(HANDSHAKE_TIMEOUT, wire::read_host_message(stream))
68        .await
69        .context("agent hello timed out")??;
70    let Message::Hello {
71        protocol,
72        host,
73        machine_id,
74        version,
75        kernel,
76        arch,
77        token: offered,
78    } = hello
79    else {
80        bail!("agent did not start with hello");
81    };
82    let mut info = HostInfo {
83        name: host.clone(),
84        machine_id,
85        version,
86        kernel,
87        arch,
88    };
89    let admitted = if protocol != wire::PROTOCOL {
90        Err(format!(
91            "agent protocol {protocol} is not supported (hub protocol {})",
92            wire::PROTOCOL
93        ))
94    } else if let Some(expected) = token
95        && !wire::same_secret(expected, offered.as_deref().unwrap_or_default())
96    {
97        Err("agent token mismatch".into())
98    } else if !valid_host_name(&info.name) {
99        Err("invalid agent host name".into())
100    } else {
101        protect_remote_host_info(&mut info);
102        admit_host_registry(registry, info.clone(), link)
103    };
104    match admitted {
105        Ok(id) => {
106            wire::write_host_message(
107                stream,
108                Message::Welcome {
109                    protocol: wire::PROTOCOL,
110                    version: env!("CARGO_PKG_VERSION").into(),
111                },
112            )
113            .await?;
114            Ok((info.name, id))
115        }
116        Err(reason) => {
117            wire::write_host_message(
118                stream,
119                Message::Reject {
120                    reason: reason.clone(),
121                },
122            )
123            .await?;
124            bail!("rejected agent: {reason}");
125        }
126    }
127}

129async fn serve_agent(
130    mut stream: TcpStream,
131    peer: SocketAddr,
132    registry: &HostRegistry,
133    token: Option<Arc<str>>,
134    mut shutdown: watch::Receiver<bool>,
135) -> Result<()> {
136    use crate::http_server::hosts::{
137        link::{close_agent_link, create_agent_link},
138        registry::{disconnected_host_registry, host_registry_is_current},
139    };
140
141    stream.set_nodelay(true)?;
142    let (link, outgoing) = create_agent_link();
143    let link = Arc::new(link);
144    let (name, id) =
145        accept_agent_handshake(&mut stream, registry, token.as_deref(), link.clone()).await?;
146    log::info!("agent {name} connected from {peer}");
147    let result = relay_agent_messages(stream, &link, outgoing, &mut shutdown, || {
148        host_registry_is_current(registry, &name, id)
149    })
150    .await;
151    disconnected_host_registry(registry, &name, id);
152    close_agent_link(&link);
153    log::info!("agent {name} disconnected");
154    result
155}

157/// Moves requests to the agent and its responses back until the link fails,
158/// idles, is replaced, or the hub shuts down.
159async fn relay_agent_messages(
160    stream: TcpStream,
161    link: &AgentLink,
162    mut outgoing: tokio::sync::mpsc::Receiver<Message>,
163    shutdown: &mut watch::Receiver<bool>,
164    current: impl Fn() -> bool,
165) -> Result<()> {
166    use crate::http_server::hosts::link::{complete_agent_link, send_agent_link};
167
168    let (mut reader, mut writer) = stream.into_split();
169    // Dedicated tasks keep partially read or written frames intact across select branches.
170    let (messages, mut incoming) = tokio::sync::mpsc::channel(8);
171    let reading = tokio::spawn(async move {
172        loop {
173            let message = wire::read_host_message(&mut reader).await;
174            let failed = message.is_err();
175            if messages.send(message).await.is_err() || failed {
176                break;
177            }
178        }
179    });
180    let (written, mut write_failed) = tokio::sync::oneshot::channel();
181    let writing = tokio::spawn(async move {
182        while let Some(message) = outgoing.recv().await {
183            if let Err(error) = wire::write_host_message(&mut writer, message).await {
184                let _ = written.send(error);
185                return;
186            }
187        }
188    });
189    let mut ping = tokio::time::interval(PING_INTERVAL);
190    let mut last_message = Instant::now();
191    let result = loop {
192        tokio::select! {
193            message = incoming.recv() => {
194                let Some(message) = message else { break Ok(()) };
195                last_message = Instant::now();
196                match message {
197                    Ok(Message::Response { id, response }) => complete_agent_link(link, id, response),
198                    // A full queue is already traffic; a lost pong only delays idleness.
199                    Ok(Message::Ping) => drop(send_agent_link(link, Message::Pong)),
200                    Ok(Message::Pong) => {}
201                    Ok(_) => break Err(anyhow::anyhow!("unexpected message from agent")),
202                    Err(error) => break Err(error),
203                }
204            }
205            error = &mut write_failed => {
206                break Err(error.unwrap_or_else(|_| anyhow::anyhow!("agent link writer stopped")));
207            }
208            _ = ping.tick() => {
209                if !current() {
210                    break Ok(());
211                }
212                if last_message.elapsed() > IDLE_TIMEOUT {
213                    break Err(anyhow::anyhow!("agent link idle for {IDLE_TIMEOUT:?}"));
214                }
215                drop(send_agent_link(link, Message::Ping));
216            }
217            _ = shutdown.changed() => break Ok(()),
218        }
219    };
220    reading.abort();
221    writing.abort();
222    result
223}
