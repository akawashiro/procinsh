1//! Decode captured instruction bytes in the unprivileged HTTP process.
2//!
3//! # Interface
4//! [`decode_response`] (`pub(super)`): `fn decode_response(response: &mut MonitorResponse) -> anyhow::Result<()>`.
5//! The response envelope is [`MonitorResponse`].
6
7use crate::http_server::protocol::{MonitorEvent, MonitorResponse};
8use anyhow::Result;
9use iced_x86::{Decoder, DecoderError, DecoderOptions, Formatter, IntelFormatter};
10use serde::{Deserialize, Serialize};
11use serde_json::Value;
12
13const MAX_BYTES: usize = 256;
14const MAX_INSTRUCTIONS: usize = 32;
15
16#[derive(Serialize)]
17struct Instruction {
18    address: String,
19    bytes: Vec<u8>,
20    text: String,
21    current: bool,
22}
23
24#[derive(Deserialize, Serialize)]
25struct Disassembly {
26    address: String,
27    bytes: Vec<u8>,
28    #[serde(skip_deserializing)]
29    instructions: Vec<Instruction>,
30    error: Option<String>,
31}
32
33impl Disassembly {
34    /// RIP is a known instruction boundary. Never guess boundaries by decoding
35    /// backwards.
36    fn decode(&mut self) -> Result<()> {
37        let address = u64::from_str_radix(self.address.trim_start_matches("0x"), 16)?;
38        self.instructions.clear();
39        let bytes = &self.bytes[..self.bytes.len().min(MAX_BYTES)];
40        let mut decoder = Decoder::with_ip(64, bytes, address, DecoderOptions::NONE);
41        let mut formatter = IntelFormatter::new();
42        formatter.options_mut().set_hex_prefix("0x");
43        formatter.options_mut().set_hex_suffix("");
44        formatter.options_mut().set_uppercase_hex(false);
45        for _ in 0..MAX_INSTRUCTIONS {
46            if !decoder.can_decode() {
47                break;
48            }
49            let offset = decoder.position();
50            let instruction = decoder.decode();
51            if instruction.is_invalid() {
52                let reason = if decoder.last_error() == DecoderError::NoMoreBytes {
53                    "The captured range ends mid-instruction."
54                } else {
55                    "Invalid or unsupported instruction encoding."
56                };
57                let detail = format!("0x{:016x}: {reason}", instruction.ip());
58                self.error = Some(match self.error.take() {
59                    Some(error) => format!("{error} {detail}"),
60                    None => detail,
61                });
62                break;
63            }
64            let mut text = String::new();
65            formatter.format(&instruction, &mut text);
66            self.instructions.push(Instruction {
67                address: format!("0x{:016x}", instruction.ip()),
68                bytes: bytes[offset..decoder.position()].to_vec(),
69                text,
70                current: offset == 0,
71            });
72        }
73        Ok(())
74    }
75}
76
77/// IPC observations carry captured bytes and acquisition errors. Decode only
78/// after receipt, before any HTTP serialization, preserving acquisition errors.
79pub(super) fn decode_response(response: &mut MonitorResponse) -> Result<()> {
80    let value = match response {
81        MonitorResponse::Data(value) => value,
82        MonitorResponse::Started {
83            initial: MonitorEvent::Observation(value),
84            ..
85        }
86        | MonitorResponse::Event {
87            event: Some(MonitorEvent::Observation(value)),
88            ..
89        } => value,
90        _ => return Ok(()),
91    };
92    if let Some(samples) = value.get_mut("live_samples").and_then(Value::as_array_mut) {
93        for sample in samples {
94            if let Some(code) = sample.get_mut("disassembly").filter(|code| !code.is_null()) {
95                let mut decoded: Disassembly = serde_json::from_value(code.take())?;
96                decoded.decode()?;
97                *code = serde_json::to_value(decoded)?;
98            }
99        }
100    }
101    Ok(())
102}

118    #[test]
119    fn decodes_initial_and_polled_observations_from_native_instruction_bytes() {
120        let code = crate::ipc::acquisition::InstructionBytes {
121            address: 0x1000,
122            bytes: vec![144, 72, 139],
123            error: Some("Partial read.".into()),
124        };
125        let code: crate::ipc::acquisition::InstructionBytes =
126            crate::ipc::codec::decode(&crate::ipc::codec::encode(&code, 4096).unwrap()).unwrap();
127        let raw = serde_json::json!({"live_samples": [
128            {"tid": 123, "disassembly": code},
129            {"tid": 124, "disassembly": null},
130            {"tid": 125, "disassembly": {"address": "0x0000000000002000", "bytes": [], "error": "Read failed."}}
131        ]});
132        let responses = [
133            MonitorResponse::Data(raw.clone()),
134            MonitorResponse::Started {
135                session: 1,
136                initial: MonitorEvent::Observation(raw.clone()),
137            },
138            MonitorResponse::Event {
139                event: Some(MonitorEvent::Observation(raw)),
140                closed: false,
141            },
142        ];
143        for mut response in responses {
144            decode_response(&mut response).unwrap();
145            let value = match response {
146                MonitorResponse::Data(value)
147                | MonitorResponse::Started {
148                    initial: MonitorEvent::Observation(value),
149                    ..
150                }
151                | MonitorResponse::Event {
152                    event: Some(MonitorEvent::Observation(value)),
153                    ..
154                } => value,
155                _ => unreachable!(),
156            };
157            let samples = &value["live_samples"];
158            assert_eq!(samples[0]["tid"], 123);
159            let code = &samples[0]["disassembly"];
160            assert_eq!(code["address"], "0x0000000000001000");
161            assert_eq!(code["bytes"], serde_json::json!([144, 72, 139]));
162            assert_eq!(
163                code["instructions"][0],
164                serde_json::json!({"address": "0x0000000000001000", "bytes": [144], "text": "nop", "current": true})
165            );
166            assert!(code["error"].as_str().unwrap().starts_with("Partial read."));
167            assert!(code["error"].as_str().unwrap().contains("mid-instruction"));
168            assert!(samples[1]["disassembly"].is_null());
169            assert_eq!(samples[2]["disassembly"]["error"], "Read failed.");
170            assert_eq!(
171                samples[2]["disassembly"]["instructions"],
172                serde_json::json!([])
173            );
174        }
175        decode_response(&mut MonitorResponse::Closed).unwrap();
176        decode_response(&mut MonitorResponse::Data(
177            serde_json::json!({"live_samples": []}),
178        ))
179        .unwrap();
180    }

182    #[test]
183    fn decodes_variable_length_instructions_and_runtime_branch_target() {
184        let mut code = Disassembly::empty(0x7fff_1234_5000);
185        code.bytes = vec![0x55, 0x48, 0x89, 0xe5, 0xe8, 0x10, 0, 0, 0, 0xc3];
186        code.decode().unwrap();
187        assert!(code.error.is_none());
188        assert_eq!(code.instructions.len(), 4);
189        assert_eq!(code.instructions[0].text, "push rbp");
190        assert_eq!(code.instructions[1].text, "mov rbp,rsp");
191        assert_eq!(code.instructions[1].address, "0x00007fff12345001");
192        assert_eq!(code.instructions[2].bytes.len(), 5);
193        assert!(code.instructions[2].text.contains("7fff12345019"));
194        assert!(code.instructions[0].current);
195        assert!(code.instructions[1..].iter().all(|i| !i.current));
196        let json = serde_json::to_value(&code).unwrap();
197        assert_eq!(json["address"], "0x00007fff12345000");
198    }

200    #[test]
201    fn stops_on_truncated_or_invalid_instructions_and_bounds_output() {
202        let mut code = Disassembly::empty(0x1000);
203        code.bytes = vec![0x90, 0x48, 0x8b];
204        code.decode().unwrap();
205        assert_eq!(code.instructions.len(), 1);
206        assert!(code.error.as_ref().unwrap().contains("mid-instruction"));
207        let mut invalid = Disassembly::empty(0x1000);
208        invalid.bytes = vec![0xf0, 0x90]; // LOCK NOP is invalid.
209        invalid.decode().unwrap();
210        assert!(invalid.instructions.is_empty());
211        assert!(invalid.error.is_some());
212        let mut many = Disassembly::empty(0x1000);
213        many.bytes = vec![0x90; 512];
214        many.decode().unwrap();
215        assert_eq!(many.instructions.len(), MAX_INSTRUCTIONS);
216    }
