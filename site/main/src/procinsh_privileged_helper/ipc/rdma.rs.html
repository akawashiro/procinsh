20/// `enum ib_qp_type`.
21#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
22pub enum RdmaQpType {
23    Smi,
24    Gsi,
25    Rc,
26    Uc,
27    Ud,
28    RawIpv6,
29    RawEthertype,
30    RawPacket,
31    XrcIni,
32    XrcTgt,
33    Driver,
34    Unknown(u8),
35}
36impl RdmaQpType {
37    pub fn from_code(code: u8) -> Self {
38        match code {
39            0 => Self::Smi,
40            1 => Self::Gsi,
41            2 => Self::Rc,
42            3 => Self::Uc,
43            4 => Self::Ud,
44            5 => Self::RawIpv6,
45            6 => Self::RawEthertype,
46            8 => Self::RawPacket,
47            9 => Self::XrcIni,
48            10 => Self::XrcTgt,
49            0xff => Self::Driver,
50            n => Self::Unknown(n),
51        }
52    }
53
54    /// Types whose QP addresses exactly one remote QP.
55    pub fn is_connected(self) -> bool {
56        matches!(self, Self::Rc | Self::Uc | Self::XrcIni | Self::XrcTgt)
57    }
58}

60    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
61        f.write_str(match self {
62            Self::Smi => "SMI",
63            Self::Gsi => "GSI",
64            Self::Rc => "RC",
65            Self::Uc => "UC",
66            Self::Ud => "UD",
67            Self::RawIpv6 => "RAW_IPV6",
68            Self::RawEthertype => "RAW_ETHERTYPE",
69            Self::RawPacket => "RAW_PACKET",
70            Self::XrcIni => "XRC_INI",
71            Self::XrcTgt => "XRC_TGT",
72            Self::Driver => "DRIVER",
73            Self::Unknown(code) => return write!(f, "TYPE {code}"),
74        })
75    }

204/// A 128-bit IB global identifier, written like sysfs `gids/N`.
205#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
206pub struct IbGid(pub [u8; 16]);
207impl IbGid {
208    pub fn from_slice(bytes: &[u8]) -> Option<Self> {
209        let gid = Self(bytes.try_into().ok()?);
210        (gid.0 != [0; 16]).then_some(gid)
211    }
212
213    /// Parses `fe80:0000:...:45c7`; the all-zero GID is unassigned.
214    pub fn parse(text: &str) -> Option<Self> {
215        let groups: Vec<_> = text.trim().split(':').collect();
216        if groups.len() != 8 {
217            return None;
218        }
219        let mut bytes = [0u8; 16];
220        for (index, group) in groups.iter().enumerate() {
221            if group.len() != 4 {
222                return None;
223            }
224            bytes[index * 2..index * 2 + 2]
225                .copy_from_slice(&u16::from_str_radix(group, 16).ok()?.to_be_bytes());
226        }
227        Self::from_slice(&bytes)
228    }
229}
230impl fmt::Display for IbGid {
231    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
232        for (index, pair) in self.0.chunks(2).enumerate() {
233            if index > 0 {
234                f.write_str(":")?;
235            }
236            write!(f, "{:02x}{:02x}", pair[0], pair[1])?;
237        }
238        Ok(())
239    }
240}

242/// A connection manager endpoint address from `__kernel_sockaddr_storage`.
243#[derive(Clone, Copy, Debug, PartialEq, Eq)]
244pub enum RdmaAddress {
245    Inet {
246        ip: IpAddr,
247        port: u16,
248    },
249    /// `struct sockaddr_ib`; the 64-bit service ID is a hexadecimal string.
250    Ib {
251        gid: IbGid,
252        service_id: u64,
253        pkey: u16,
254    },
255    Unknown {
256        family: u16,
257    },
258}
259const AF_IB: u16 = 27;
260impl RdmaAddress {
261    /// Decodes a CM address. A zero family means no address was assigned.
262    pub fn decode(bytes: &[u8]) -> Option<Self> {
263        let family = u16::from_ne_bytes(bytes.get(0..2)?.try_into().ok()?);
264        let port = || Some(u16::from_be_bytes(bytes.get(2..4)?.try_into().ok()?));
265        Some(match family {
266            0 => return None,
267            f if f == libc::AF_INET as u16 => Self::Inet {
268                ip: IpAddr::from(<[u8; 4]>::try_from(bytes.get(4..8)?).ok()?),
269                port: port()?,
270            },
271            f if f == libc::AF_INET6 as u16 => {
272                let ip = std::net::Ipv6Addr::from(<[u8; 16]>::try_from(bytes.get(8..24)?).ok()?);
273                Self::Inet {
274                    ip: ip
275                        .to_ipv4_mapped()
276                        .map(IpAddr::V4)
277                        .unwrap_or(IpAddr::V6(ip)),
278                    port: port()?,
279                }
280            }
281            AF_IB => Self::Ib {
282                pkey: port()?,
283                gid: IbGid::from_slice(bytes.get(8..24)?)?,
284                service_id: u64::from_be_bytes(bytes.get(24..32)?.try_into().ok()?),
285            },
286            family => Self::Unknown { family },
287        })
288    }
289
290    pub fn ip(&self) -> Option<IpAddr> {
291        match self {
292            Self::Inet { ip, .. } => Some(*ip),
293            _ => None,
294        }
295    }
296}
297impl fmt::Display for RdmaAddress {
298    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
299        match self {
300            Self::Inet { ip, port } => write!(f, "{}", std::net::SocketAddr::new(*ip, *port)),
301            Self::Ib {
302                gid, service_id, ..
303            } => write!(f, "{gid} SID 0x{service_id:016x}"),
304            Self::Unknown { family } => write!(f, "AF {family}"),
305        }
306    }
307}
