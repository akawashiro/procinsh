9use crate::ipc::model::socket_types::{SocketConnectionStatus, SocketProtocol, SocketState};
10use std::net::SocketAddr;
11#[derive(Clone, Debug)]
12/// Socket kind and independently observed connection metadata.
13/// A known protocol does not imply a known state, address or peer.
14pub struct SocketInfo {
15    pub protocol: SocketProtocol,
16    pub state: Option<SocketState>,
17    pub local: Option<SocketAddr>,
18    pub remote: Option<SocketAddr>,
19    pub path: Option<String>,
20    pub peer_inode: Option<u64>,
21}
22
23impl SocketInfo {
24    /// Preserve direct observations; table fallbacks only confirm positive facts.
25    /// CLOSE, zero addresses, and UNIX state names never prove ENOTCONN.
26    pub fn connection_status(&self, observed: SocketConnectionStatus) -> SocketConnectionStatus {
27        if observed != SocketConnectionStatus::Unknown {
28            observed
29        } else if self.state == Some(SocketState::Listen) {
30            SocketConnectionStatus::Listening
31        } else if self.peer_inode.is_some()
32            || (self.protocol.is_tcp_or_udp()
33                && self
34                    .remote
35                    .is_some_and(|remote| remote.port() != 0 && !remote.ip().is_unspecified()))
36        {
37            SocketConnectionStatus::PeerPresent
38        } else {
39            SocketConnectionStatus::Unknown
40        }
41    }
42}
