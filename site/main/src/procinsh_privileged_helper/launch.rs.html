1//! Validates the inherited private endpoint and owns helper lifetime.
2//!
3//! # Interface
4//! [`run`] (`pub(crate)`): `fn run() -> anyhow::Result<()>`.
5use super::helper_service::{MonitorService, MonitorState};
6use crate::ipc::{schema::*, transport};
7use anyhow::{Result, ensure};
8use std::{
9    net::Shutdown,
10    os::{
11        fd::{AsRawFd, FromRawFd},
12        unix::net::UnixStream,
13    },
14    sync::{
15        Arc,
16        atomic::{AtomicBool, Ordering},
17        mpsc,
18    },
19    time::{Duration, Instant},
20};
21
22static STOP: AtomicBool = AtomicBool::new(false);
23extern "C" fn stop(_: i32) {
24    STOP.store(true, Ordering::Relaxed);
25}
26
27pub(crate) fn run() -> Result<()> {
28    let args: Vec<_> = std::env::args().skip(1).collect();
29    ensure!(
30        args.len() == 4 && args[0] == "--ipc-fd" && args[1] == "3" && args[2] == "--parent-pid",
31        "helper requires an inherited private IPC endpoint"
32    );
33    let parent: i32 = args[3].parse()?;
34    let mut peer: libc::ucred = unsafe { std::mem::zeroed() };
35    let mut size = std::mem::size_of_val(&peer) as libc::socklen_t;
36    ensure!(
37        unsafe {
38            libc::getsockopt(
39                3,
40                libc::SOL_SOCKET,
41                libc::SO_PEERCRED,
42                &mut peer as *mut _ as *mut _,
43                &mut size,
44            )
45        } == 0
46            && peer.pid == parent
47            && parent == unsafe { libc::getppid() }
48            && peer.uid == unsafe { libc::getuid() },
49        "invalid IPC parent identity"
50    );
51    let mut socket = unsafe { UnixStream::from_raw_fd(3) };
52    ensure!(
53        unsafe { libc::fcntl(3, libc::F_SETFD, libc::FD_CLOEXEC) } == 0,
54        "seal inherited endpoint"
55    );
56    super::privileges::restrict_monitor_privileges()?;
57    let mut action: libc::sigaction = unsafe { std::mem::zeroed() };
58    action.sa_sigaction = stop as *const () as libc::sighandler_t;
59    unsafe {
60        libc::sigemptyset(&mut action.sa_mask);
61    }
62    for signal in [libc::SIGTERM, libc::SIGINT] {
63        ensure!(
64            unsafe { libc::sigaction(signal, &action, std::ptr::null_mut()) } == 0,
65            "install shutdown signal handler"
66        );
67    }
68    ensure!(
69        unsafe { libc::prctl(libc::PR_SET_PDEATHSIG, libc::SIGTERM, 0, 0, 0) } == 0
70            && unsafe { libc::getppid() } == parent,
71        "parent exited during helper startup"
72    );
73    transport::write(
74        &mut socket,
75        &HelperIdentity {
76            pid: std::process::id(),
77            capabilities: super::privileges::effective_capabilities()?,
78        },
79        MAX_REQUEST,
80    )?;
81    let state = Arc::new(MonitorState::new(Duration::from_secs(1)));
82    let service = MonitorService::new(state.clone());
83    let shutdown = socket.try_clone()?;
84    let (finished, completion) = mpsc::channel();
85    std::thread::Builder::new()
86        .name("privileged-ipc".into())
87        .spawn(move || {
88            let _ = finished.send(service.serve(socket));
89        })?;
90    loop {
91        match completion.try_recv() {
92            Ok(result) => {
93                if let Err(error) = result {
94                    log::debug!("helper IPC ended: {error:#}");
95                }
96                break;
97            }
98            Err(mpsc::TryRecvError::Disconnected) => anyhow::bail!("helper IPC worker panicked"),
99            Err(mpsc::TryRecvError::Empty) => {}
100        }
101        let mut fd = libc::pollfd {
102            fd: shutdown.as_raw_fd(),
103            events: libc::POLLRDHUP,
104            revents: 0,
105        };
106        unsafe {
107            libc::poll(&mut fd, 1, 100);
108        }
109        if STOP.load(Ordering::Relaxed)
110            || fd.revents & (libc::POLLRDHUP | libc::POLLHUP | libc::POLLERR) != 0
111        {
112            break;
113        }
114    }
115    state.stop();
116    let _ = shutdown.shutdown(Shutdown::Both);
117    let (done, joined) = mpsc::channel();
118    std::thread::spawn(move || {
119        let _ = done.send(state.join_collectors());
120    });
121    let deadline = Instant::now() + Duration::from_secs(5);
122    joined.recv_timeout(deadline.saturating_duration_since(Instant::now()))??;
123    Ok(())
124}
