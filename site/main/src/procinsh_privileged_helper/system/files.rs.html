22pub(super) const COVERAGE: &str = "Regular-file read/write, pread/pwrite and vectored I/O, including page cache. Not physical disk traffic. mmap, io_uring, splice/sendfile and kernel workers are not observed.";
23const MAX_AGGREGATES: usize = 8192;
24use crate::ipc::activity::FileActivity;
25#[derive(Default)]
26struct Batch {
27    events: Vec<FileActivity>,
28    dropped: u64,
29}
30fn add_batch(state: &mut Batch, event: FileActivity) {
31    if state.events.len() < MAX_AGGREGATES {
32        state.events.push(event);
33    } else {
34        state.dropped += 1;
35    }
36}
37/// Owns BPF resources and captures bounded regular-file I/O observations.
38pub(super) struct FileActivityCollector {
39    ring: libbpf_rs::RingBuffer<'static>,
40    _links: Vec<libbpf_rs::Link>,
41    obj: libbpf_rs::Object,
42    batch: Arc<Mutex<Batch>>,
43}
44pub(super) fn create_file_activity_collector() -> Result<FileActivityCollector> {
45    let obj = ObjectBuilder::default()
46        .open_memory(include_bytes!(concat!(env!("OUT_DIR"), "/files.bpf.o")))?
47        .load()
48        .context("File I/O requires CAP_BPF / CAP_PERFMON and compatible VFS BTF hooks")?;
49    let mut links = Vec::new();
50    for prog in obj.progs_mut() {
51        links.push(
52            prog.attach()
53                .with_context(|| format!("attach {:?}", prog.name()))?,
54        );
55    }
56    let batch = Arc::new(Mutex::new(Batch::default()));
57    let shared = batch.clone();
58    let map = obj
59        .maps()
60        .find(|m| m.name() == "events")
61        .context("file events map")?;
62    let mut builder = RingBufferBuilder::new();
63    builder.add(&map, move |data| {
64        let mut batch = shared.lock().unwrap();
65        if let Some(event) = decode_file(data) {
66            add_batch(&mut batch, event);
67        } else {
68            batch.dropped += 1;
69        }
70        0
71    })?;
72    Ok(FileActivityCollector {
73        ring: builder.build()?,
74        _links: links,
75        obj,
76        batch,
77    })
78}

80pub(super) fn poll_file_activity_collector(state: &FileActivityCollector) -> Result<()> {
81    let result = state.ring.consume_raw_n(8192);
82    if result < 0 {
83        return Err(std::io::Error::from_raw_os_error(-result).into());
84    }
85    Ok(())
86}

88pub(super) fn drain_file_activity_collector(state: &FileActivityCollector) -> Vec<FileActivity> {
89    std::mem::take(&mut state.batch.lock().unwrap().events)
90}

92pub(super) fn lost_file_activity_collector(state: &FileActivityCollector) -> u64 {
93    state
94        .obj
95        .maps()
96        .find(|m| m.name() == "lost")
97        .and_then(|m| m.lookup(&0u32.to_ne_bytes(), MapFlags::ANY).ok().flatten())
98        .and_then(|bytes| bytes.try_into().ok())
99        .map(u64::from_ne_bytes)
100        .unwrap_or(0)
101        + state.batch.lock().unwrap().dropped
102}

104fn decode_file(data: &[u8]) -> Option<FileActivity> {
105    use crate::ipc::model::resource::device_id_from_kernel;
106
107    if data.len() != 4144 {
108        return None;
109    }
110    let u64_at = |i| u64::from_ne_bytes(data[i..i + 8].try_into().unwrap());
111    let u32_at = |i| u32::from_ne_bytes(data[i..i + 4].try_into().unwrap());
112    let bytes = u64_at(24);
113    if bytes == 0 || u32_at(36) > 1 {
114        return None;
115    }
116    let len = u32_at(40) as usize;
117    let path = if (2..=4096).contains(&len) && data[48 + len - 1] == 0 {
118        Some(String::from_utf8_lossy(&data[48..48 + len - 1]).into_owned())
119    } else {
120        None
121    };
122    let dev = u64_at(16);
123    Some(FileActivity {
124        process_id: ProcessId {
125            pid: u32_at(32) as i32,
126            start_time_ticks: ((u64_at(0) as u128
127                * crate::ipc::model::process::ticks_per_second() as u128)
128                / 1_000_000_000) as u64,
129        },
130        file: FileIdentity {
131            device: device_id_from_kernel(dev),
132            inode: u64_at(8),
133            generation: u32_at(44),
134        },
135        path,
136        write: u32_at(36) != 0,
137        bytes,
138        count: 1,
139    })
140}

198    #[test]
199    fn aggregation_separates_direction_and_process_lifetime_and_is_bounded() {
200        use crate::http_server::analysis::activity::add_file_batch;
201
202        let e = decode_file(&event()).unwrap();
203        let mut batch = FileBatch::default();
204        add_file_batch(&mut batch, e.clone());
205        add_file_batch(&mut batch, e.clone());
206        let mut write = e.clone();
207        write.write = true;
208        add_file_batch(&mut batch, write);
209        let mut reused = e.clone();
210        reused.process_id.start_time_ticks += 1;
211        add_file_batch(&mut batch, reused);
212        assert_eq!(batch.events.len(), 3);
213        assert_eq!(batch.events[&(e.process_id, e.file, false)].bytes, 14);
214        for i in 0..MAX_AGGREGATES {
215            let mut next = e.clone();
216            next.file.inode = i as u64 + 1000;
217            add_file_batch(&mut batch, next);
218        }
219        assert_eq!(batch.events.len(), MAX_AGGREGATES);
220        assert_eq!(batch.dropped, 3);
221        add_file_batch(&mut batch, e.clone());
222        assert_eq!(batch.events[&(e.process_id, e.file, false)].count, 3);
223    }
