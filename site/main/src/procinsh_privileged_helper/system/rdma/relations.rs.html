18struct Endpoint<'a> {
19    entry: &'a QueuePairEntry,
20    cm: Option<&'a CmIdEntry>,
21    remote_gid: Option<IbGid>,
22}
23impl Endpoint<'_> {
24    /// QPNs are unique per device, except the SMI and GSI QPs (0 and 1) of each port.
25    fn id(&self) -> String {
26        match self.entry.port {
27            Some(port) => format!("rdma:{}/{port}:{}", self.entry.device, self.entry.qpn),
28            None => format!("rdma:{}:{}", self.entry.device, self.entry.qpn),
29        }
30    }
31}
32
33fn resolve_rdma_owner(
34    owner: &ResourceOwner,
35    processes: &HashMap<i32, ProcessId>,
36) -> RdmaEndpointOwner {
37    match owner {
38        ResourceOwner::Kernel(name) => RdmaEndpointOwner::Kernel { name: name.clone() },
39        ResourceOwner::Process(pid) => match processes.get(&(*pid as i32)) {
40            Some(id) => RdmaEndpointOwner::Process { process_id: *id },
41            None => RdmaEndpointOwner::Unknown { pid: Some(*pid) },
42        },
43        ResourceOwner::Unknown => RdmaEndpointOwner::Unknown { pid: None },
44    }
45}

47fn build_rdma_endpoint(
48    entry: &QueuePairEntry,
49    processes: &HashMap<i32, ProcessId>,
50) -> RdmaEndpoint {
51    RdmaEndpoint {
52        owner: resolve_rdma_owner(&entry.owner, processes),
53        device: entry.device.clone(),
54        port: entry.port,
55        qpn: entry.qpn,
56        qp_type: entry.qp_type,
57        state: entry.state,
58    }
59}

61/// Whether two QPs with swapped QPNs are the two ends of one connection on this host.
62/// QPNs are unique per device only, so another device needs address evidence, and
63/// any contradicting address rejects the pair.
64fn same_connection(a: &Endpoint, b: &Endpoint, gids: &HashMap<(&str, u32), IbGid>) -> bool {
65    let local = |e: &Endpoint| {
66        e.entry
67            .port
68            .and_then(|port| gids.get(&(e.entry.device.as_str(), port)).copied())
69    };
70    let gid_match = |from: &Endpoint, to: &Endpoint| match (from.remote_gid, local(to)) {
71        (Some(remote), Some(local)) => Some(remote == local),
72        _ => None,
73    };
74    let cm_match = match (a.cm, b.cm) {
75        (Some(a), Some(b)) => match (a.source, a.destination, b.source, b.destination) {
76            (Some(a_src), Some(a_dst), Some(b_src), Some(b_dst)) => {
77                Some(a_src == b_dst && a_dst == b_src)
78            }
79            _ => None,
80        },
81        _ => None,
82    };
83    let evidence = [gid_match(a, b), gid_match(b, a), cm_match];
84    if evidence.contains(&Some(false)) {
85        return false;
86    }
87    a.entry.device == b.entry.device || evidence.contains(&Some(true))
88}

90fn format_rdma_relation_label(endpoint: &Endpoint, peer: Option<&Endpoint>) -> String {
91    let entry = endpoint.entry;
92    let mut label = format!("RDMA {} QP {}", entry.qp_type, entry.qpn);
93    if let Some(peer) = peer {
94        label += &format!(" ↔ QP {}", peer.entry.qpn);
95    } else if let Some(remote) = entry.remote_qpn {
96        label += &format!(" → QP {remote}");
97    }
98    if peer.is_none() {
99        if let Some(destination) = endpoint.cm.and_then(|cm| cm.destination) {
100            label += &format!(" {destination}");
101        } else if let Some(gid) = endpoint.remote_gid {
102            label += &format!(" {gid}");
103        }
104    }
105    label
106}

108/// Builds one relation per unpaired QP and one per QP pair on this host.
109/// `processes` maps PIDs of this snapshot to their identities.
110pub(super) fn build_rdma_relations(
111    resources: &NldevResources,
112    ports: &[RdmaPortInfo],
113    neighbors: &HashMap<IpAddr, IbGid>,
114    processes: &HashMap<i32, ProcessId>,
115) -> Vec<RdmaRelation> {
116    let gids: HashMap<_, _> = ports
117        .iter()
118        .filter_map(|p| Some(((p.device.as_str(), p.port), p.gid?)))
119        .collect();
120    let mut cms: HashMap<(&str, u32), &CmIdEntry> = HashMap::new();
121    for cm in &resources.cm_ids {
122        if let Some(qpn) = cm.qpn {
123            cms.entry((cm.device.as_str(), qpn)).or_insert(cm);
124        }
125    }
126    let endpoints: Vec<_> = resources
127        .queue_pairs
128        .iter()
129        .map(|entry| {
130            let cm = cms.get(&(entry.device.as_str(), entry.qpn)).copied();
131            let remote_gid = cm.and_then(|cm| match cm.destination? {
132                RdmaAddress::Ib { gid, .. } => Some(gid),
133                destination => neighbors.get(&destination.ip()?).copied(),
134            });
135            Endpoint {
136                entry,
137                cm,
138                remote_gid,
139            }
140        })
141        .collect();
142    let mut by_qpn: HashMap<u32, Vec<usize>> = HashMap::new();
143    for (index, e) in endpoints.iter().enumerate() {
144        if e.entry.qp_type.is_connected() && e.entry.remote_qpn.is_some() {
145            by_qpn.entry(e.entry.qpn).or_default().push(index);
146        }
147    }
148    let peer_of = |index: usize| -> Option<usize> {
149        let a = &endpoints[index];
150        if !a.entry.qp_type.is_connected() {
151            return None;
152        }
153        let matches: Vec<_> = by_qpn
154            .get(&a.entry.remote_qpn?)?
155            .iter()
156            .copied()
157            .filter(|&b| {
158                b != index
159                    && endpoints[b].entry.remote_qpn == Some(a.entry.qpn)
160                    && same_connection(a, &endpoints[b], &gids)
161            })
162            .collect();
163        (matches.len() == 1).then(|| matches[0])
164    };
165    let mut emitted = HashSet::new();
166    let mut result = Vec::new();
167    for (index, e) in endpoints.iter().enumerate() {
168        if emitted.contains(&index) {
169            continue;
170        }
171        // Pair only mutual unique matches, so each QP belongs to one relation.
172        let peer = peer_of(index).filter(|&b| peer_of(b) == Some(index));
173        let (first, second) = match peer {
174            Some(b) if endpoints[b].id() < e.id() => (b, Some(index)),
175            Some(b) => (index, Some(b)),
176            None => (index, None),
177        };
178        emitted.insert(first);
179        let first_endpoint = &endpoints[first];
180        let second_endpoint = second.map(|s| {
181            emitted.insert(s);
182            &endpoints[s]
183        });
184        result.push(RdmaRelation {
185            id: match second_endpoint {
186                Some(peer) => format!("{}/{}", first_endpoint.id(), peer.id()),
187                None => first_endpoint.id(),
188            },
189            endpoint: build_rdma_endpoint(first_endpoint.entry, processes),
190            peer: second_endpoint.map(|peer| build_rdma_endpoint(peer.entry, processes)),
191            remote_qpn: first_endpoint.entry.remote_qpn,
192            cm: first_endpoint.cm.map(|cm| RdmaCmInfo {
193                id: cm.id,
194                owner: resolve_rdma_owner(&cm.owner, processes),
195                state: cm.state,
196                port_space: cm.port_space,
197                source: cm.source,
198                destination: cm.destination,
199            }),
200            remote_gid: first_endpoint.remote_gid,
201            remote_hostname: None,
202            label: format_rdma_relation_label(first_endpoint, second_endpoint),
203        });
204    }
205    result
206}

361    #[test]
362    fn local_pairs_need_unique_swaps_and_no_contradicting_addresses() {
363        let process_owner_fixture = || ResourceOwner::Process(1);
364        // Same device: a QPN swap identifies a loopback connection.
365        let same = NldevResources {
366            queue_pairs: vec![
367                queue_pair_fixture("ib0", 20, Some(10), process_owner_fixture()),
368                queue_pair_fixture("ib0", 10, Some(20), process_owner_fixture()),
369            ],
370            ..Default::default()
371        };
372        let result = build_rdma_relations(&same, &[], &HashMap::new(), &HashMap::new());
373        assert_eq!(result.len(), 1);
374        assert_eq!(result[0].id, "rdma:ib0/1:10/rdma:ib0/1:20");
375        assert_eq!(result[0].peer.as_ref().unwrap().qpn, 20);
376        assert_eq!(result[0].label, "RDMA RC QP 10 ↔ QP 20");
377        // Different devices without address evidence remain separate.
378        let devices = NldevResources {
379            queue_pairs: vec![
380                queue_pair_fixture("ib0", 10, Some(20), process_owner_fixture()),
381                queue_pair_fixture("ib1", 20, Some(10), process_owner_fixture()),
382            ],
383            ..Default::default()
384        };
385        assert_eq!(
386            build_rdma_relations(&devices, &[], &HashMap::new(), &HashMap::new()).len(),
387            2
388        );
389        // Swapped CM addresses confirm a pair across devices.
390        let confirmed = NldevResources {
391            cm_ids: vec![
392                connection_id_fixture("ib0", 10, "10.0.0.1:1", "10.0.0.2:2"),
393                connection_id_fixture("ib1", 20, "10.0.0.2:2", "10.0.0.1:1"),
394            ],
395            ..devices
396        };
397        assert_eq!(
398            build_rdma_relations(&confirmed, &[], &HashMap::new(), &HashMap::new()).len(),
399            1
400        );
401        // A remote GID naming another port contradicts a same-device QPN swap.
402        let contradicted = NldevResources {
403            cm_ids: vec![connection_id_fixture("ib0", 10, "10.0.0.1:1", "10.0.0.9:2")],
404            ..same
405        };
406        let neighbors =
407            HashMap::from([("10.0.0.9".parse().unwrap(), IbGid::parse(REMOTE).unwrap())]);
408        let result = build_rdma_relations(
409            &contradicted,
410            &[rdma_port_fixture("ib0", LOCAL)],
411            &neighbors,
412            &HashMap::new(),
413        );
414        assert_eq!(result.len(), 2);
415        // An ambiguous swap is not paired.
416        let ambiguous = NldevResources {
417            queue_pairs: vec![
418                queue_pair_fixture("ib0", 10, Some(20), process_owner_fixture()),
419                queue_pair_fixture("ib1", 20, Some(10), process_owner_fixture()),
420                queue_pair_fixture("ib2", 20, Some(10), process_owner_fixture()),
421            ],
422            cm_ids: vec![
423                connection_id_fixture("ib0", 10, "10.0.0.1:1", "10.0.0.2:2"),
424                connection_id_fixture("ib1", 20, "10.0.0.2:2", "10.0.0.1:1"),
425                connection_id_fixture("ib2", 20, "10.0.0.2:2", "10.0.0.1:1"),
426            ],
427            ..Default::default()
428        };
429        assert_eq!(
430            build_rdma_relations(&ambiguous, &[], &HashMap::new(), &HashMap::new()).len(),
431            3
432        );
433    }
