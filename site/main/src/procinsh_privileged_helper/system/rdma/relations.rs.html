13struct Endpoint<'a> {
14    entry: &'a QueuePairEntry,
15    cm: Option<&'a CmIdEntry>,
16    remote_gid: Option<IbGid>,
17}
18impl Endpoint<'_> {
19    /// QPNs are unique per device, except the SMI and GSI QPs (0 and 1) of each port.
20    fn id(&self) -> String {
21        match self.entry.port {
22            Some(port) => format!("rdma:{}/{port}:{}", self.entry.device, self.entry.qpn),
23            None => format!("rdma:{}:{}", self.entry.device, self.entry.qpn),
24        }
25    }
26}
27
28fn owner(owner: &ResourceOwner, processes: &HashMap<i32, ProcessId>) -> RdmaEndpointOwner {
29    match owner {
30        ResourceOwner::Kernel(name) => RdmaEndpointOwner::Kernel { name: name.clone() },
31        ResourceOwner::Process(pid) => match processes.get(&(*pid as i32)) {
32            Some(id) => RdmaEndpointOwner::Process { process_id: *id },
33            None => RdmaEndpointOwner::Unknown { pid: Some(*pid) },
34        },
35        ResourceOwner::Unknown => RdmaEndpointOwner::Unknown { pid: None },
36    }
37}
38
39fn endpoint(entry: &QueuePairEntry, processes: &HashMap<i32, ProcessId>) -> RdmaEndpoint {
40    RdmaEndpoint {
41        owner: owner(&entry.owner, processes),
42        device: entry.device.clone(),
43        port: entry.port,
44        qpn: entry.qpn,
45        qp_type: entry.qp_type,
46        state: entry.state,
47    }
48}

50/// Whether two QPs with swapped QPNs are the two ends of one connection on this host.
51/// QPNs are unique per device only, so another device needs address evidence, and
52/// any contradicting address rejects the pair.
53fn same_connection(a: &Endpoint, b: &Endpoint, gids: &HashMap<(&str, u32), IbGid>) -> bool {
54    let local = |e: &Endpoint| {
55        e.entry
56            .port
57            .and_then(|port| gids.get(&(e.entry.device.as_str(), port)).copied())
58    };
59    let gid_match = |from: &Endpoint, to: &Endpoint| match (from.remote_gid, local(to)) {
60        (Some(remote), Some(local)) => Some(remote == local),
61        _ => None,
62    };
63    let cm_match = match (a.cm, b.cm) {
64        (Some(a), Some(b)) => match (a.source, a.destination, b.source, b.destination) {
65            (Some(a_src), Some(a_dst), Some(b_src), Some(b_dst)) => {
66                Some(a_src == b_dst && a_dst == b_src)
67            }
68            _ => None,
69        },
70        _ => None,
71    };
72    let evidence = [gid_match(a, b), gid_match(b, a), cm_match];
73    if evidence.contains(&Some(false)) {
74        return false;
75    }
76    a.entry.device == b.entry.device || evidence.contains(&Some(true))
77}

79fn label(endpoint: &Endpoint, peer: Option<&Endpoint>) -> String {
80    let entry = endpoint.entry;
81    let mut label = format!("RDMA {} QP {}", entry.qp_type, entry.qpn);
82    if let Some(peer) = peer {
83        label += &format!(" ↔ QP {}", peer.entry.qpn);
84    } else if let Some(remote) = entry.remote_qpn {
85        label += &format!(" → QP {remote}");
86    }
87    if peer.is_none() {
88        if let Some(destination) = endpoint.cm.and_then(|cm| cm.destination) {
89            label += &format!(" {destination}");
90        } else if let Some(gid) = endpoint.remote_gid {
91            label += &format!(" {gid}");
92        }
93    }
94    label
95}

97/// Builds one relation per unpaired QP and one per QP pair on this host.
98/// `processes` maps PIDs of this snapshot to their identities.
99pub(super) fn relations(
100    resources: &NldevResources,
101    ports: &[RdmaPortInfo],
102    neighbors: &HashMap<IpAddr, IbGid>,
103    processes: &HashMap<i32, ProcessId>,
104) -> Vec<RdmaRelation> {
105    let gids: HashMap<_, _> = ports
106        .iter()
107        .filter_map(|p| Some(((p.device.as_str(), p.port), p.gid?)))
108        .collect();
109    let mut cms: HashMap<(&str, u32), &CmIdEntry> = HashMap::new();
110    for cm in &resources.cm_ids {
111        if let Some(qpn) = cm.qpn {
112            cms.entry((cm.device.as_str(), qpn)).or_insert(cm);
113        }
114    }
115    let endpoints: Vec<_> = resources
116        .queue_pairs
117        .iter()
118        .map(|entry| {
119            let cm = cms.get(&(entry.device.as_str(), entry.qpn)).copied();
120            let remote_gid = cm.and_then(|cm| match cm.destination? {
121                RdmaAddress::Ib { gid, .. } => Some(gid),
122                destination => neighbors.get(&destination.ip()?).copied(),
123            });
124            Endpoint {
125                entry,
126                cm,
127                remote_gid,
128            }
129        })
130        .collect();
131    let mut by_qpn: HashMap<u32, Vec<usize>> = HashMap::new();
132    for (index, e) in endpoints.iter().enumerate() {
133        if e.entry.qp_type.is_connected() && e.entry.remote_qpn.is_some() {
134            by_qpn.entry(e.entry.qpn).or_default().push(index);
135        }
136    }
137    let peer_of = |index: usize| -> Option<usize> {
138        let a = &endpoints[index];
139        if !a.entry.qp_type.is_connected() {
140            return None;
141        }
142        let matches: Vec<_> = by_qpn
143            .get(&a.entry.remote_qpn?)?
144            .iter()
145            .copied()
146            .filter(|&b| {
147                b != index
148                    && endpoints[b].entry.remote_qpn == Some(a.entry.qpn)
149                    && same_connection(a, &endpoints[b], &gids)
150            })
151            .collect();
152        (matches.len() == 1).then(|| matches[0])
153    };
154    let mut emitted = HashSet::new();
155    let mut result = Vec::new();
156    for (index, e) in endpoints.iter().enumerate() {
157        if emitted.contains(&index) {
158            continue;
159        }
160        // Pair only mutual unique matches, so each QP belongs to one relation.
161        let peer = peer_of(index).filter(|&b| peer_of(b) == Some(index));
162        let (first, second) = match peer {
163            Some(b) if endpoints[b].id() < e.id() => (b, Some(index)),
164            Some(b) => (index, Some(b)),
165            None => (index, None),
166        };
167        emitted.insert(first);
168        let first_endpoint = &endpoints[first];
169        let second_endpoint = second.map(|s| {
170            emitted.insert(s);
171            &endpoints[s]
172        });
173        result.push(RdmaRelation {
174            id: match second_endpoint {
175                Some(peer) => format!("{}/{}", first_endpoint.id(), peer.id()),
176                None => first_endpoint.id(),
177            },
178            endpoint: endpoint(first_endpoint.entry, processes),
179            peer: second_endpoint.map(|peer| endpoint(peer.entry, processes)),
180            remote_qpn: first_endpoint.entry.remote_qpn,
181            cm: first_endpoint.cm.map(|cm| RdmaCmInfo {
182                id: cm.id,
183                owner: owner(&cm.owner, processes),
184                state: cm.state,
185                port_space: cm.port_space,
186                source: cm.source,
187                destination: cm.destination,
188            }),
189            remote_gid: first_endpoint.remote_gid,
190            remote_hostname: None,
191            label: label(first_endpoint, second_endpoint),
192        });
193    }
194    result
195}

329    #[test]
330    fn local_pairs_need_unique_swaps_and_no_contradicting_addresses() {
331        let owner = || ResourceOwner::Process(1);
332        // Same device: a QPN swap identifies a loopback connection.
333        let same = NldevResources {
334            queue_pairs: vec![
335                qp("ib0", 20, Some(10), owner()),
336                qp("ib0", 10, Some(20), owner()),
337            ],
338            ..Default::default()
339        };
340        let result = relations(&same, &[], &HashMap::new(), &HashMap::new());
341        assert_eq!(result.len(), 1);
342        assert_eq!(result[0].id, "rdma:ib0/1:10/rdma:ib0/1:20");
343        assert_eq!(result[0].peer.as_ref().unwrap().qpn, 20);
344        assert_eq!(result[0].label, "RDMA RC QP 10 ↔ QP 20");
345        // Different devices without address evidence remain separate.
346        let devices = NldevResources {
347            queue_pairs: vec![
348                qp("ib0", 10, Some(20), owner()),
349                qp("ib1", 20, Some(10), owner()),
350            ],
351            ..Default::default()
352        };
353        assert_eq!(
354            relations(&devices, &[], &HashMap::new(), &HashMap::new()).len(),
355            2
356        );
357        // Swapped CM addresses confirm a pair across devices.
358        let confirmed = NldevResources {
359            cm_ids: vec![
360                cm("ib0", 10, "10.0.0.1:1", "10.0.0.2:2"),
361                cm("ib1", 20, "10.0.0.2:2", "10.0.0.1:1"),
362            ],
363            ..devices
364        };
365        assert_eq!(
366            relations(&confirmed, &[], &HashMap::new(), &HashMap::new()).len(),
367            1
368        );
369        // A remote GID naming another port contradicts a same-device QPN swap.
370        let contradicted = NldevResources {
371            cm_ids: vec![cm("ib0", 10, "10.0.0.1:1", "10.0.0.9:2")],
372            ..same
373        };
374        let neighbors =
375            HashMap::from([("10.0.0.9".parse().unwrap(), IbGid::parse(REMOTE).unwrap())]);
376        let result = relations(
377            &contradicted,
378            &[port("ib0", LOCAL)],
379            &neighbors,
380            &HashMap::new(),
381        );
382        assert_eq!(result.len(), 2);
383        // An ambiguous swap is not paired.
384        let ambiguous = NldevResources {
385            queue_pairs: vec![
386                qp("ib0", 10, Some(20), owner()),
387                qp("ib1", 20, Some(10), owner()),
388                qp("ib2", 20, Some(10), owner()),
389            ],
390            cm_ids: vec![
391                cm("ib0", 10, "10.0.0.1:1", "10.0.0.2:2"),
392                cm("ib1", 20, "10.0.0.2:2", "10.0.0.1:1"),
393                cm("ib2", 20, "10.0.0.2:2", "10.0.0.1:1"),
394            ],
395            ..Default::default()
396        };
397        assert_eq!(
398            relations(&ambiguous, &[], &HashMap::new(), &HashMap::new()).len(),
399            3
400        );
401    }
