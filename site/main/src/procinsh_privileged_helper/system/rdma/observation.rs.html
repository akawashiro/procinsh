23/// Observes RDMA resources visible in the collector's network and PID namespaces,
24/// with devices and ports read under the sysfs root `sys`. Failures become
25/// warnings; an IPoIB neighbor failure only omits remote GIDs.
26pub(in crate::privileged_helper::system) fn collect_rdma_observation(
27    sys: &Path,
28    processes: &HashMap<i32, ProcessId>,
29    deadline: Instant,
30) -> RdmaObservation {
31    use crate::ipc::rdma::{ip_rdma_address, parse_ib_gid};
32
33    let root = &sys.join("class/infiniband");
34    let mut result = RdmaObservation::default();
35    // Opening NETLINK_RDMA can autoload ib_core; a host without devices is not queried.
36    if ports::list_rdma_devices(root).is_empty() {
37        return result;
38    }
39    result.ports = ports::read_rdma_ports(root);
40    let resources = match nldev::read_resources(deadline) {
41        Ok(resources) => resources,
42        Err(error) => {
43            result.warnings.push(format!("RDMA: {error:#}"));
44            return result;
45        }
46    };
47    for error in &resources.errors {
48        result.warnings.push(format!("RDMA: {error}"));
49    }
50    if resources.truncated {
51        result.warnings.push(format!(
52            "RDMA queue pair limit of {} reached. Some queue pairs are missing.",
53            nldev::QUEUE_PAIR_LIMIT
54        ));
55    }
56    let mut neighbors = HashMap::new();
57    if resources.cm_ids.iter().any(|cm| {
58        cm.destination
59            .is_some_and(|d| ip_rdma_address(&d).is_some())
60    }) {
61        match ipoib::ipoib_gids(deadline) {
62            Ok(found) => {
63                neighbors = found
64                    .into_iter()
65                    .filter_map(|(ip, gid)| Some((ip, parse_ib_gid(&gid)?)))
66                    .collect()
67            }
68            Err(error) => result
69                .warnings
70                .push(format!("RDMA remote GIDs unavailable: {error:#}")),
71        }
72    }
73    result.relations =
74        relations::build_rdma_relations(&resources, &result.ports, &neighbors, processes);
75    result
76}
