22/// Owns a monitoring registration until dropped.

33/// Manages subscribers, collection workers, snapshots, and event delivery.
34pub(in crate::privileged_helper) struct SystemMonitor {
35    viewers: Mutex<usize>,
36    snapshot: RwLock<Arc<snapshot::SystemSnapshot>>,
37    events: broadcast::Sender<SystemMonitorEvent>,
38    stop: AtomicBool,
39    started: AtomicBool,
40    workers: Mutex<Vec<std::thread::JoinHandle<()>>>,
41}
42impl Default for SystemMonitor {
43    fn default() -> Self {
44        let (events, _) = broadcast::channel(16);
45        Self {
46            viewers: Mutex::new(0),
47            snapshot: RwLock::new(Arc::new(snapshot::SystemSnapshot::default())),
48            events,
49            stop: AtomicBool::new(false),
50            started: AtomicBool::new(false),
51            workers: Mutex::new(Vec::new()),
52        }
53    }
54}
55impl SystemMonitor {
56    pub(in crate::privileged_helper) fn stopped(&self) -> bool {
57        self.stop.load(Ordering::Relaxed)
58    }
59
60    pub(in crate::privileged_helper) fn stop(&self) {
61        let _viewers = self.viewers.lock().unwrap();
62        self.stop.store(true, Ordering::Relaxed);
63    }
64
65    pub(in crate::privileged_helper) fn join_workers(&self) -> anyhow::Result<()> {
66        let workers = std::mem::take(&mut *self.workers.lock().unwrap());
67        let mut failed = false;
68        for worker in workers {
69            failed |= worker.join().is_err();
70        }
71        anyhow::ensure!(!failed, "system monitoring worker panicked");
72        Ok(())
73    }
74
75    pub(in crate::privileged_helper) fn active(&self) -> bool {
76        !self.stopped() && *self.viewers.lock().unwrap() > 0
77    }
78
79    pub(in crate::privileged_helper) fn subscribe(
80        self: &Arc<Self>,
81    ) -> Result<Subscription, SubscribeError> {
82        let mut viewers = self.viewers.lock().unwrap();
83        if self.stopped() {
84            return Err(SubscribeError::Stopped);
85        }
86        if *viewers >= 32 {
87            return Err(SubscribeError::TooManySubscribers);
88        }
89        *viewers += 1;
90        let viewer = Subscription {
91            monitor: self.clone(),
92            receiver: self.events.subscribe(),
93            initial: self.snapshot.read().unwrap().clone(),
94        };
95        self.start();
96        drop(viewers);
97        Ok(viewer)
98    }
99
100    fn send(&self, event: SystemMonitorEvent) {
101        let _ = self.events.send(event);
102    }
103
104    pub(in crate::privileged_helper) fn snapshot(&self) -> Arc<SystemSnapshot> {
105        self.snapshot.read().unwrap().clone()
106    }
107
108    fn run_activity(&self) {
109        let mut collector: Option<activity::ActivityCollector> = None;
110        let mut last = Instant::now();
111        let mut logged = super::StatusLog::default();
112        log::info!("System activity worker started");
113        while !self.stopped() {
114            if !self.active() {
115                logged.observe(&SystemMonitorStatus::default());
116                collector = None;
117                std::thread::sleep(Duration::from_millis(100));
118                continue;
119            }
120            let collector = collector.get_or_insert_with(|| {
121                log::info!("System observation active");
122                activity::ActivityCollector::new()
123            });
124            let snapshot = self.snapshot();
125            collector.poll(&snapshot);
126            if last.elapsed() >= Duration::from_secs(1) {
127                let batch = collector.drain(monotonic_ns(), &snapshot);
128                let mut status = collector.status();
129                status.active = true;
130                self.send(SystemMonitorEvent::Activity(Arc::new(SystemActivity {
131                    captured_at: crate::privileged_helper::process::timestamp_ms(),
132                    window_ms: last.elapsed().as_millis() as u64,
133                    files: Vec::new(),
134                    ipc: Vec::new(),
135                    cpu: Vec::new(),
136                    signals: Vec::new(),
137                    raw_activity: Some(crate::ipc::activity::RawActivity {
138                        files: batch.files,
139                        ipc: batch.ipc,
140                        cpu: batch.cpu,
141                        signals: batch.signals,
142                    }),
143                    status,
144                })));
145                last = Instant::now();
146            }
147            logged.observe(&collector.status());
148            std::thread::sleep(Duration::from_millis(10));
149        }
150        log::info!("System activity worker stopped");
151    }
152
153    fn start(self: &Arc<Self>) {
154        if self.started.swap(true, Ordering::SeqCst) {
155            return;
156        }
157        let monitor = self.clone();
158        let mut workers = self.workers.lock().unwrap();
159        workers.push(spawn_worker("activity", move || monitor.run_activity()));
160        let monitor = self.clone();
161        workers.push(spawn_worker("snapshot", move || {
162            log::info!("System snapshot worker started");
163            let mut previous_warnings = Vec::new();
164            let mut discovery = crate::privileged_helper::process::ProcessScanner::default();
165            let mut full = Instant::now() - Duration::from_secs(10);
166            while !monitor.stopped() {
167                if !monitor.active() {
168                    full = Instant::now() - Duration::from_secs(10);
169                    std::thread::sleep(Duration::from_millis(100));
170                    continue;
171                }
172                if full.elapsed() >= Duration::from_secs(1) {
173                    let data = snapshot::collect(&mut discovery);
174                    if data.warnings != previous_warnings {
175                        if data.warnings.is_empty() {
176                            log::info!("System snapshot recovered");
177                        } else {
178                            for warning in &data.warnings {
179                                log::warn!("System snapshot: {warning}");
180                            }
181                        }
182                        previous_warnings = data.warnings.clone();
183                    }
184                    log::debug!(
185                        "System snapshot collected processes={} fd_relations={} rdma_relations={}",
186                        data.processes.len(),
187                        data.fd_relations.len(),
188                        data.rdma_relations.len()
189                    );
190                    let data = Arc::new(data);
191                    monitor.send(SystemMonitorEvent::Snapshot(data.clone()));
192                    *monitor.snapshot.write().unwrap() = data;
193                    full = Instant::now();
194                }
195                std::thread::sleep(Duration::from_millis(100));
196            }
197            log::info!("System snapshot worker stopped");
198        }));
199    }
200}

202pub(in crate::privileged_helper) fn monotonic_ns() -> u64 {
203    let mut ts = libc::timespec {
204        tv_sec: 0,
205        tv_nsec: 0,
206    };
207    unsafe { libc::clock_gettime(libc::CLOCK_MONOTONIC, &mut ts) };
208    ts.tv_sec as u64 * 1_000_000_000 + ts.tv_nsec as u64
209}

211fn spawn_worker(
212    name: &'static str,
213    work: impl FnOnce() + Send + 'static,
214) -> std::thread::JoinHandle<()> {
215    std::thread::spawn(move || {
216        if let Err(panic) = std::panic::catch_unwind(std::panic::AssertUnwindSafe(work)) {
217            log::error!("System {name} worker panicked");
218            std::panic::resume_unwind(panic);
219        }
220    })
