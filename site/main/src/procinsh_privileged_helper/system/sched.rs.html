11use anyhow::{Context, Result};
12use libbpf_rs::{MapCore, MapFlags, ObjectBuilder};
13/// Owns BPF resources and captures scheduler counters.
14pub(super) struct CpuActivityCollector {
15    _links: Vec<libbpf_rs::Link>,
16    obj: libbpf_rs::Object,
17}

20pub(super) fn create_cpu_activity_collector() -> Result<CpuActivityCollector> {
21    let open = ObjectBuilder::default()
22        .open_memory(include_bytes!(concat!(env!("OUT_DIR"), "/sched.bpf.o")))?;
23    let obj = open
24        .load()
25        .context("CAP_BPF / CAP_PERFMON and compatible BTF required")?;
26    let mut links = Vec::new();
27    for prog in obj.progs_mut() {
28        links.push(
29            prog.attach()
30                .with_context(|| format!("attach {:?}", prog.name()))?,
31        );
32    }
33    Ok(CpuActivityCollector { _links: links, obj })
34}

36pub(super) fn collect_cpu_activity_collector(
37    state: &mut CpuActivityCollector,
38    now: u64,
39    _snapshot: &crate::ipc::model::system::SystemSnapshot,
40) -> Result<crate::ipc::activity::CpuCounters> {
41    let current = state
42        .obj
43        .maps()
44        .find(|map| map.name() == "cpu_current")
45        .context("cpu_current map")?
46        .lookup_percpu(&0u32.to_ne_bytes(), MapFlags::ANY)?
47        .unwrap_or_default();
48    let totals = state
49        .obj
50        .maps()
51        .find(|map| map.name() == "cpu_totals")
52        .context("cpu_totals map")?;
53
54    let mut captured = Vec::new();
55    for key in totals.keys().take(65_536) {
56        if let Some(values) = totals.lookup_percpu(&key, MapFlags::ANY)? {
57            let values = values
58                .into_iter()
59                .filter_map(|bytes| {
60                    let runtime = u64::from_ne_bytes(bytes.get(..8)?.try_into().ok()?);
61                    let switches = u64::from_ne_bytes(bytes.get(8..16)?.try_into().ok()?);
62                    (runtime != 0 || switches != 0).then_some([runtime, switches])
63                })
64                .collect();
65            captured.push((key, values));
66        }
67    }
68    Ok(crate::ipc::activity::CpuCounters {
69        now,
70        current,
71        totals: captured,
72    })
73}
