1//! InfiniBand port GIDs of network destinations reached over IPoIB.
2//!
3//! An IPoIB neighbor's hardware address is 20 bytes: flags and the remote IPoIB
4//! QPN, then the remote port GID. `/proc/net/arp` truncates it, so neighbors are
5//! dumped with rtnetlink in the collector's network namespace. Only sockets of
6//! that namespace can use the result; other namespaces have their own tables.
7
8use crate::privileged_helper::netlink::{self, NetlinkDumpRequest};
9use anyhow::Result;
10use std::{collections::HashMap, net::IpAddr, path::Path, time::Instant};
11
12const RTM_NEWNEIGH: u16 = 28;
13const RTM_GETNEIGH: u16 = 30;
14const NDA_DST: u16 = 1;
15const NDA_LLADDR: u16 = 2;
16const NUD_INCOMPLETE: u16 = 0x01;
17const NUD_FAILED: u16 = 0x20;
18const IPOIB_HARDWARE_ADDRESS: usize = 20;
19const ARPHRD_INFINIBAND: &str = "32";
20
21/// A GID written like sysfs `gids/N`: eight groups of four hexadecimal digits.
22pub(super) fn format_gid(bytes: &[u8]) -> String {
23    bytes
24        .chunks(2)
25        .map(|pair| format!("{:02x}{:02x}", pair[0], pair[1]))
26        .collect::<Vec<_>>()
27        .join(":")
28}
29
30/// Whether any network interface under `sys` is an IPoIB interface.
31pub(super) fn has_ipoib_interface(sys: &Path) -> bool {
32    std::fs::read_dir(sys.join("class/net"))
33        .into_iter()
34        .flatten()
35        .flatten()
36        .any(|entry| {
37            std::fs::read_to_string(entry.path().join("type"))
38                .is_ok_and(|kind| kind.trim() == ARPHRD_INFINIBAND)
39        })
40}

42/// One `RTM_NEWNEIGH` body; only resolved IPoIB entries yield a GID.
43fn parse_neighbor(body: &[u8]) -> Result<Option<(IpAddr, String)>> {
44    anyhow::ensure!(body.len() >= 12, "short neighbor message");
45    let state = u16::from_ne_bytes(body[8..10].try_into()?);
46    if state & (NUD_INCOMPLETE | NUD_FAILED) != 0 {
47        return Ok(None);
48    }
49    let (mut destination, mut gid) = (None, None);
50    for (kind, data) in netlink::attributes(&body[12..])? {
51        match kind {
52            NDA_DST => {
53                destination = match data.len() {
54                    4 => Some(IpAddr::from(<[u8; 4]>::try_from(data)?)),
55                    16 => {
56                        let ip = std::net::Ipv6Addr::from(<[u8; 16]>::try_from(data)?);
57                        Some(
58                            ip.to_ipv4_mapped()
59                                .map(IpAddr::V4)
60                                .unwrap_or(IpAddr::V6(ip)),
61                        )
62                    }
63                    _ => None,
64                }
65            }
66            NDA_LLADDR if data.len() == IPOIB_HARDWARE_ADDRESS && data[4..] != [0; 16] => {
67                gid = Some(format_gid(&data[4..]))
68            }
69            _ => {}
70        }
71    }
72    Ok(destination.zip(gid))
73}

75/// Resolved IPoIB neighbors. An address seen with different GIDs is ambiguous and omitted.
76pub(super) fn ipoib_gids(deadline: Instant) -> Result<HashMap<IpAddr, String>> {
77    let fd = netlink::open_socket(libc::NETLINK_ROUTE, "NETLINK_ROUTE")?;
78    let mut found: HashMap<IpAddr, Option<String>> = HashMap::new();
79    // struct ndmsg with AF_UNSPEC dumps every neighbor table.
80    NetlinkDumpRequest {
81        label: "IPoIB neighbor dump",
82        message_type: RTM_GETNEIGH,
83        reply_type: RTM_NEWNEIGH,
84        sequence: 1,
85        payload: &[0u8; 12],
86    }
87    .run(&fd, deadline, |body| {
88        if let Some((ip, gid)) = parse_neighbor(body)? {
89            found
90                .entry(ip)
91                .and_modify(|known| {
92                    if known.as_ref() != Some(&gid) {
93                        *known = None;
94                    }
95                })
96                .or_insert(Some(gid));
97        }
98        Ok(())
99    })?;
100    Ok(found
101        .into_iter()
102        .filter_map(|(ip, gid)| Some((ip, gid?)))
103        .collect())
104}

106#[cfg(test)]
107mod tests {
108    use super::*;
109
110    fn attribute(kind: u16, data: &[u8]) -> Vec<u8> {
111        let mut bytes = Vec::new();
112        bytes.extend_from_slice(&((4 + data.len()) as u16).to_ne_bytes());
113        bytes.extend_from_slice(&kind.to_ne_bytes());
114        bytes.extend_from_slice(data);
115        bytes.resize((bytes.len() + 3) & !3, 0);
116        bytes
117    }
118
119    fn neighbor(state: u16, destination: &[u8], hardware: &[u8]) -> Vec<u8> {
120        let mut body = vec![0u8; 12];
121        body[8..10].copy_from_slice(&state.to_ne_bytes());
122        body.extend(attribute(NDA_DST, destination));
123        body.extend(attribute(NDA_LLADDR, hardware));
124        body
125    }

127    #[test]
128    fn ipoib_hardware_addresses_yield_remote_gids() {
129        let hardware = [
130            0x80, 0x00, 0x02, 0x22, 0xfe, 0x80, 0, 0, 0, 0, 0, 0, 0xf4, 0x52, 0x14, 0x03, 0x00,
131            0x80, 0xde, 0x51,
132        ];
133        let (ip, gid) = parse_neighbor(&neighbor(0x04, &[10, 10, 10, 1], &hardware))
134            .unwrap()
135            .unwrap();
136        assert_eq!(ip.to_string(), "10.10.10.1");
137        assert_eq!(gid, "fe80:0000:0000:0000:f452:1403:0080:de51");
138        // Ethernet neighbors and unresolved entries carry no GID.
139        assert!(
140            parse_neighbor(&neighbor(0x02, &[192, 168, 2, 1], &[0xde; 6]))
141                .unwrap()
142                .is_none()
143        );
144        assert!(
145            parse_neighbor(&neighbor(NUD_FAILED, &[10, 10, 10, 1], &hardware))
146                .unwrap()
147                .is_none()
148        );
149        assert!(parse_neighbor(&[0; 4]).is_err());
150    }

152    #[test]
153    fn ipoib_interfaces_are_found_by_hardware_type() {
154        let root = std::env::temp_dir().join(format!("procinsh-ipoib-{}", std::process::id()));
155        let _ = std::fs::remove_dir_all(&root);
156        for (name, kind) in [("eno1", "1"), ("ibp133s0", "32")] {
157            std::fs::create_dir_all(root.join("class/net").join(name)).unwrap();
158            std::fs::write(
159                root.join("class/net").join(name).join("type"),
160                format!("{kind}\n"),
161            )
162            .unwrap();
163        }
164        assert!(has_ipoib_interface(&root));
165        std::fs::remove_dir_all(root.join("class/net/ibp133s0")).unwrap();
166        assert!(!has_ipoib_interface(&root));
167        std::fs::remove_dir_all(&root).unwrap();
168    }
