40pub(super) fn page_size() -> u64 {
41    unsafe { libc::sysconf(libc::_SC_PAGESIZE) }.max(1) as u64
42}
43
44pub(in crate::privileged_helper) fn ticks_per_second() -> f64 {
45    unsafe { libc::sysconf(libc::_SC_CLK_TCK) }.max(1) as f64
46}

48fn parse_process_stat(text: &str) -> Result<Stat> {
49    let open = text.find('(').context("stat: missing comm")?;
50    let close = text.rfind(')').context("stat: missing comm end")?;
51    ensure!(close > open, "stat: invalid comm");
52    let f: Vec<_> = text[close + 1..].split_whitespace().collect();
53    let number = |i: usize| -> Result<u64> {
54        f.get(i)
55            .context("stat: missing field")?
56            .parse()
57            .context("stat: invalid number")
58    };
59    let signed = |i: usize| -> Result<i64> {
60        f.get(i)
61            .context("stat: missing field")?
62            .parse()
63            .context("stat: invalid signed number")
64    };
65    Ok(Stat {
66        pid: text[..open].trim().parse()?,
67        parent_pid: signed(1)? as i32,
68        name: text[open + 1..close].to_owned(),
69        state: f.first().context("stat: missing state")?.to_string(),
70        ticks: number(11)?.saturating_add(number(12)?),
71        start_time: number(19)?,
72        minor_faults: number(7)?,
73        major_faults: number(9)?,
74        priority: signed(15)?,
75        nice: signed(16)?,
76        thread_count: number(17)? as u32,
77        vms: number(20)?,
78        rss: (signed(21)?.max(0) as u64).saturating_mul(page_size()),
79        cpu: signed(36)? as i32,
80        policy: number(38)? as u32,
81    })
82}

84/// Read at most `limit` bytes, rejecting input that contains any additional byte.
85pub(super) fn read_bounded_bytes(reader: impl Read, limit: usize) -> Result<Vec<u8>> {
86    let mut bytes = Vec::new();
87    reader.take(limit as u64 + 1).read_to_end(&mut bytes)?;
88    ensure!(
89        bytes.len() <= limit,
90        "The read limit of {limit} bytes was exceeded."
91    );
92    Ok(bytes)
93}

95/// Bounds procfs text allocation even when a target creates very large maps.
96pub(super) fn read_procfs_text(path: &str) -> Result<String> {
97    const MAX_TEXT: u64 = 16 * 1024 * 1024;
98    let mut text = String::new();
99    fs::File::open(path)?
100        .take(MAX_TEXT + 1)
101        .read_to_string(&mut text)?;
102    ensure!(
103        text.len() as u64 <= MAX_TEXT,
104        "procfs text exceeds 16 MiB limit"
105    );
106    Ok(text)
107}

109pub(super) fn read_process_stat(path: &str) -> Result<Stat> {
110    parse_process_stat(&read_procfs_text(path).with_context(|| format!("read {path}"))?)
111}

113pub(in crate::privileged_helper) fn read_procfs_fields(
114    path: &str,
115) -> Result<HashMap<String, String>> {
116    Ok(read_procfs_text(path)?
117        .lines()
118        .filter_map(|line| line.split_once(':'))
119        .map(|(k, v)| (k.to_owned(), v.trim().to_owned()))
120        .collect())
121}

123pub(super) fn parse_procfs_u64_field(fields: &HashMap<String, String>, key: &str) -> Option<u64> {
124    fields.get(key)?.split_whitespace().next()?.parse().ok()
125}

129pub(super) fn read_process_io_stats(id: ProcessId) -> Result<IoStats> {
130    let pid = id.pid;
131    let f = read_procfs_fields(&format!("/proc/{pid}/io"))?;
132    Ok(IoStats {
133        read_bytes: parse_procfs_u64_field(&f, "read_bytes")
134            .ok_or_else(|| anyhow!("missing read_bytes"))?,
135        write_bytes: parse_procfs_u64_field(&f, "write_bytes")
136            .ok_or_else(|| anyhow!("missing write_bytes"))?,
137    })
138}

140#[cfg(test)]
141mod tests {
142    use super::*;
143
144    #[test]
145    fn stat_handles_spaces_and_parentheses_and_signed_priority() {
146        let mut f = vec!["0"; 50];
147        f[0] = "S";
148        f[1] = "7";
149        f[11] = "20";
150        f[12] = "3";
151        f[15] = "-40";
152        f[19] = "918";
153        f[21] = "-1";
154        let stat = parse_process_stat(&format!("42 (a ) tricky (name)) {}", f.join(" "))).unwrap();
155        assert_eq!(stat.name, "a ) tricky (name)");
156        assert_eq!(stat.parent_pid, 7);
157        assert_eq!(stat.ticks, 23);
158        assert_eq!(stat.priority, -40);
159        assert_eq!(stat.start_time, 918);
160        assert_eq!(stat.rss, 0);
161        assert!(parse_process_stat("42 (bad) S").is_err());
162    }
163}
