24fn executable_header(pid: i32) -> Option<[u8; 20]> {
25    use std::io::Read;
26    let mut header = [0; 20];
27    std::fs::File::open(format!("/proc/{pid}/exe"))
28        .and_then(|mut file| file.read_exact(&mut header))
29        .ok()?;
30    Some(header)
31}

99    /// Captures the initial ptrace snapshot and schedules the next attempt ten seconds
100    /// after completion, including failed capture attempts.
101    pub(in crate::privileged_helper::process) fn bootstrap(
102        &mut self,
103        maps: &[MemoryMap],
104    ) -> Result<()> {
105        // Discover all threads and open perf events before interrupting any thread.
106        self.next_ptrace = None;
107        self.poll(maps)?;
108        self.capture_ptrace(maps)
109    }
110    fn capture_ptrace(&mut self, maps: &[MemoryMap]) -> Result<()> {
111        for (&tid, thread) in &mut self.threads {
112            thread.ptrace_error = None;
113            thread.latest.error = thread.setup_error.clone();
114            match super::ptrace::capture(tid, maps) {
115                Ok(sample) => process_sample(
116                    self.id.pid,
117                    #[cfg(test)]
118                    &mut self.unwind,
119                    #[cfg(test)]
120                    &mut self.symbols,
121                    thread,
122                    sample,
123                    maps,
124                    Arc::clone(&self.context),
125                ),
126                Err(error) => {
127                    let message = format!("ptrace sampling: {error:#}");
128                    thread.ptrace_error = Some(message.clone());
129                    thread.latest.error = Some(match thread.latest.error.take() {
130                        Some(error) => format!("{error}; {message}"),
131                        None => message,
132                    });
133                }
134            }
135        }
136        // Schedule from completion, including failed attempts; never catch up in a burst.
137        self.next_ptrace = Some(Instant::now() + PTRACE_INTERVAL);
138        process::validate_process_identity(self.id)
139    }
140    /// Drains perf samples and captures current threads when the ptrace deadline is due.
141    pub(in crate::privileged_helper::process) fn poll(&mut self, maps: &[MemoryMap]) -> Result<()> {
142        process::validate_process_identity(self.id)?;
143        if self.context.maps.as_slice() != maps {
144            self.context = Arc::new(MappingSnapshot {
145                epoch: self.context.epoch + 1,
146                maps: maps.to_vec(),
147            });
148        }
149        #[cfg(test)]
150        {
151            for map in maps.iter().filter(|map| map.executable && map.inode != 0) {
152                if self.symbols.get(map).is_none()
153                    && let Some(file) = process::elf::read(self.id.pid, map)
154                {
155                    self.symbols.insert(map, file);
156                }
157            }
158            self.unwind.refresh(maps, &self.symbols);
159        }
160        let mut alive = BTreeSet::new();
161        for tid in super::super::threads::tids(self.id.pid)? {
162            let Ok(stat) = procfs::read_stat(&format!("/proc/{}/task/{tid}/stat", self.id.pid))
163            else {
164                continue;
165            };
166            alive.insert(tid);
167            if self
168                .threads
169                .get(&tid)
170                .is_some_and(|thread| thread.start_time != stat.start_time)
171            {
172                self.threads.remove(&tid);
173            }
174            let thread = self.threads.entry(tid).or_insert_with(|| {
175                let mut latest = ThreadSample::waiting(tid, self.executable_header);
176                let event = match PerfEventReader::open(tid) {
177                    Ok(event) => Some(event),
178                    Err(e) => {
179                        latest.error = Some(format!("{e:#}"));
180                        None
181                    }
182                };
183                let switch_event = match PerfEventReader::context_switch(tid) {
184                    Ok(event) => Some(event),
185                    Err(e) => {
186                        let message = format!("context-switch perf: {e:#}");
187                        latest.error = Some(match latest.error.take() {
188                            Some(error) => format!("{error}; {message}"),
189                            None => message,
190                        });
191                        None
192                    }
193                };
194                Thread {
195                    switch_event,
196                    setup_error: latest.error.clone(),
197                    ptrace_error: None,
198                    start_time: stat.start_time,
199                    event,
200                    latest,
201                    sampled_ns: None,
202                }
203            });
204            let mut newest = None;
205            let mut lost = 0u64;
206            thread.latest.error = match (&thread.setup_error, &thread.ptrace_error) {
207                (Some(perf), Some(ptrace)) => Some(format!("{perf}; {ptrace}")),
208                (perf, ptrace) => perf.clone().or_else(|| ptrace.clone()),
209            };
210            for event in [&mut thread.event, &mut thread.switch_event]
211                .into_iter()
212                .flatten()
213            {
214                match event.drain() {
215                    Ok(Some(sample)) if sample.tid == tid => {
216                        if newest
217                            .as_ref()
218                            .is_none_or(|old: &super::sample::RawSample| {
219                                sample.time_ns > old.time_ns
220                            })
221                        {
222                            newest = Some(sample);
223                        }
224                    }
225                    Ok(_) => {}
226                    Err(e) => thread.latest.error = Some(format!("{e:#}")),
227                }
228                lost = lost.saturating_add(event.lost);
229            }
230            thread.latest.lost_samples = lost;
231            if let Some(sample) = newest
232                && thread.sampled_ns.is_none_or(|time| sample.time_ns > time)
233            {
234                process_sample(
235                    self.id.pid,
236                    #[cfg(test)]
237                    &mut self.unwind,
238                    #[cfg(test)]
239                    &mut self.symbols,
240                    thread,
241                    sample,
242                    maps,
243                    Arc::clone(&self.context),
244                );
245            }
246        }
247        self.threads.retain(|tid, _| alive.contains(tid));
248        if self
249            .next_ptrace
250            .is_some_and(|deadline| Instant::now() >= deadline)
251        {
252            self.capture_ptrace(maps)?;
253        }
254        process::validate_process_identity(self.id)?;
255        Ok(())
256    }
257    /// Returns each mapping context referenced by retained thread samples once.
258    pub(in crate::privileged_helper::process) fn mapping_snapshots(&self) -> Vec<MappingSnapshot> {
259        let mut snapshots = BTreeMap::new();
260        for thread in self.threads.values() {
261            if let Some(context) = &thread.latest.context {
262                snapshots
263                    .entry(context.epoch)
264                    .or_insert_with(|| context.as_ref().clone());
265            }
266        }
267        snapshots.into_values().collect()
268    }
269    /// Returns the latest samples with ages refreshed from monotonic time.
270    pub(in crate::privileged_helper::process) fn latest(&self) -> Vec<ThreadSample> {
271        let now = monotonic_ns();
272        self.threads
273            .values()
274            .map(|thread| {
275                let mut latest = thread.latest.clone();
276                latest.sample_age_ms = thread
277                    .sampled_ns
278                    .map(|time| now.saturating_sub(time) / 1_000_000);
279                latest
280            })
281            .collect()
282    }

284fn process_sample(
285    pid: i32,
286    #[cfg(test)] unwind: &mut StackUnwinder,
287    #[cfg(test)] symbols: &mut ElfCache,
288    thread: &mut Thread,
289    sample: super::sample::RawSample,
290    maps: &[MemoryMap],
291    context: Arc<MappingSnapshot>,
292) {
293    if thread.sampled_ns.is_some_and(|time| sample.time_ns <= time) {
294        return;
295    }
296    let executable_header = thread.latest.executable_header;
297    let tid = sample.tid;
298    let lost = thread.latest.lost_samples;
299    let r = &sample.registers;
300    #[cfg(test)]
301    let (frames, reason) = {
302        let (mut frames, reason) = unwind.walk(
303            r.0[native::PC],
304            r.0[native::SP],
305            r.0[native::FP],
306            maps,
307            &sample.stack,
308        );
309        for (index, frame) in frames.iter_mut().enumerate() {
310            let address = instruction_address(frame.address, index > 0);
311            let info = maps
312                .iter()
313                .find(|m| m.contains(address) && m.inode != 0)
314                .and_then(|map| {
315                    let elf = symbols.get(map)?;
316                    resolve_frame(elf_address(address, map, &elf, procfs::page_size())?, &elf)
317                });
318            apply_symbol_info(frame, info);
319        }
320        (frames, reason)
321    };
322    #[cfg(not(test))]
323    let (frames, reason) = (Vec::new(), String::new());
324    let code = InstructionBytes::capture(pid, r.0[native::PC], maps);
325    let age = monotonic_ns().saturating_sub(sample.time_ns) / 1_000_000;
326    thread.sampled_ns = Some(sample.time_ns);
327    thread.latest = ThreadSample {
328        tid,
329        executable_header,
330        sampled_at: Some(process::timestamp_ms().saturating_sub(age)),
331        sample_time_ns: Some(sample.time_ns),
332        #[cfg(test)]
333        sample_source: Some(sample.source),
334        sample_age_ms: Some(age),
335        cpu: sample.cpu,
336        lost_samples: lost,
337        #[cfg(test)]
338        registers: registers::from_sample(r, maps),
339        #[cfg(not(test))]
340        registers: Vec::new(),
341        raw_registers: Some(r.clone()),
342        stack_bytes: sample.stack.clone(),
343        map_epoch: Some(context.epoch),
344        context: Some(context),
345        call_stack: frames,
346        disassembly: Some(code),
347        unwind_stop: reason,
348        error: thread.latest.error.clone(),
349    };
350}

352// Replace all resolved fields so repeated application cannot append inline frames
353// or preserve stale data from an earlier resolution.
354#[cfg(test)]
355fn apply_symbol_info(frame: &mut StackFrame, info: Option<SymbolInfo>) {
356    let info = info.unwrap_or_default();
357    frame.symbol = info.name;
358    frame.symbol_offset = info.offset.map(|offset| format!("0x{offset:x}"));
359    frame.source_file = info.file;
360    frame.line = info.line;
361    frame.inline_frames = info.inline_frames;
362}

462    fn poll_until(sampler: &mut Sampler, maps: &[MemoryMap], condition: impl Fn(&Sampler) -> bool) {
463        let deadline = Instant::now() + Duration::from_secs(10);
464        while Instant::now() < deadline {
465            sampler.poll(maps).unwrap();
466            if condition(sampler) {
467                return;
468            }
469            std::thread::sleep(Duration::from_millis(20));
470        }
471        panic!("no expected live sample: {:?}", sampler.latest());
472    }

706    fn live_blocked_samples_refresh_known_registers_and_bound_deep_stacks() {
707        let target = TestTarget::new("blocked_stack");
708        PerfEventReader::open(target.id.pid)
709            .expect("perf access is required in the Docker test container");
710        PerfEventReader::context_switch(target.id.pid).unwrap();
711        let maps = process::maps::read_maps(target.id.pid).unwrap();
712        let mut sampler = Sampler::new(target.id);
713        let mut complete = false;
714        let mut truncated = false;
715        let mut previous = None;
716        let mut updates = 0;
717        let deadline = Instant::now() + Duration::from_secs(3);
718        while Instant::now() < deadline {
719            sampler.poll(&maps).unwrap();
720            let sample = &sampler.latest()[0];
721            if sample.sample_source == Some(SampleSource::ContextSwitch { preempted: false }) {
722                assert!(sample.error.is_none(), "{:?}", sample.error);
723                for (name, expected) in [
724                    ("R12", 0x12345678u64),
725                    ("R13", 0x23456789),
726                    ("R14", 0x3456789a),
727                    ("R15", 0x456789ab),
728                ] {
729                    let register = sample.registers.iter().find(|r| r.name == name).unwrap();
730                    assert_eq!(register.value, expected);
731                }
732                assert!(
733                    sample
734                        .call_stack
735                        .iter()
736                        .any(|f| f.symbol.as_deref() == Some("blocked_leaf"))
737                );
738                complete |= sample
739                    .call_stack
740                    .iter()
741                    .any(|f| f.symbol.as_deref() == Some("main"));
742                truncated |= sample.unwind_stop.contains("stack snapshot exhausted");
743                if sample.sampled_at != previous {
744                    updates += 1;
745                    previous = sample.sampled_at;
746                }
747            }
748            std::thread::sleep(Duration::from_millis(20));
749        }
750        assert!(
751            complete && truncated && updates >= 5,
752            "complete={complete} truncated={truncated} updates={updates}: {:?}",
753            sampler.latest()
754        );
755        eprintln!(
756            "Blocked fixture: {updates} distinct updates in 3s; R12–R15 matched; complete and bounded deep stacks verified"
757        );
758        target.assert_detached();
759    }

848    fn live_thread_lifecycle_discovers_and_removes_acknowledged_threads() {
849        let mut controller = ThreadController::new();
850        let id = controller.target.id;
851        let maps = process::maps::read_maps(id.pid).unwrap();
852        let mut sampler = Sampler::new(id);
853        sampler.poll(&maps).unwrap();
854        for _ in 0..3 {
855            let tid = controller.command("start");
856            assert_ne!(tid, id.pid);
857            sampler.poll(&maps).unwrap();
858            let thread = sampler
859                .threads
860                .get(&tid)
861                .expect("new worker was not discovered");
862            assert!(thread.event.is_some(), "{:?}", thread.setup_error);
863            assert!(thread.switch_event.is_some(), "{:?}", thread.setup_error);
864            assert_eq!(controller.command("stop"), tid);
865            // pthread_join has completed. Wait only for procfs to drop the task,
866            // without racing an uncontrolled stream of replacement threads.
867            poll_until(&mut sampler, &maps, |s| !s.threads.contains_key(&tid));
868            assert!(!sampler.latest().iter().any(|sample| sample.tid == tid));
869            assert!(sampler.threads.contains_key(&id.pid));
870        }
871        controller.target.assert_detached();
872    }
