1//! Acquire bounded environment bytes after checking privacy and process identity.
2//!
3//! # Interface
4//! - [`read_environment`] (`pub(in crate::privileged_helper)`): `fn read_environment(id: `[`ProcessId`]`) -> `[`anyhow::Result`]`<`[`EnvironmentData`](crate::ipc::acquisition::EnvironmentData)`>`.
5
6use super::procfs::read_bounded_bytes;
7use crate::privileged_helper::timestamp_ms;
8use anyhow::{Result, ensure};
9use std::fs::File;
10use {
11    super::{memory, permission_help, validate_process_id},
12    crate::ipc::schema::ProcessId,
13};
14
15const MAX_ENVIRONMENT: usize = 1024 * 1024;
16
17pub(in crate::privileged_helper) fn read_environment(
18    id: ProcessId,
19) -> Result<crate::ipc::acquisition::EnvironmentData> {
20    ensure!(
21        !memory::privacy_enabled(),
22        "Environment variables are not read in privacy mode."
23    );
24    validate_process_id(id)?;
25    let bytes = read_bounded_bytes(
26        File::open(format!("/proc/{}/environ", id.pid))?,
27        MAX_ENVIRONMENT,
28    )
29    .map_err(|e| anyhow::anyhow!(permission_help("/proc/PID/environ", e)))?;
30    validate_process_id(id)?;
31    Ok(crate::ipc::acquisition::EnvironmentData {
32        process_id: id,
33        captured_at: timestamp_ms(),
34        raw_environment: bytes,
35    })
36}
