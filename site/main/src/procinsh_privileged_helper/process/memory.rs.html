11use super::permission_help;
12use anyhow::{Result, bail, ensure};
13use std::sync::atomic::{AtomicBool, Ordering};
14
15const MAX_READ: usize = 65536;
16
17/// Set once at startup by `--privacy`: no process memory or environment is read.
18static PRIVACY: AtomicBool = AtomicBool::new(false);
19
20pub(in crate::privileged_helper) fn enable_privacy() {
21    PRIVACY.store(true, Ordering::Relaxed);
22}
23
24pub(in crate::privileged_helper) fn privacy_enabled() -> bool {
25    PRIVACY.load(Ordering::Relaxed)
26}
27
28/// Read memory through a kernel task ID, including a non-leader thread TID.
29/// Callers validate the containing process identity around acquisition.
30pub(super) fn read_process_memory(pid: i32, address: u64, length: usize) -> Result<Vec<u8>> {
31    ensure!(
32        !privacy_enabled(),
33        "Process memory is not read in privacy mode."
34    );
35    ensure!(
36        (1..=MAX_READ).contains(&length),
37        "length must be 1..=65536 bytes"
38    );
39    ensure!(
40        address.checked_add(length as u64).is_some(),
41        "address range overflow"
42    );
43    let mut bytes = vec![0u8; length];
44    let local = libc::iovec {
45        iov_base: bytes.as_mut_ptr().cast(),
46        iov_len: length,
47    };
48    let remote = libc::iovec {
49        iov_base: address as *mut libc::c_void,
50        iov_len: length,
51    };
52    // SAFETY: local points at an allocated buffer; the kernel validates remote memory.
53    let count = unsafe { libc::process_vm_readv(pid, &local, 1, &remote, 1, 0) };
54    if count < 0 {
55        bail!(permission_help(
56            "process_vm_readv",
57            std::io::Error::last_os_error()
58        ));
59    }
60    bytes.truncate(count as usize);
61    Ok(bytes)
62}

64#[cfg(test)]
65mod tests {
66    use super::*;
67
68    #[test]
69    fn reads_own_memory_and_rejects_invalid_ranges() {
70        let value = *b"procinsh";
71        let pid = std::process::id() as i32;
72        assert_eq!(
73            read_process_memory(pid, value.as_ptr() as u64, value.len()).unwrap(),
74            value
75        );
76        assert!(read_process_memory(pid, 0, MAX_READ + 1).is_err());
77        assert!(read_process_memory(pid, u64::MAX, 16).is_err());
78    }
